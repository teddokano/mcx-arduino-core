# NXP FRDM MCX Series
# boards.txt

# Tools menus (from 0.13.0). The first option of each is the default, and
# is what a board gets with no option given (arduino-cli, the CI).
# Optimize: -O2 as before; -Os for a sketch that doesn't fit, smaller by
# 1-11% (FRDM-MCXA153: hello_world 39.8KB -> 38.7KB, release_check/01
# 71.7KB -> 64.1KB), a little slower
menu.opt=Optimize
# EEPROM (FRDM-MCXA153 only, the board with the least flash): the EEPROM
# library's 16KB at the top of the flash, or none and all 128KB for the
# sketch. Switching loses what EEPROM kept: the area is the sketch's then
menu.eeprom=EEPROM
# Upload method (the MCX A and N boards, which probe-rs knows): LinkServer,
# NXP's tool, installed by the user, or probe-rs. The choice is for the
# Upload button and the Debug button both: the board's gdb-bridge .cfg says
# which one gdb-bridge runs
menu.upload_method=Upload method

##############################################################
# FRDM-MCXA153
##############################################################

frdm_mcxa153.name=FRDM-MCXA153 (mcx-arduino-core)

# CPU
frdm_mcxa153.build.mcu=cortex-m33+nodsp
frdm_mcxa153.build.fpu_flags=
frdm_mcxa153.build.f_cpu=96000000UL

# Arduino IDE internal identifiers
frdm_mcxa153.build.board=FRDM_MCXA153
frdm_mcxa153.build.core=arduino
frdm_mcxa153.build.variant=frdm_mcxa153

# Board-specific preprocessor defines
frdm_mcxa153.build.board_defines=-DCPU_MCXA153VLH -DCPU_MCXA153VLH_cm33_nodsp -DTARGET_A153 -DFRDM_MCXA153

# Optimization: the default, and Tools > Optimize (menu.opt above)
frdm_mcxa153.build.opt_flags=-O2
frdm_mcxa153.menu.opt.o2=Faster (-O2)
frdm_mcxa153.menu.opt.o2.build.opt_flags=-O2
frdm_mcxa153.menu.opt.os=Smaller (-Os)
frdm_mcxa153.menu.opt.os.build.opt_flags=-Os

# Flash / RAM
frdm_mcxa153.upload.maximum_size=114688
frdm_mcxa153.upload.maximum_data_size=24576

# Linker script filename (inside variants/frdm_mcxa153/linker/)
frdm_mcxa153.build.ldscript=MCXA153.ld

# Tools > EEPROM (menu.eeprom at the top): none gives the sketch the 16KB
# the EEPROM library keeps, and makes #include <EEPROM.h> stop the build
frdm_mcxa153.menu.eeprom.on=1KB in the top 16KB of flash (sketch: 112KB)
frdm_mcxa153.menu.eeprom.off=None (sketch: all 128KB)
frdm_mcxa153.menu.eeprom.off.build.ldscript=MCXA153_noeeprom.ld
frdm_mcxa153.menu.eeprom.off.upload.maximum_size=131072
frdm_mcxa153.menu.eeprom.off.build.eeprom_defines=-DMCX_EEPROM_NONE

# Upload
# build.pyocd_target is currently unused -- no recipe in platform.txt
# references it (only the LinkServer upload path is implemented).
# Kept as forward-looking metadata for if pyOCD upload support is added.
frdm_mcxa153.build.pyocd_target=mcxa153
frdm_mcxa153.build.linkserver_target=MCXA153:FRDM-MCXA153
frdm_mcxa153.upload.tool=linkserver
# Tools > Upload method (menu.upload_method at the top)
frdm_mcxa153.build.probe_rs_chip=MCXA153
frdm_mcxa153.menu.upload_method.linkserver=LinkServer
frdm_mcxa153.menu.upload_method.probe_rs=probe-rs
frdm_mcxa153.menu.upload_method.probe_rs.upload.tool=probe_rs
frdm_mcxa153.menu.upload_method.probe_rs.upload.tool.default=probe_rs
frdm_mcxa153.menu.upload_method.probe_rs.debug.server.openocd.script=a153-probe-rs.cfg
frdm_mcxa153.upload.tool.default=linkserver
frdm_mcxa153.upload.protocol=

# upload_port.0.board is the board's name in the USB product string of its
# MCU-LINK, reported by tools/mcx-discovery (see platform.txt).
frdm_mcxa153.vid.0=0x1FC9
frdm_mcxa153.pid.0=0x0143
frdm_mcxa153.upload_port.0.vid=0x1FC9
frdm_mcxa153.upload_port.0.pid=0x0143
frdm_mcxa153.upload_port.0.board=FRDM-MCXA153

# Debug (see platform.txt's debug.* section and tools/gdb-bridge/src/main.go).
# The launcher and the binaries are board-agnostic; the script below is
# what tells gdb-bridge which board this is.
frdm_mcxa153.debug.server.openocd.path={runtime.platform.path}/tools/gdb-bridge/launch.sh
frdm_mcxa153.debug.server.openocd.path.windows={runtime.platform.path}\tools\gdb-bridge\gdb-bridge-windows-amd64.exe
frdm_mcxa153.debug.server.openocd.script=a153.cfg
frdm_mcxa153.debug.svd_file={runtime.platform.path}/variants/frdm_mcxa153/svd/MCXA153.svd
