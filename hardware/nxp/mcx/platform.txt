name=NXP FRDM MCX Series
version=0.4.0
# version.major/minor/patch mirror the version= line above (split by hand,
# since platform.txt's {} substitution can't parse a dot-separated string
# itself) -- keep these three in lockstep with version= at every release.
# Consumed by compiler.defines below, which turns them into the
# MCX_ARDUINO_CORE_VERSION_MAJOR/_MINOR/_PATCH macros a sketch/library
# sees via mcx_arduino_core_version.h (included from Arduino.h).
version.major=0
version.minor=4
version.patch=0

compiler.path={runtime.tools.arm-none-eabi-gcc.path}/bin/
compiler.cpu_flags=-mcpu={build.mcu} {build.fpu_flags} -mthumb
compiler.defines=-DPRINTF_FLOAT_ENABLE=1 -DSCANF_FLOAT_ENABLE=1 -DPRINTF_ADVANCED_ENABLE=1 -DSCANF_ADVANCED_ENABLE=1 -D__NEWLIB__ -DSDK_OS_BAREMETAL -DSERIAL_PORT_TYPE_UART=1 -DSDK_DEBUGCONSOLE=0 -D__USE_CMSIS -DARDUINO=10819 -DARDUINO_ARCH_MCX -DARDUINO_{build.board} -DF_CPU={build.f_cpu} -DMCX_ARDUINO_CORE_VERSION_MAJOR={version.major} -DMCX_ARDUINO_CORE_VERSION_MINOR={version.minor} -DMCX_ARDUINO_CORE_VERSION_PATCH={version.patch} -DSDK_DELAY_USE_DWT {build.board_defines}
compiler.opt_flags={build.opt_flags} -fno-common -g3 -gdwarf-4 -ffunction-sections -fdata-sections -fno-builtin -fmerge-constants -specs=nano.specs
compiler.warning_flags.none=
compiler.warning_flags.default=-Wall
compiler.warning_flags.more=-Wall
compiler.warning_flags.all=-Wall -Wextra
compiler.includes=-I{build.core.path} -I{build.core.path}/sdk -I{build.core.path}/r01lib -I{build.core.path}/arduino_api -I{build.variant.path}/include

compiler.c.cmd=arm-none-eabi-gcc
compiler.c.flags={compiler.cpu_flags} {compiler.defines} {compiler.opt_flags} {compiler.warning_flags} -Wno-unused-parameter -Wno-missing-field-initializers -Wno-old-style-declaration -c -D__NEWLIB__ -fstack-usage
recipe.c.o.pattern="{compiler.path}{compiler.c.cmd}" {compiler.c.flags} {compiler.includes} {includes} -o "{object_file}" "{source_file}"

compiler.cpp.cmd=arm-none-eabi-c++
compiler.cpp.flags={compiler.cpu_flags} {compiler.defines} {compiler.opt_flags} {compiler.warning_flags} -Wno-address-of-packed-member -Wno-unused-parameter -Wno-missing-field-initializers -std=c++20 -c -fno-rtti -fno-exceptions -D__NEWLIB__ -fstack-usage
recipe.cpp.o.pattern="{compiler.path}{compiler.cpp.cmd}" {compiler.cpp.flags} {compiler.includes} {includes} -o "{object_file}" "{source_file}"

compiler.S.cmd=arm-none-eabi-gcc
compiler.S.flags={compiler.cpu_flags} -c -x assembler-with-cpp
recipe.S.o.pattern="{compiler.path}{compiler.S.cmd}" {compiler.S.flags} {compiler.includes} {includes} -o "{object_file}" "{source_file}"

compiler.ar.cmd=arm-none-eabi-ar
compiler.ar.flags=rcs
recipe.ar.pattern="{compiler.path}{compiler.ar.cmd}" {compiler.ar.flags} "{archive_file_path}" "{object_file}"

compiler.c.elf.cmd=arm-none-eabi-c++
compiler.c.elf.flags=-nostdlib -Xlinker --gc-sections -Xlinker -print-memory-usage -Xlinker --sort-section=alignment -u _printf_float -u _scanf_float {compiler.cpu_flags}
recipe.c.combine.pattern="{compiler.path}{compiler.c.elf.cmd}" {compiler.c.elf.flags} -T "{build.variant.path}/linker/{build.ldscript}" -o "{build.path}/{build.project_name}.elf" {object_files} -Wl,--whole-archive "{archive_file_path}" -Wl,--no-whole-archive -lc -lm -lgcc

compiler.objcopy.cmd=arm-none-eabi-objcopy
compiler.elf2hex.cmd=arm-none-eabi-objcopy
compiler.elf2hex.flags=-O binary
recipe.objcopy.bin.pattern="{compiler.path}{compiler.elf2hex.cmd}" {compiler.elf2hex.flags} "{build.path}/{build.project_name}.elf" "{build.path}/{build.project_name}.bin"

compiler.size.cmd=arm-none-eabi-size
recipe.size.pattern="{compiler.path}{compiler.size.cmd}" -A "{build.path}/{build.project_name}.elf"
recipe.size.regex=^(?:\.text|\.data|\.rodata)\s+([0-9]+).*
recipe.size.regex.data=^(?:\.data|\.bss)\s+([0-9]+).*

# Upload
tools.linkserver.upload.params.verbose=
tools.linkserver.upload.params.quiet=
tools.linkserver.upload.pattern={runtime.platform.path}/tools/upload.sh "{build.path}/{build.project_name}.elf" "{build.linkserver_target}"
tools.linkserver.upload.pattern.windows={runtime.platform.path}/tools/upload.bat "{build.path}/{build.project_name}.elf" "{build.linkserver_target}"

upload.tool=linkserver
upload.tool.default=linkserver

# Debug
# arduino-cli's debug support only knows how to drive something it calls
# "openocd" -- the path to that executable is configurable per board
# (boards.txt sets debug.server.openocd.path), but the string "openocd" in
# debug.server itself is not. The path actually points at tools/gdb-bridge,
# which relays gdb to NXP LinkServer's own gdbserver instead of running real
# OpenOCD (which has no MCX chip support). See
# tools/gdb-bridge/src/main.go for the full explanation.
debug.executable={build.path}/{build.project_name}.elf
debug.toolchain=gcc
debug.toolchain.path={runtime.tools.arm-none-eabi-gcc.path}/bin/
debug.toolchain.prefix=arm-none-eabi-
debug.server=openocd
# Arduino IDE 2's debug UI won't start a session without at least one
# "OpenOCD configuration file" configured, even though gdb-bridge doesn't
# actually run OpenOCD and never reads this file -- see dummy-openocd.cfg.
debug.server.openocd.scripts_dir={runtime.platform.path}/tools/gdb-bridge/
debug.server.openocd.script=dummy-openocd.cfg
