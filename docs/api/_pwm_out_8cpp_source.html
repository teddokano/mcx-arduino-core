97PwmOut::PwmOut( int pin )
98 : Obj( true ), _submodule( 0 ), _channel( 0 ), _pin( -1 ), _period_us( 20000u ), _pulse_us( 0u )
99{
100 resolve_pin( pin );
101
102 if ( -1 == _pin )
103 {
104 panic( "PwmOut: unsupported PWM pin" );
105 return;
106 }
107
108 _acquire_module();
109 CLOCK_EnableClock( s_sm_clock[ _submodule ] );
110
111 {
112 DigitalInOut pin_io( (uint8_t)pin );
113 pin_io.pin_mux( (int)kPORT_MuxAlt5 ); // FlexPWM0 function on this pin
114 }
115
116 if ( !s_sm_init[ _submodule ] )
117 {
118 pwm_config_t cfg;
119 PWM_GetDefaultConfig( &cfg );
120 cfg.pairOperation = kPWM_Independent; // chA/chB run independently (duty), period still shared by HW
121 cfg.enableDebugMode = true;
122 PWM_Init( FLEXPWM0, (pwm_submodule_t)_submodule, &cfg );
123
124 // Out of reset every FAULTx input is mapped to disable this submodule's
125 // outputs (SDK doc: "a reset sets all bits in this field"). No fault
126 // pins are wired on this design, so clear the map for both channels.
127 PWM_SetupFaultDisableMap( FLEXPWM0, (pwm_submodule_t)_submodule, kPWM_PwmA, kPWM_faultchannel_0, 0u );
128 PWM_SetupFaultDisableMap( FLEXPWM0, (pwm_submodule_t)_submodule, kPWM_PwmB, kPWM_faultchannel_0, 0u );
129
130 s_sm_init[ _submodule ] = true;
131 }
132
133 // One-time setup of OUTEN/polarity/fault-state for this channel. The
134 // frequency passed here is a throwaway placeholder (safe at the default
135 // prescale of /1) — apply() immediately below overwrites the period/duty
136 // registers with the real default (20ms / 0%) at a correctly chosen
137 // prescaler, so the value used here does not need to match _period_us.
138 pwm_signal_param_t sig;
139 sig.pwmChannel = sdk_channel( _channel );
140 sig.level = kPWM_HighTrue;
141 sig.dutyCyclePercent = 0;
142 sig.deadtimeValue = 0;
143 sig.faultState = kPWM_PwmFaultState0;
144 sig.pwmchannelenable = true;
145 PWM_SetupPwm( FLEXPWM0, (pwm_submodule_t)_submodule, &sig, 1, kPWM_EdgeAligned, 1000u,
146 CLOCK_GetFreq( kCLOCK_MainClk ) );
147
148 apply(); // sets the real default period/duty and calls PWM_SetPwmLdok()
149 PWM_StartTimer( FLEXPWM0, (uint8_t)( 1u << _submodule ) );
150}

163void PwmOut::apply( void )
164{
165 if ( -1 == _pin )
166 return;
167
168 uint32_t src_clk = CLOCK_GetFreq( kCLOCK_MainClk );
169
170 // Pick the smallest prescaler (highest resolution) for which the period
171 // still fits the 16-bit tick counter, clamping to the fastest/slowest
172 // achievable period at the extremes (memo: clamp to the actual range
173 // the FlexPWM prescaler/clock can produce).
174 uint8_t prescale = 0;
175 uint64_t pulseCnt64 = 0;
176 for ( prescale = 0; prescale <= 7; prescale++ )
177 {
178 pulseCnt64 = ( (uint64_t)_period_us * ( src_clk >> prescale ) ) / 1000000ULL;
179 if ( pulseCnt64 <= 0xFFFFu || prescale == 7 )
180 break;
181 }
182 if ( pulseCnt64 < 1 )
183 pulseCnt64 = 1;
184 if ( pulseCnt64 > 0xFFFFu )
185 pulseCnt64 = 0xFFFFu;
186
187 uint16_t pulseCnt = (uint16_t)pulseCnt64;
188
189 uint16_t cur_prsc = (uint16_t)( ( FLEXPWM0->SM[ _submodule ].CTRL & PWM_CTRL_PRSC_MASK ) >> PWM_CTRL_PRSC_SHIFT );
190 if ( cur_prsc != prescale )
191 PWM_SetClockMode( FLEXPWM0, (pwm_submodule_t)_submodule, (pwm_clock_prescale_t)prescale );
192
193 uint64_t dutyTicks = ( (uint64_t)_pulse_us * ( src_clk >> prescale ) ) / 1000000ULL;
194 if ( dutyTicks > pulseCnt )
195 dutyTicks = pulseCnt;
196 uint16_t duty16 = pulseCnt ? (uint16_t)( ( dutyTicks * 65535ULL ) / pulseCnt ) : 0u;
197
198 PWM_UpdatePwmPeriodAndDutycycle( FLEXPWM0, (pwm_submodule_t)_submodule, sdk_channel( _channel ),
199 kPWM_EdgeAligned, pulseCnt, duty16 );
200 PWM_SetPwmLdok( FLEXPWM0, (uint8_t)( 1u << _submodule ), true );
201}

203void PwmOut::period( float seconds )
204{
205 if ( -1 == _pin )
206 return;
207
208 uint32_t src_clk = CLOCK_GetFreq( kCLOCK_MainClk );
209 float min_period = 1.0e6f / (float)src_clk; // 1 tick @ prescale/1
210 float max_period = 1.0e6f * 65535.0f * 128.0f / (float)src_clk; // 65535 ticks @ prescale/128
211 float period_us = seconds * 1.0e6f;
212
213 if ( period_us < min_period )
214 period_us = min_period;
215 if ( period_us > max_period )
216 period_us = max_period;
217
218 _period_us = (uint32_t)period_us;
219 if ( _pulse_us > _period_us )
220 _pulse_us = _period_us;
221
222 apply();
223}

292struct PwmPinDescriptor {
293 int pin; // io.h logical pin (PWM0..PWM5)
294 uint32_t port_pin; // PORT2 pin number
295 uint8_t submodule; // 0=sm0, 1=sm1, 2=sm2 (this chip's FlexPWM1)
296 uint8_t channel; // 0=chA, 1=chB
297 uint8_t alt; // PORT mux ALT value (not uniform across pins here)
298};
299
300// Physical pins are P2_2..P2_7 on FlexPWM1 (see io.h for why -- P3_6..P3_11
301// used earlier were unrouted test points). PWM0..PWM5 assignment below
302// matches the "PWM0".."PWM5" silkscreen labels on the Arduino Shield
303// Compatible Headers schematic sheet (FRDM-MCXN947SH.pdf page 12), which is
304// why it isn't in physical pin order.
305//
306// ALT is uniformly 5 for all six pins -- confirmed against Zephyr's
307// silicon-derived pinctrl header (MCXN947VDF-pinctrl.h, N9X_MUX(port,pin,
308// mux) macros), not derived by counting position in pin_mux.c's
309// pin_signal string. That position-counting method (used successfully
310// elsewhere in this codebase, e.g. I3C1_SDA's Alt10) breaks down for
311// P2_2/P2_3 specifically: their pin_signal lists have an extra "CLKOUT"
312// entry ahead of PWM1_A2/B2 that doesn't actually consume a mux slot, so
313// counting gave Alt6/Alt4 -- both wrong, and both silently produced no
314// PWM output at all on real hardware (caught via logic analyzer, no
315// signal on any probed pin) before this fix.
316const PwmPinDescriptor s_pins[] = {
317 // pin pin# sm ch alt
318 { PWM0, 3u, 2u, 1u, 5u }, // P2_3, PWM1_B2
319 { PWM1, 2u, 2u, 0u, 5u }, // P2_2, PWM1_A2
320 { PWM2, 5u, 1u, 1u, 5u }, // P2_5, PWM1_B1
321 { PWM3, 4u, 1u, 0u, 5u }, // P2_4, PWM1_A1
322 { PWM4, 7u, 0u, 1u, 5u }, // P2_7, PWM1_B0
323 { PWM5, 6u, 0u, 0u, 5u }, // P2_6, PWM1_A0
324};
