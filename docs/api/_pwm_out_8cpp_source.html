183// Per-submodule state, shared by the PwmOut on each of its two channels.
184// Times are in ticks of the source clock, before the prescaler, so the
185// duty has as many steps as the hardware can make at any frequency
186struct SubmoduleState {
187 bool init; // PWM_Init() and the fault map done
188 uint8_t users; // bit 0 = chA has a PwmOut, bit 1 = chB
189 uint32_t period_ticks;
190 uint32_t pulse_ticks[ 2 ];
191};
192
193SubmoduleState s_sm[ PWM_MODULES ][ 3 ] = {};
194uint8_t s_module_users[ PWM_MODULES ] = {};

196constexpr uint32_t MIN_TICKS = 1u; // 1 tick @ prescale/1
197constexpr uint32_t MAX_TICKS = 0xFFFFu << 7; // 65535 ticks @ prescale/128

209// Source-clock ticks counted with a prescaler of 2^prescale, rounded
210inline uint32_t ticks_at( uint32_t ticks, uint8_t prescale )
211{
212 return (uint32_t)( ( (uint64_t)ticks + ( ( 1u << prescale ) >> 1 ) ) >> prescale );
213}

215uint32_t us_to_ticks( int64_t us )
216{
217 if ( us <= 0 )
218 return 0u;
219
220 uint64_t ticks = ( (uint64_t)us * source_clock() + 500000u ) / 1000000u;
221
222 return ( ticks > 0xFFFFFFFFu ) ? 0xFFFFFFFFu : (uint32_t)ticks;
223}
224
225uint32_t seconds_to_ticks( float seconds )
226{
227 float ticks = seconds * (float)source_clock() + 0.5f;
228
229 if ( !( ticks >= 1.0f ) ) // NaN too
230 return 0u;
231 if ( ticks >= 4294967040.0f ) // the largest float below 2^32
232 return 0xFFFFFFFFu;
233
234 return (uint32_t)ticks;
235}

261PwmOut::PwmOut( int pin )
262 : Obj( true ), _module( 0 ), _submodule( 0 ), _channel( 0 ), _alt( 0 ), _pin( -1 )
263{
264 const PwmPinDescriptor *d = find_pin( pin );
265
266 if ( nullptr == d )
267 {
268 panic( "PwmOut: unsupported PWM pin" );
269 return;
270 }
271
272 _module = d->module;
273 _submodule = d->submodule;
274 _channel = d->channel;
275 _alt = d->alt;
276 _pin = pin;
277
278 PWM_Type *base = s_module[ _module ];
279 SubmoduleState &sm = s_sm[ _module ][ _submodule ];
280
281 if ( 0 == s_module_users[ _module ]++ )
282 RESET_ReleasePeripheralReset( s_reset[ _module ] );
283
284 CLOCK_EnableClock( s_sm_clock[ _module ][ _submodule ] );
285
286 {
287 DigitalInOut pin_io( (uint8_t)pin );
288 pin_io.pin_mux( (int)_alt );
289 }
290
291 if ( !sm.init )
292 {
293 pwm_config_t cfg;
294 PWM_GetDefaultConfig( &cfg );
295 cfg.pairOperation = kPWM_Independent; // chA/chB run independently (duty), period still shared by HW
296 cfg.enableDebugMode = true;
297 PWM_Init( base, (pwm_submodule_t)_submodule, &cfg );
298
299 PWM_SetupFaultDisableMap( base, (pwm_submodule_t)_submodule, kPWM_PwmA, kPWM_faultchannel_0, 0u );
300 PWM_SetupFaultDisableMap( base, (pwm_submodule_t)_submodule, kPWM_PwmB, kPWM_faultchannel_0, 0u );
301
302 sm.init = true;
303 }
304
305 // The first PwmOut on the submodule sets the Mbed default of 20ms; one
306 // made next to a running channel takes over its period
307 if ( !sm.users )
308 sm.period_ticks = us_to_ticks( 20000 );
309
310 sm.users |= (uint8_t)( 1u << _channel );
311 sm.pulse_ticks[ _channel ] = 0u;
312
313 // One-time setup of OUTEN/polarity/fault-state for this channel. The
314 // period it writes is a placeholder that apply() replaces before LDOK.
315 pwm_signal_param_t sig;
316 sig.pwmChannel = sdk_channel( _channel );
317 sig.level = kPWM_HighTrue;
318 sig.dutyCyclePercent = 0;
319 sig.deadtimeValue = 0;
320 sig.faultState = kPWM_PwmFaultState0;
321 sig.pwmchannelenable = true;
322 PWM_SetupPwm( base, (pwm_submodule_t)_submodule, &sig, 1, kPWM_EdgeAligned, 1000u,
323 CLOCK_GetFreq( PWM_SOURCE_CLOCK ) );
324
325 apply(); // the real period and duty, then PWM_SetPwmLdok()
326 PWM_StartTimer( base, (uint8_t)( 1u << _submodule ) );
327
328 uint8_t pin8 = (uint8_t)_pin;
329 pin_registry_note( this, "PwmOut", &pin8, 1, _alt );
330}

367void PwmOut::apply( void )
368{
369 if ( -1 == _pin )
370 return;
371
372 PWM_Type *base = s_module[ _module ];
373 SubmoduleState &sm = s_sm[ _module ][ _submodule ];
374
375 // Pick the smallest prescaler (highest resolution) for which the period
376 // still fits the 16-bit tick counter. period() keeps it within
377 // MIN_TICKS..MAX_TICKS, so the count is 1..0xFFFF.
378 uint8_t prescale = 0;
379 uint32_t count = 0;
380 for ( prescale = 0; prescale <= 7; prescale++ )
381 {
382 count = ticks_at( sm.period_ticks, prescale );
383 if ( count <= 0xFFFFu || prescale == 7 )
384 break;
385 }
386 if ( count < 1 )
387 count = 1;
388 if ( count > 0xFFFFu )
389 count = 0xFFFFu;
390
391 uint16_t cur_prsc = (uint16_t)( ( base->SM[ _submodule ].CTRL & PWM_CTRL_PRSC_MASK ) >> PWM_CTRL_PRSC_SHIFT );
392 if ( cur_prsc != prescale )
393 PWM_SetClockMode( base, (pwm_submodule_t)_submodule, (pwm_clock_prescale_t)prescale );
394
395 // Edge-aligned, as PWM_SetupPwm() set the channel up: the counter runs
396 // from INIT = 0 to VAL1, and a channel is high from VAL2 (A) or VAL4
397 // (B), 0, until VAL3 or VAL5. A pulse as long as the period never
398 // reaches its VAL3/VAL5 and stays high. These are the registers the
399 // SDK's PWM_UpdatePwmPeriodAndDutycycle() writes, here in ticks: that
400 // function takes the duty as a 16-bit fraction and turns it back into
401 // ticks, which loses steps. Like it, the values wait for LDOK.
402 base->SM[ _submodule ].INIT = 0u;
403 base->SM[ _submodule ].VAL0 = (uint16_t)( count / 2u );
404 base->SM[ _submodule ].VAL1 = (uint16_t)( count - 1u );
405
406 // Both channels, so the one this PwmOut isn't on keeps its pulse width
407 // against the period it now shares
408 for ( uint8_t ch = 0; ch < 2; ch++ )
409 {
410 if ( !( sm.users & ( 1u << ch ) ) )
411 continue;
412
413 uint32_t high = ticks_at( sm.pulse_ticks[ ch ], prescale );
414 if ( sm.pulse_ticks[ ch ] >= sm.period_ticks || high > count )
415 high = count;
416
417 if ( ch == 0 )
418 {
419 base->SM[ _submodule ].VAL2 = 0u;
420 base->SM[ _submodule ].VAL3 = (uint16_t)high;
421 }
422 else
423 {
424 base->SM[ _submodule ].VAL4 = 0u;
425 base->SM[ _submodule ].VAL5 = (uint16_t)high;
426 }
427 }
428 PWM_SetPwmLdok( base, (uint8_t)( 1u << _submodule ), true );
429}

431void PwmOut::set_period( uint32_t ticks )
432{
433 if ( -1 == _pin )
434 return;
435
436 SubmoduleState &sm = s_sm[ _module ][ _submodule ];
437
438 if ( ticks < MIN_TICKS )
439 ticks = MIN_TICKS;
440 if ( ticks > MAX_TICKS )
441 ticks = MAX_TICKS;
442
443 sm.period_ticks = ticks;
444 for ( uint8_t ch = 0; ch < 2; ch++ )
445 {
446 if ( sm.pulse_ticks[ ch ] > sm.period_ticks )
447 sm.pulse_ticks[ ch ] = sm.period_ticks;
448 }
449
450 apply();
451}

609constexpr uint32_t MAX_TICKS = 0xFFFFu << 7; // 65535 ticks @ prescale/128
610
611// Source-clock ticks counted with a prescaler of 2^prescale, rounded
612inline uint32_t ticks_at( uint32_t ticks, uint8_t prescale )
613{
614 return (uint32_t)( ( (uint64_t)ticks + ( ( 1u << prescale ) >> 1 ) ) >> prescale );
615}
616
617uint32_t us_to_ticks( int64_t us )
618{
619 if ( us <= 0 )
620 return 0u;
621
622 uint64_t ticks = ( (uint64_t)us * source_clock() + 500000u ) / 1000000u;
623
624 return ( ticks > 0xFFFFFFFFu ) ? 0xFFFFFFFFu : (uint32_t)ticks;
625}
626
627uint32_t seconds_to_ticks( float seconds )
628{
629 float ticks = seconds * (float)source_clock() + 0.5f;
630
631 if ( !( ticks >= 1.0f ) ) // NaN too
632 return 0u;
633 if ( ticks >= 4294967040.0f ) // the largest float below 2^32
634 return 0xFFFFFFFFu;
635
636 return (uint32_t)ticks;
637}

639// At most how long until the counter next goes from MOD to 0, when the
640// buffers hand their values on, in microseconds, rounded up. An upper
641// bound, not a measurement: MOD reads back as last written, which counts
642// only from the next overflow, so for a period after a MOD write the old
643// one may still be running; and CNT, read just after it was cleared with
644// the counter stopped, can still show the old count. A count above MOD
645// runs on to 0xFFFF and wraps first. Getting this short leaves a write in
646// a buffer that is still taken, and the TPM ignores it
647uint32_t us_to_overflow( int t, uint32_t now )
648{
649 TPM_Type *base = s_tpm[ t ];
650 TpmState &st = s_state[ t ];
651 uint32_t ps = ( base->SC & TPM_SC_PS_MASK ) >> TPM_SC_PS_SHIFT;
652 uint32_t mod = st.mod;
653 uint32_t cnt = base->CNT;
654
655 if ( (int32_t)( now - st.mod_before_until ) < 0 && st.mod_before > mod )
656 mod = st.mod_before;
657
658 uint32_t ticks = ( cnt <= mod ) ? mod + 1u - cnt : 0x10000u - cnt + mod + 1u;
659
660 return ( ( ticks << ps ) + ( source_clock() / 1000000u ) - 1u ) / ( source_clock() / 1000000u );
661}

663void write_cnv( int t, int ch, uint16_t value, uint32_t now )
664{
665 s_tpm[ t ]->CONTROLS[ ch ].CnV = value;
666 s_state[ t ].cnv[ ch ] = value;
667 s_state[ t ].free_at[ ch ] = now + us_to_overflow( t, now ) + 2u;

670void write_mod( int t, uint32_t value, uint32_t now )
671{
672 TpmState &st = s_state[ t ];
673 uint32_t until = now + us_to_overflow( t, now ) + 2u;
674
675 s_tpm[ t ]->MOD = value;
676 st.mod_before = st.mod;
677 st.mod_before_until = until;
678 st.mod = value;

682// MOD is buffered as CnV is (below), and the same way round
683void set_mod( int t, uint32_t value, uint32_t now )
684{
685 TPM_Type *base = s_tpm[ t ];
686 TpmState &st = s_state[ t ];
687 uint32_t primask = __get_PRIMASK();
688
689 __disable_irq();
690
691 if ( !( base->SC & TPM_SC_CMOD_MASK ) )
692 {
693 // Stopped: the register takes the value at once
694 base->MOD = value;
695 st.mod = value;
696 st.mod_free_at = now;
697 st.mod_before = value;
698 st.mod_before_until = now;
699 st.waiting &= (uint8_t)~MOD_BIT;
700 }
701 else if ( st.waiting & MOD_BIT )
702 {
703 st.mod_later = value; // replaces the value waiting
704 }
705 else if ( value == st.mod )
706 {
707 // already there, or on its way
708 }
709 else if ( (int32_t)( now - st.mod_free_at ) < 0 )
710 {
711 st.mod_later = value;
712 st.waiting |= MOD_BIT;
713 s_any_waiting = true;
714 }
715 else
716 {
717 write_mod( t, value, now );
718 }
719
720 __set_PRIMASK( primask );
721}
722
723// While the counter runs, a CnV write waits in a buffer for the end of the
724// period, and further writes to that buffer before then are ignored
725// (reference manual, TPMx_CnV; seen on the board too). The TPM's own flag
726// for the end of a period, TOF, could not be cleared reliably on the board,
727// so the time the buffer frees up is worked out from the counter instead,
728// and a value for a channel whose buffer is still taken waits here until
729// pwm_tpm_service() (SysTick, every millisecond) writes it. analogWrite()
730// never waits, and the last value always lands
731void set_cnv( int t, int ch, uint16_t value, uint32_t now )
732{
733 TPM_Type *base = s_tpm[ t ];
734 TpmState &st = s_state[ t ];
735 uint8_t bit = (uint8_t)( 1u << ch );
736 uint32_t primask = __get_PRIMASK();
737
738 __disable_irq();
739
740 if ( !( base->SC & TPM_SC_CMOD_MASK ) )
741 {
742 // Stopped: the register takes the value at once
743 base->CONTROLS[ ch ].CnV = value;
744 st.cnv[ ch ] = value;
745 st.free_at[ ch ] = now;
746 st.waiting &= (uint8_t)~bit;
747 }
748 else if ( st.waiting & bit )
749 {
750 st.later[ ch ] = value; // replaces the value waiting
751 }
752 else if ( value == st.cnv[ ch ] )
753 {
754 // already there, or on its way
755 }
756 else if ( (int32_t)( now - st.free_at[ ch ] ) < 0 )
757 {
758 st.later[ ch ] = value;
759 st.waiting |= bit;
760 s_any_waiting = true;
761 }
762 else
763 {
764 write_cnv( t, ch, value, now );
765 }
766
767 __set_PRIMASK( primask );
768}

772void pwm_tpm_service( uint32_t now )
773{
774 if ( !s_any_waiting )
775 return;
776
777 bool still = false;
778
779 for ( int t = 0; t < TPMS; t++ )
780 {
781 TpmState &st = s_state[ t ];
782
783 if ( st.waiting & MOD_BIT )
784 {
785 if ( (int32_t)( now - st.mod_free_at ) >= 0 )
786 {
787 write_mod( t, st.mod_later, now );
788 st.waiting &= (uint8_t)~MOD_BIT;
789 }
790 else
791 still = true;
792 }
793
794 for ( int ch = 0; ch < CHANNELS; ch++ )
795 {
796 uint8_t bit = (uint8_t)( 1u << ch );
797
798 if ( !( st.waiting & bit ) )
799 continue;
800
801 if ( (int32_t)( now - st.free_at[ ch ] ) >= 0 )
802 {
803 write_cnv( t, ch, st.later[ ch ], now );
804 st.waiting &= (uint8_t)~bit;
805 }
806 else
807 still = true;
808 }
809 }
810
811 s_any_waiting = still;
812}

879void PwmOut::take_channel( void )
880{
881 TpmState &st = s_state[ _tpm ];
882 PwmOut *prev = st.owner[ _channel ];

924void PwmOut::apply( void )
925{
926 if ( -1 == _pin )
927 return;
928
929 TPM_Type *base = s_tpm[ _tpm ];
930 TpmState &st = s_state[ _tpm ];
931 uint32_t now = (uint32_t)micros();
932
933 // The smallest prescaler (highest resolution) for which the period
934 // still fits the 16-bit counter. At most 0xFFFF ticks, so MOD stays
935 // below 0xFFFF: a 100% pulse needs CnV > MOD, and CnV is 16 bits too
936 uint8_t prescale = 0;
937 uint32_t ticks = 0;
938 for ( prescale = 0; prescale <= 7; prescale++ )
939 {
940 ticks = ticks_at( st.period_ticks, prescale );
941 if ( ticks <= 0xFFFFu || prescale == 7 )
942 break;
943 }
944 if ( ticks < 2 )
945 ticks = 2;
946 if ( ticks > 0xFFFFu )
947 ticks = 0xFFFFu;
948
949 // The prescaler can only change with the counter stopped. Stopped, the
950 // registers take MOD and CnV at once, so nothing is left waiting
951 uint32_t cur_ps = ( base->SC & TPM_SC_PS_MASK ) >> TPM_SC_PS_SHIFT;
952 if ( cur_ps != prescale || !( base->SC & TPM_SC_CMOD_MASK ) )
953 {
954 TPM_StopTimer( base );
955 base->SC = ( base->SC & ~( TPM_SC_PS_MASK | TPM_SC_TOF_MASK ) ) | TPM_SC_PS( prescale );
956 base->CNT = 0u;
957 st.waiting = 0u;
958 }
959
960 set_mod( _tpm, ticks - 1u, now );
961
962 for ( int ch = 0; ch < CHANNELS; ch++ )
963 {
964 PwmOut *o = st.owner[ ch ];
965
966 if ( !o )
967 continue;
968
969 // A pulse as long as the period needs CnV = MOD + 1 to stay high
970 uint32_t cnv = ticks_at( o->_pulse_ticks, prescale );
971 if ( o->_pulse_ticks >= st.period_ticks || cnv > ticks )
972 cnv = ticks;
973
974 set_cnv( _tpm, ch, (uint16_t)cnv, now );
975 }
976
977 if ( !( base->SC & TPM_SC_CMOD_MASK ) )
978 TPM_StartTimer( base, kTPM_SystemClock );
979}

984 return;
