2 * MCXEthernet: EthernetServer
3 *
4 * A connection made to the server takes a TCP slot in lwIP's accept callback and
5 * stays tied to the server's port until accept() hands it out, as in the official
6 * library: available() returns the tied ones that have data, and frees the ones the
7 * peer has closed with nothing left to read.

14static err_t on_accept( void *arg, struct tcp_pcb *pcb, err_t err )
15{
16 if ( err != ERR_OK || !pcb )
17 return ERR_VAL;
18
19 uint16_t port = (uint16_t)(uintptr_t)arg;
20 if ( tcp_slot_take( pcb, TCP_CONNECTED, port ) < 0 )
21 {
22 tcp_abort( pcb );
23 return ERR_ABRT;
24 }
25 return ERR_OK;
26}

28void EthernetServer::begin()
29{
30 if ( _listening || !started() )
31 return;
32
33 Lock lock;
34 struct tcp_pcb *pcb = tcp_new();
35 if ( !pcb )
36 return;
37
38 ip_set_option( pcb, SOF_REUSEADDR );
39 if ( tcp_bind( pcb, IP_ANY_TYPE, _port ) != ERR_OK )
40 {
41 tcp_abort( pcb );
42 return;
43 }
44
45 struct tcp_pcb *listen = tcp_listen( pcb ); // frees pcb on success
46 if ( !listen )
47 {
48 tcp_abort( pcb );
49 return;
50 }
51 tcp_arg( listen, (void *)(uintptr_t)_port );
52 tcp_accept( listen, on_accept );
53 _listening = true;
54}

75EthernetClient EthernetServer::accept()
76{
77 Lock lock;
78
79 // as the official library: open connections only, each once; reset ones are dropped
80 for ( int i = 0; i < MAX_SOCK_NUM; i++ )
81 {
82 TcpSlot *s = &tcp_slots[ i ];
83 if ( s->state == TCP_FREE || s->server_port != _port )
84 continue;
85
86 if ( s->state == TCP_CONNECTED || s->state == TCP_PEER_CLOSED )
87 {
88 s->server_port = 0;
89 return EthernetClient( i );
90 }
91 tcp_slot_release( i );
92 }
93 return EthernetClient();
94}
