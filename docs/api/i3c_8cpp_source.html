51{
52#ifdef CPU_MCXN947VDF
53 if ( (sda == I3C_SDA) && (scl == I3C_SCL) )
54 ;
55 else
56 panic( "FRDM-MCXN947 only support I3C_SDA/I3C_SCL pins for I3C" );
57#elif CPU_MCXN236VDF
58 if ( (sda == I3C_SDA) && (scl == I3C_SCL) )
59 ;
60 else
61 panic( "FRDM-MCXN236 only support I3C_SDA/I3C_SCL pins for I3C" );
62#elif CPU_MCXA156VLL
63 if ( (sda == I3C_SDA) && (scl == I3C_SCL) )
64 ;
65 else if ( (sda == I2C_SDA) && (scl == I2C_SCL) )
66 ;
67 else
68 panic( "FRDM-MCXA153 supports I3C_SDA/I3C_SCL or I2C_SDA(D18)/I2C_SCL(D19) pins for I3C" );
69#elif CPU_MCXA153VLH
70 if ( (sda == I3C_SDA) && (scl == I3C_SCL) )
71 ;
72 else if ( (sda == I2C_SDA) && (scl == I2C_SCL) )
73 ;
74 else
75 panic( "FRDM-MCXA153 supports I3C_SDA/I3C_SCL or I2C_SDA(D18)/I2C_SCL(D19) pins for I3C" );
76#else
77 #error Target CPU is not supported
78#endif // CPU_MCXN947VDF

80 I3C_MasterGetDefaultConfig( &masterConfig );
81
82 masterConfig.baudRate_Hz.i2cBaud = i2c_freq ? i2c_freq : I2C::FREQ;
83 masterConfig.baudRate_Hz.i3cOpenDrainBaud = i3c_od_freq ? i3c_od_freq : OD_FREQ;
84 masterConfig.baudRate_Hz.i3cPushPullBaud = i3c_pp_freq ? i3c_pp_freq : PP_FREQ;
85 masterConfig.enableOpenDrainStop = false;
86 masterConfig.disableTimeout = true;

91
92 /* Create I3C handle. */
93 I3C_MasterTransferCreateHandle( EXAMPLE_MASTER, &g_i3c_m_handle, &masterCallback, NULL );
94
95 first_broadcast = true;
96
97 DigitalInOut _scl( sda );
98 DigitalInOut _sda( scl );
99
100 _scl.pin_mux( kPORT_MuxAlt10 );
101 _sda.pin_mux( kPORT_MuxAlt10 );

103 /*
104 * On FRDM-MCXN947, the pin_mux.c code that sets PORT1_16/PORT1_17's
105 * input buffer enable (IBE) bit for I3C1_SDA/I3C1_SCL only exists
106 * inside BOARD_InitDEBUG_UARTPins() -- a generated function that is
107 * never actually called anywhere (init_mcu() only calls
108 * BOARD_InitBootPins()/BOARD_InitBootClocks()/BOARD_InitBootPeripherals(),
109 * none of which touch these pins). PORT_SetPinMux() above only writes
110 * the MUX field, not IBE, so without this the pins reset-default to
111 * IBE disabled -- the I3C peripheral can drive them but can't sense
112 * the target's response. This is the same bug class as the Serial1
113 * D0/D1 input-buffer bug found on A153 (see arduino_serial.cpp
114 * history). Confirmed as the cause via GDB: with IBE left disabled,
115 * I3C_MasterTransferBlocking() succeeded (status 0) for writes but
116 * returned kStatus_I3C_Nak on every read's address phase, regardless
117 * of pull-up, repeated-start vs stop, or SCL frequency.
118 */
119 _scl.input_buffer( true );
120 _sda.input_buffer( true );
121}

211status_t I3C::ccc_broadcast( uint8_t ccc, const uint8_t *dp, uint8_t length, bool first_time )
212{
213 uint8_t bp[ REG_RW_BUFFER_SIZE ];
214 status_t r_code;
215
216 bp[ 0 ] = ccc;
217 memcpy( (uint8_t *)bp + 1, (uint8_t *)dp, length );
218
219 if ( first_time || first_broadcast )
220 {
221 first_broadcast = false;
222
223 frequency( 0, 2000000, 2000000 ); // I2C_freq = default, I3C_OD_freq = 2MHz, I3C_PP_freq = 2MHz
224 r_code = write( BROADCAST_ADDR, bp, length + 1 );
225 frequency(); // revert to default frequency
226 }
227 else
228 {
229 r_code = write( BROADCAST_ADDR, bp, length + 1 );
230 }
231 return r_code;
232}

264void I3C::master_ibi_callback( I3C_Type *base, i3c_master_handle_t *handle, i3c_ibi_type_t ibiType, i3c_ibi_state_t ibiState )
265{
266 g_ibiWonFlag = true;
267 g_ibiAddress = handle->ibiAddress;
268
269 switch ( ibiType )
270 {
271 case kI3C_IbiNormal:
272 if ( ibiState == kI3C_IbiDataBuffNeed )
273 {
274 handle->ibiBuff = g_ibiBuff;
275 }
276 else
277 {
278 memcpy( g_ibiUserBuff, (void *)handle->ibiBuff, handle->ibiPayloadSize );
279 g_ibiUserBuffUsed = handle->ibiPayloadSize;
280 }
281 break;
282
283 default:
284 assert(false);
285 break;
286 }
287
288 if ( g_ibi_callback )
289 g_ibi_callback();
290}
