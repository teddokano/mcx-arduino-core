75/*
76 * millis() / micros()
77 *
78 * DWT->CYCCNT already runs free (enabled by SDK_DelayAtLeastUs() for wait()),
79 * but as a bare 32bit cycle counter it wraps every ~45s at 96MHz — too short
80 * to use directly. SysTick is set to a 1ms tick to extend the range to the
81 * full 32bit millisecond count (~49 days, matching classic Arduino millis()).
82 * DWT->CYCCNT is kept only for sub-millisecond resolution inside micros().
83 */
84
85namespace {
86
87volatile uint32_t ms_ticks = 0;
88volatile uint32_t ms_tick_dwt = 0;
89uint32_t cycles_per_us = 1;
90uint32_t cycles_per_ms = 1000;
91bool millis_ready = false;
92
93void millis_init( void )
94{
95 uint32_t core_hz = CLOCK_GetCoreSysClkFreq();
96
97 cycles_per_us = core_hz / 1000000u;
98 cycles_per_ms = core_hz / 1000u;
99
100#if !defined( __ARM_ARCH_6M__ ) // Cortex-M0+ has no cycle counter: micros() reads SysTick there
101 CoreDebug->DEMCR |= CoreDebug_DEMCR_TRCENA_Msk;
102 DWT->CTRL |= DWT_CTRL_CYCCNTENA_Msk;
103
104 // DWT->CYCCNT has been free-running since boot (wait()/delay() enabled
105 // it). Snapshot "now" as the baseline so micros() doesn't report the
106 // raw since-boot cycle count before the first SysTick tick has fired.
107 // Taken just before SysTick starts, so each tick comes a few cycles
108 // after a whole number of milliseconds from it (SysTick_Handler())
109 ms_tick_dwt = DWT->CYCCNT;
110#endif
111
112 millis_ready = true;
113
114 SysTick_Config( cycles_per_ms );
115}
116
117} // namespace

119#if defined( __ARM_ARCH_6M__ )
120// wait_us() goes through double and 64-bit arithmetic, done in software on
121// a Cortex-M0+: about 45us before it starts to wait. Count SysTick's own
122// down-counter instead, which runs with interrupts off too
123void delayMicroseconds( unsigned int us )
124{
125 if ( !millis_ready )
126 millis_init();
127
128 uint32_t load = SysTick->LOAD + 1u;
129 uint32_t target = us * cycles_per_us;
130 uint32_t elapsed = 0;
131 uint32_t prev = SysTick->VAL;
132
133 while ( elapsed < target )
134 {
135 uint32_t now = SysTick->VAL;
136
137 elapsed += ( prev >= now ) ? ( prev - now ) : ( prev + load - now );
138 prev = now;
139 }
140}
141
142// wait() has no cycle counter to read here either: SDK_DelayAtLeastUs()
143// counts its own loop, so every interrupt taken during it added its time on
144// top (delay( 500 ) took 501 ms, and 514 ms under a 10 kHz TimerOne).
145// Count SysTick's down-counter, as delayMicroseconds() does, a whole
146// millisecond at a time so that a long delay doesn't overflow. Only an
147// interrupt running for over 1 ms between two reads would go uncounted
148void delay( unsigned long ms )
149{
150 if ( !millis_ready )
151 millis_init();
152
153 uint32_t load = SysTick->LOAD + 1u;
154 uint32_t elapsed = 0;
155 uint32_t prev = SysTick->VAL;
156
157 while ( ms > 0 )
158 {
159 uint32_t now = SysTick->VAL;
160
161 elapsed += ( prev >= now ) ? ( prev - now ) : ( prev + load - now );
162 prev = now;
163
164 while ( ( ms > 0 ) && ( elapsed >= cycles_per_ms ) )
165 {
166 elapsed -= cycles_per_ms;
167 ms--;
168 }
169 }
170}
171#endif

173extern "C" void SysTick_Handler( void )
174{
175#if !defined( __ARM_ARCH_6M__ )
176 // The whole milliseconds since the last tick, by the cycle counter. This
177 // handler can run late (SysTick has the lowest priority, and waits behind
178 // MCXEthernet's tick, for one), and the time it runs mustn't become the
179 // start of the millisecond, or micros() reads short until the next tick.
180 // Ticks missed while interrupts were off for over 1 ms come back, and a
181 // tick that comes in while this one counts it adds 0 when it runs
182 uint32_t n = ( DWT->CYCCNT - ms_tick_dwt ) / cycles_per_ms;
183
184 ms_ticks = ms_ticks + n;
185 ms_tick_dwt = ms_tick_dwt + n * cycles_per_ms;
186#else
187 uint32_t n = 1;
188
189 ms_ticks = ms_ticks + 1;
190#endif
191#if defined( CPU_MCXC444VLH )
192 pwm_tpm_service( ms_ticks * 1000u ); // micros() at this tick
193#endif
194
195 void ( *hook )( uint32_t ) = systick_hook;
196
197 if ( hook && n )
198 hook( n );
199
200 hook = systick_ticker_hook;
201
202 if ( hook && n )
203 hook( n );
204}

214unsigned long micros( void )
215{
216 if ( !millis_ready )
217 millis_init();
218
219#if defined( __ARM_ARCH_6M__ )
220 // No cycle counter on a Cortex-M0+: how far SysTick has counted down in
221 // this millisecond gives the rest. A tick pending but not yet taken
222 // (interrupts are off here) means the count has wrapped: one more ms.
223 uint32_t ms;
224 uint32_t val;
225 uint32_t load = SysTick->LOAD;
226 uint32_t primask = __get_PRIMASK(); // called with interrupts off, leave them off
227
228 __disable_irq();
229 ms = ms_ticks;
230 val = SysTick->VAL;
231 if ( ( SCB->ICSR & SCB_ICSR_PENDSTSET_Msk ) && val > load / 2 )
232 ms++;
233 __set_PRIMASK( primask );
234
235 uint32_t delta_us = ( load - val ) / cycles_per_us;
236#else
237 uint32_t ms;
238 uint32_t base;
239 uint32_t primask = __get_PRIMASK(); // called with interrupts off, leave them off
240
241 __disable_irq();
242 ms = ms_ticks;
243 base = ms_tick_dwt;
244 __set_PRIMASK( primask );
245
246 uint32_t delta_us = (uint32_t)( DWT->CYCCNT - base ) / cycles_per_us;
247#endif
248
249 return (unsigned long)( (uint64_t)ms * 1000u + delta_us );
250}
