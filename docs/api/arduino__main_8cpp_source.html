50/*
51 * millis() / micros()
52 *
53 * DWT->CYCCNT already runs free (enabled by SDK_DelayAtLeastUs() for wait()),
54 * but as a bare 32bit cycle counter it wraps every ~45s at 96MHz — too short
55 * to use directly. SysTick is set to a 1ms tick to extend the range to the
56 * full 32bit millisecond count (~49 days, matching classic Arduino millis()).
57 * DWT->CYCCNT is kept only for sub-millisecond resolution inside micros().
58 */
59
60namespace {
61
62volatile uint32_t ms_ticks = 0;
63volatile uint32_t ms_tick_dwt = 0;
64uint32_t cycles_per_us = 1;
65bool millis_ready = false;
66
67void millis_init( void )
68{
69 uint32_t core_hz = CLOCK_GetCoreSysClkFreq();
70
71 cycles_per_us = core_hz / 1000000u;
72
73 CoreDebug->DEMCR |= CoreDebug_DEMCR_TRCENA_Msk;
74 DWT->CTRL |= DWT_CTRL_CYCCNTENA_Msk;
75
76 // DWT->CYCCNT has been free-running since boot (wait()/delay() enabled
77 // it). Snapshot "now" as the baseline so micros() doesn't report the
78 // raw since-boot cycle count before the first SysTick tick has fired.
79 ms_tick_dwt = DWT->CYCCNT;
80
81 millis_ready = true;
82
83 SysTick_Config( core_hz / 1000u );
84}
85
86} // namespace
87
88extern "C" void SysTick_Handler( void )
89{
90 ms_ticks = ms_ticks + 1;
91 ms_tick_dwt = DWT->CYCCNT;
92}

94unsigned long millis( void )
95{
96 if ( !millis_ready )
97 millis_init();
98
99 return (unsigned long)ms_ticks;
100}

102unsigned long micros( void )
103{
104 if ( !millis_ready )
105 millis_init();
106
107 uint32_t ms;
108 uint32_t base;
109
110 __disable_irq();
111 ms = ms_ticks;
112 base = ms_tick_dwt;
113 __enable_irq();
114
115 uint32_t delta_us = (uint32_t)( DWT->CYCCNT - base ) / cycles_per_us;
116
117 return (unsigned long)( (uint64_t)ms * 1000u + delta_us );
118}
