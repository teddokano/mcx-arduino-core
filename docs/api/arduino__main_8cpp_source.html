36/*
37 * millis() / micros()
38 *
39 * DWT->CYCCNT already runs free (enabled by SDK_DelayAtLeastUs() for wait()),
40 * but as a bare 32bit cycle counter it wraps every ~45s at 96MHz — too short
41 * to use directly. SysTick is set to a 1ms tick to extend the range to the
42 * full 32bit millisecond count (~49 days, matching classic Arduino millis()).
43 * DWT->CYCCNT is kept only for sub-millisecond resolution inside micros().
44 */
45
46namespace {
47
48volatile uint32_t ms_ticks = 0;
49volatile uint32_t ms_tick_dwt = 0;
50uint32_t cycles_per_us = 1;
51bool millis_ready = false;
52
53void millis_init( void )
54{
55 uint32_t core_hz = CLOCK_GetCoreSysClkFreq();
56
57 cycles_per_us = core_hz / 1000000u;
58
59 CoreDebug->DEMCR |= CoreDebug_DEMCR_TRCENA_Msk;
60 DWT->CTRL |= DWT_CTRL_CYCCNTENA_Msk;
61
62 // DWT->CYCCNT has been free-running since boot (wait()/delay() enabled
63 // it). Snapshot "now" as the baseline so micros() doesn't report the
64 // raw since-boot cycle count before the first SysTick tick has fired.
65 ms_tick_dwt = DWT->CYCCNT;
66
67 millis_ready = true;
68
69 SysTick_Config( core_hz / 1000u );
70}
71
72} // namespace
73
74extern "C" void SysTick_Handler( void )
75{
76 ms_ticks = ms_ticks + 1;
77 ms_tick_dwt = DWT->CYCCNT;
78}
79
80unsigned long millis( void )
81{
82 if ( !millis_ready )
83 millis_init();
84
85 return (unsigned long)ms_ticks;
86}
87
88unsigned long micros( void )
89{
90 if ( !millis_ready )
91 millis_init();
92
93 uint32_t ms;
94 uint32_t base;
95
96 __disable_irq();
97 ms = ms_ticks;
98 base = ms_tick_dwt;
99 __enable_irq();
100
101 uint32_t delta_us = (uint32_t)( DWT->CYCCNT - base ) / cycles_per_us;
102
103 return (unsigned long)( (uint64_t)ms * 1000u + delta_us );
104}
