82uint32_t mcxTicker::ms_of( float seconds )
83{
84 float ms = seconds * 1000.0f + 0.5f;
85
86 if ( !( ms >= 1.0f ) ) // NaN too
87 return 1u;
88 if ( ms >= 4294967040.0f ) // the largest float below 2^32
89 return 0xFFFFFFFFu;
90
91 return (uint32_t)ms;

104void mcxTicker::link( uint32_t ms, bool repeat, thunk_t thunk, callback_t callback, const arg_t &bits )
105{
106 if ( callback == nullptr )
107 {
108 detach();
109 return;
110 }
111
112 if ( ms == 0 )
113 ms = 1;
114
115 uint32_t primask = irq_save();
116
117 this->thunk = thunk;
118 func = callback;
119 arg = bits;
120 period = ms;
121 this->repeat = repeat;
122
123 // From loop(), the next tick comes in less than 1 ms, so one more
124 // makes the first call at least ms away. From an mcxTicker's function,
125 // at a tick, it is a whole ms away. seen keeps the tick that may be
126 // going on from counting this one
127 left = ( in_systick() || ms == 0xFFFFFFFFu ) ? ms : ms + 1u;
128 seen = serial;
129
130 if ( !linked )
131 {
132 next = head;
133 head = this;
134 linked = true;
135 }
136
137 if ( systick_ticker_hook == nullptr )
138 systick_ticker_hook = tick;
139
140 irq_restore( primask );
141}

143// With interrupts off
144void mcxTicker::unlink()
145{
146 if ( !linked )
147 return;
148
149 for ( mcxTicker **p = &head; *p != nullptr; p = &(*p)->next )
150 {
151 if ( *p == this )
152 {
153 *p = next; // next stays as it is, so that tick() can go on past this one
154 break;
155 }
156 }
157
158 linked = false;
159}

161void mcxTicker::tick( uint32_t ms )
162{
163 serial = serial + 1;
164
165 for ( mcxTicker *t = head; t != nullptr; )
166 {
167 mcxTicker *following = t->next; // a function may detach or move its own object
168
169 if ( t->linked && t->seen != serial )
170 {
171 t->seen = serial;
172
173 if ( t->left > ms )
174 t->left = t->left - ms;
175 else
176 {
177 // Late by more than a period (interrupts were off): the
178 // missed calls are skipped, on the same timing
179 uint32_t over = ms - t->left;
180
181 if ( t->repeat )
182 t->left = t->period - over % t->period;
183 else
184 t->unlink();
185
186 t->thunk( t );
187 }
188 }
189
190 t = following;
191 }
