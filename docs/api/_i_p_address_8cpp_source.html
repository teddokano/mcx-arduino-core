63// Exactly four decimal fields of 1-3 digits, each 0-255, separated by dots
64bool IPAddress::parse_v4( const char *s, uint8_t *out )
65{
66 for ( int field = 0; field < 4; field++ )
67 {
68 if ( field > 0 && *s++ != '.' )
69 return false;
70
71 int value = 0;
72 int digits = 0;
73
74 while ( '0' <= *s && *s <= '9' )
75 {
76 if ( ++digits > 3 )
77 return false;
78
79 value = value * 10 + ( *s++ - '0' );
80 }
81
82 if ( digits == 0 || value > 255 )
83 return false;
84
85 out[ field ] = (uint8_t)value;
86 }
87
88 return *s == '\0';
89}

91static int hex_value( char c )
92{
93 if ( '0' <= c && c <= '9' )
94 return c - '0';
95 if ( 'a' <= c && c <= 'f' )
96 return c - 'a' + 10;
97 if ( 'A' <= c && c <= 'F' )
98 return c - 'A' + 10;
99
100 return -1;
101}

103// Groups of 1-4 hex digits separated by ':', eight of them, or fewer with
104// one "::" standing for the zero groups that are left out (at least one)
105bool IPAddress::parse_v6( const char *s, uint8_t *out )
106{
107 uint16_t groups[ 8 ];
108 int count = 0; // groups read so far
109 int gap = -1; // where "::" was, as a group index
110
111 if ( s[ 0 ] == ':' )
112 {
113 if ( s[ 1 ] != ':' )
114 return false;
115
116 gap = 0;
117 s += 2;
118 }
119
120 while ( *s != '\0' )
121 {
122 if ( count == 8 )
123 return false;
124
125 int value = 0;
126 int digits = 0;
127 int h;
128
129 while ( ( h = hex_value( *s ) ) >= 0 )
130 {
131 if ( ++digits > 4 )
132 return false;
133
134 value = value * 16 + h;
135 s++;
136 }
137
138 if ( digits == 0 )
139 return false;
140
141 groups[ count++ ] = (uint16_t)value;
142
143 if ( *s == '\0' )
144 break;
145 if ( *s++ != ':' )
146 return false;
147
148 if ( *s == ':' )
149 {
150 if ( gap >= 0 )
151 return false;
152
153 gap = count;
154 s++;
155 }
156 else if ( *s == '\0' )
157 {
158 return false; // a single ':' at the end
159 }
160 }
161
162 if ( gap < 0 ? count != 8 : count > 7 )
163 return false;
164
165 int zeros = 8 - count;
166 int g = 0;
167
168 for ( int i = 0; i < 8; i++ )
169 {
170 uint16_t v = ( gap <= i && i < gap + zeros ) ? 0 : groups[ g++ ];
171 out[ i * 2 ] = v >> 8;
172 out[ i * 2 + 1 ] = v & 0xFF;
173 }
174
175 return true;
176}

219// Writes value in decimal or lower-case hex without leading zeros, returns
220// where the next character goes
221static char *put_number( char *t, unsigned value, unsigned base )
222{
223 char digits[ 5 ];
224 int n = 0;
225
226 do
227 {
228 digits[ n++ ] = "0123456789abcdef"[ value % base ];
229 value /= base;
230 }
231 while ( value );
232
233 while ( n )
234 *t++ = digits[ --n ];
235
236 return t;
237}

239// Writes the text and its NUL into text (TEXT_SIZE bytes), returns its length
240size_t IPAddress::format( char *text ) const
241{
242 char *t = text;
243
244 if ( _type == IPv4 )
245 {
246 for ( int i = 0; i < 4; i++ )
247 {
248 if ( i > 0 )
249 *t++ = '.';
250
251 t = put_number( t, _bytes[ V4_OFFSET + i ], 10 );
252 }
253
254 *t = '\0';
255 return t - text;
256 }
257
258 // The longest run of zero groups, the first one if two are as long.
259 // A single zero group is written as "0", not "::" (RFC 5952 4.2.2).
260 int run_start = -1;
261 int run_length = 1;
262
263 for ( int i = 0; i < 8; )
264 {
265 int n = 0;
266
267 while ( i + n < 8 && _bytes[ ( i + n ) * 2 ] == 0 && _bytes[ ( i + n ) * 2 + 1 ] == 0 )
268 n++;
269
270 if ( n > run_length )
271 {
272 run_start = i;
273 run_length = n;
274 }
275
276 i += n ? n : 1;
277 }
278
279 for ( int i = 0; i < 8; i++ )
280 {
281 if ( i == run_start )
282 {
283 *t++ = ':';
284 *t++ = ':';
285 i += run_length - 1;
286 continue;
287 }
288
289 if ( i > 0 && i != run_start + run_length )
290 *t++ = ':';
291
292 t = put_number( t, ( _bytes[ i * 2 ] << 8 ) | _bytes[ i * 2 + 1 ], 16 );
293 }
294
295 *t = '\0';
296 return t - text;
297}
