19void TwoWire::begin( int baud )
20{
21 baudrate = baud;
22
23 bool is_i3c = ( I3C_SDA == _sda ) && ( I3C_SCL == _scl );
24
25 if ( !i2c )
26 {
27 if ( is_i3c )
28 {
29 I3C *i3c;
30 i3c = new I3C( _sda, _scl );
31
32#ifdef CPU_MCXN947VDF
33 /*
34 * At least one native-I3C-mode bus operation needs to happen on
35 * this bus before switching it down to legacy I2C_MODE -- on
36 * FRDM-MCXN947, without this, I2C-mode reads on this bus
37 * reliably NAK on the address phase (writes still succeed).
38 * Confirmed via a controlled A/B test on real hardware (GDB,
39 * then reflashed with/without this line): consistently fails
40 * without it, consistently works with it. Root cause not fully
41 * understood -- per the I3C spec the master's pull-up assist
42 * should already be active during bus-free/START/address-header
43 * regardless of prior traffic, so this looks like an SDK/
44 * silicon init-ordering quirk (I3C_MasterInit() alone may not
45 * be enough to arm the pull-up-assist state machine; one real
46 * START/STOP cycle might be needed first) rather than a spec
47 * violation we're intentionally working around. This priming
48 * broadcast is expected to itself fail (kStatus_I3C_WriteAbort,
49 * nothing has a dynamic address yet) -- only the bus activity
50 * matters. Matches the flow NXP's own P3T1755_FRDM_MCXN947_demo_DAA
51 * reference example always does (RSTDAA before any I2C-mode
52 * access).
53 *
54 * Confirmed NOT needed on A153 (its I3C_SDA/I3C_SCL are the
55 * same physical pins as the general-purpose Arduino I2C
56 * connector, which has real populated pull-up resistors --
57 * unlike N947's dedicated onboard-sensor-only I3C bus, whose
58 * fixed pull-ups are DNP on the schematic). Scoped to just
59 * this chip rather than done unconditionally, so boards that
60 * don't need it don't get a mystery extra I2C-bus transaction
61 * on every Wire.begin().
62 */
63 i3c->ccc_broadcast( CCC::BROADCAST_RSTDAA, nullptr, 0 );
64#endif // CPU_MCXN947VDF
65
66 i3c->mode( I3C::MODE::I2C_MODE );
67 i2c = i3c;
68 }
69 else
70 {
71 i2c = new I2C( _sda, _scl );
72 }
73 }
74
75 /*
76 * I2C(sda, scl, no_hw=true) — the base-class constructor I3C delegates
77 * to — skips hardware init entirely (`if (no_hw) return;`), leaving
78 * I2C::unit_base uninitialized. I3C's own
79 * frequency(uint32_t,uint32_t,uint32_t)/frequency(void) only hide, not
80 * override, I2C::frequency(uint32_t) (different signature), so calling
81 * the generic i2c->frequency(baudrate) for an I3C instance would
82 * dispatch to I2C::frequency() and dereference that uninitialized
83 * unit_base -> BusFault. Route to I3C's own overload instead.
84 */
85 if ( is_i3c )
86 static_cast<I3C *>( i2c )->frequency( baudrate, 0, 0 );
87 else
88 i2c->frequency( baudrate );
89
90 i2c->err_callback( nullptr );
91}
