29/*
30 * Target (slave) mode, on the SDK's LPI2C_Slave* transfer API.
31 *
32 * An LPI2C's master and slave halves run independently on the same pins,
33 * so a board can be a controller and a target at once, even addressing
34 * its own target. Four things the SDK leaves to the caller, each seen on
35 * hardware:
36 *
37 * - FRDM-MCXA153's SDK (fsl_lpi2c 2.5.4) turns the master off when it
38 * arms the slave (LPI2C_SlaveTransferNonBlocking(): "Enable the slave
39 * function and disable the master function"); FRDM-MCXN947's (2.2.4)
40 * doesn't. The master is turned back on after arming, or every later
41 * controller transfer waits forever on a disabled master.
42 * - When the reply a controller reads runs out, the SDK sends nothing and
43 * the slave stretches SCL forever. 0xFF is sent from there on, as AVR
44 * does.
45 * - A write longer than the buffer: the byte that doesn't fit is NAKed
46 * (ACKSTALL, deciding each ACK in the TransmitAck event), and still
47 * read into a scratch byte, since an unread byte leaves the receive
48 * flag set and the interrupt firing.
49 * - Deleting the master (end(), or the reset after a timeout) gates the
50 * LPI2C's clock (A153) or de-initializes its FlexComm (N947), which
51 * takes the slave with it, so start() re-arms the target after
52 * re-creating the master.
53 */
54struct WireTarget
55{
56 lpi2c_slave_handle_t handle;
57 uint8_t address;
58 bool active;
59 bool reading; // the controller is reading, not writing
60 bool rx_given; // rx[] handed to the SDK for this write
61 bool rx_full; // ... and it filled up
62 bool tx_given; // onRequest() already asked for this read
63 bool in_request; // inside onRequest(): write() fills tx[]
64 uint8_t rx[ WIRE_BUFFER_SIZE ]; // the SDK receives into this
65 uint8_t received[ WIRE_BUFFER_SIZE ]; // read() reads a finished write here
66 uint8_t tx[ WIRE_BUFFER_SIZE ];
67 size_t tx_size;
68 uint8_t fill; // what a controller gets past the reply
69 uint8_t sink; // where a NAKed byte is read to
70};

102void TwoWire::begin( uint8_t address )
103{
104 if ( on_i3c_pins() )
105 panic( "Wire1: target (slave) mode isn't supported on Wire1, which runs on the I3C peripheral. Use Wire" );
106#ifdef CPU_MCXN947VDF
107 /*
108 * LPI2C3 (FlexComm3), behind Wire2, never sees the bus as a target:
109 * its slave status stays 0 through a transfer addressed to it, while
110 * its own master side on the same pins works, and LPI2C2 set up
111 * identically (Wire) works. Every slave register matches LPI2C2's.
112 * Cause not found; checked with Wire and Wire2 jumpered together.
113 */
114 if ( _sda == MB_SDA && _scl == MB_SCL )
115 panic( "Wire2: target (slave) mode doesn't work on Wire2 (LPI2C3 on FRDM-MCXN947). Use Wire" );
116#endif
117 if ( address > 127 )
118 panic( "Wire: begin(address) given an address over 127" );
119
120 if ( !target )
121 target = new WireTarget();
122 target->address = address;
123 target->active = true;
124
125 start( 100000 ); // a controller too, as on AVR; start() arms the target
126}

151void TwoWire::target_arm( void )
152{
153 LPI2C_Type *base = i2c->lpi2c_base();
154 WireTarget *t = target;
155
156 t->reading = false;
157 t->rx_given = false;
158 t->rx_full = false;
159 t->tx_given = false;
160 t->in_request = false;
161 t->fill = 0xFF;
162
163 lpi2c_slave_config_t cfg;
164 LPI2C_SlaveGetDefaultConfig( &cfg );
165 cfg.address0 = t->address;
166 cfg.sclStall.enableAck = true; // ACKSTALL, so a full buffer can NAK
167
168 LPI2C_SlaveInit( base, &cfg, i2c->lpi2c_clock() );
169#if defined( CPU_MCXN236VDF )
170 flexcomm_keep_shared( LPI2C_GetInstance( base ) ); // Serial1's FlexComm too
171#endif
172 LPI2C_SlaveTransferCreateHandle( base, &t->handle, target_callback, this );
173 LPI2C_SlaveTransferNonBlocking( base, &t->handle,
174 kLPI2C_SlaveAddressMatchEvent | kLPI2C_SlaveTransmitAckEvent |
175 kLPI2C_SlaveRepeatedStartEvent | kLPI2C_SlaveCompletionEvent );
176
177 LPI2C_MasterEnable( base, true ); // A153's SDK turned it off (see top)
178 i2c->wait_bus_idle();
179}

197void TwoWire::target_event( void *p )
198{
199 lpi2c_slave_transfer_t *x = static_cast<lpi2c_slave_transfer_t *>( p );
200 WireTarget *t = target;
201
202 switch ( x->event )
203 {
204 case kLPI2C_SlaveAddressMatchEvent:
205 t->reading = x->receivedAddress & 1u;
206 t->rx_given = false;
207 t->rx_full = false;
208 t->tx_given = false;
209 x->data = nullptr;
210 x->dataSize = 0;
211 break;
212
213 case kLPI2C_SlaveTransmitAckEvent:
214 {
215 // Before the byte is stored: ACK it only if it will fit
216 size_t stored = t->rx_given ? sizeof( t->rx ) - x->dataSize : 0;
217 LPI2C_SlaveTransmitAck( i2c->lpi2c_base(), !t->rx_full && stored < sizeof( t->rx ) );
218 break;
219 }
220
221 case kLPI2C_SlaveReceiveEvent:
222 if ( !t->rx_given )
223 {
224 x->data = t->rx;
225 x->dataSize = sizeof( t->rx );
226 t->rx_given = true;
227 }
228 else
229 {
230 t->rx_full = true; // NAKed above; read it and drop it
231 x->data = &t->sink;
232 x->dataSize = 1;
233 }
234 break;
235
236 case kLPI2C_SlaveTransmitEvent:
237 if ( !t->tx_given )
238 {
239 t->tx_given = true;
240 t->tx_size = 0;
241 if ( user_onRequest )
242 {
243 t->in_request = true;
244 user_onRequest();
245 t->in_request = false;
246 }
247 if ( t->tx_size )
248 {
249 x->data = t->tx;
250 x->dataSize = t->tx_size;
251 break;
252 }
253 }
254 x->data = &t->fill; // past the reply: 0xFF, as on AVR
255 x->dataSize = 1;
256 break;
257
258 case kLPI2C_SlaveRepeatedStartEvent:
259 case kLPI2C_SlaveCompletionEvent:
260 // A controller's write just ended: hand it to onReceive(), as
261 // AVR does even for a write of no bytes
262 if ( !t->reading && x->completionStatus == kStatus_Success )
263 {
264 size_t n = t->rx_full ? sizeof( t->rx ) : t->rx_given ? x->transferredCount : 0;
265
266 // Its own buffer, not rx_buf: this can land in the middle of
267 // a requestFrom() of our own, which reads into rx_buf
268 memcpy( t->received, t->rx, n );
269 rx_data = t->received;
270 rx_index = 0;
271 rx_size = n;
272
273 if ( user_onReceive )
274 user_onReceive( (int)n );
275 }
276 t->reading = false;
277 t->rx_given = false;
278 t->rx_full = false;
279 break;
280
281 default:
282 break;
283 }
284}

286void TwoWire::start( int baud )
287{
288 baudrate = baud;
289
290 bool is_i3c = on_i3c_pins();
291
292 if ( !i2c )
293 {
294 if ( is_i3c )
295 {
296 I3C *i3c;
297 i3c = new I3C( _sda, _scl );
298
299#ifdef CPU_MCXN947VDF
300 /*
301 * At least one native-I3C-mode bus operation needs to happen on
302 * this bus before switching it down to legacy I2C_MODE -- on
303 * FRDM-MCXN947, without this, I2C-mode reads on this bus
304 * reliably NAK on the address phase (writes still succeed).
305 * Confirmed via a controlled A/B test on real hardware (GDB,
306 * then reflashed with/without this line): consistently fails
307 * without it, consistently works with it. Root cause not fully
308 * understood -- per the I3C spec the master's pull-up assist
309 * should already be active during bus-free/START/address-header
310 * regardless of prior traffic, so this looks like an SDK/
311 * silicon init-ordering quirk (I3C_MasterInit() alone may not
312 * be enough to arm the pull-up-assist state machine; one real
313 * START/STOP cycle might be needed first) rather than a spec
314 * violation we're intentionally working around. This priming
315 * broadcast is expected to itself fail (kStatus_I3C_WriteAbort,
316 * nothing has a dynamic address yet) -- only the bus activity
317 * matters. Matches the flow NXP's own P3T1755_FRDM_MCXN947_demo_DAA
318 * reference example always does (RSTDAA before any I2C-mode
319 * access).
320 *
321 * Confirmed NOT needed on A153 (its I3C_SDA/I3C_SCL are the
322 * same physical pins as the general-purpose Arduino I2C
323 * connector, which has real populated pull-up resistors --
324 * unlike N947's dedicated onboard-sensor-only I3C bus, whose
325 * fixed pull-ups are DNP on the schematic). Scoped to just
326 * this chip rather than done unconditionally, so boards that
327 * don't need it don't get a mystery extra I2C-bus transaction
328 * on every Wire.begin().
329 */
330 i3c->ccc_broadcast( CCC::BROADCAST_RSTDAA, nullptr, 0 );
331#endif // CPU_MCXN947VDF
332
333 i3c->mode( I3C::MODE::I2C_MODE );
334 i2c = i3c;
335 }
336 else
337 {
338 i2c = new I2C( _sda, _scl );
339
340 // Internal pull-ups on, as AVR's twi_init() and UNO R4's
341 // begin() do: FRDM-MCXA153 has no pull-up resistors on D18/D19,
342 // so without them a bus with nothing else on it floats, reads
343 // low, and the first transfer hangs or finds the bus busy.
344 // Harmless next to external pull-ups, which set the rise time.
345 i2c->pullup( true );
346 }
347 }
348
349 /*
350 * I2C(sda, scl, no_hw=true) — the base-class constructor I3C delegates
351 * to — skips hardware init entirely (`if (no_hw) return;`), leaving
352 * I2C::unit_base uninitialized. I3C's own
353 * frequency(uint32_t,uint32_t,uint32_t)/frequency(void) only hide, not
354 * override, I2C::frequency(uint32_t) (different signature), so calling
355 * the generic i2c->frequency(baudrate) for an I3C instance would
356 * dispatch to I2C::frequency() and dereference that uninitialized
357 * unit_base -> BusFault. Route to I3C's own overload instead.
358 */
359 if ( is_i3c )
360 static_cast<I3C *>( i2c )->frequency( baudrate, 0, 0 );
361 else
362 {
363 i2c->pin_low_timeout( timeout_us );
364 i2c->frequency( baudrate );
365 }
366
367 i2c->err_callback( nullptr );
368
369 if ( target && target->active )
370 target_arm();
371}

403size_t TwoWire::write( uint8_t data )
404{
405 if ( target && target->in_request ) // onRequest(): the reply
406 {
407 if ( target->tx_size >= sizeof( target->tx ) )
408 {
409 setWriteError();
410 return 0;
411 }
412 target->tx[ target->tx_size++ ] = data;
413 return 1;
414 }
415
416 if ( tx_size >= sizeof( tx_buf ) )
417 {
418 setWriteError();
419 return 0;
420 }
421
422 tx_buf[ tx_size++ ] = data;
423
424 return 1;
425}

445uint8_t TwoWire::requestFrom( const uint8_t address, const size_t length, bool stop )
446{
447 size_t n = ( length < sizeof( rx_buf ) ) ? length : sizeof( rx_buf );
448
449 rx_data = rx_buf;
450 rx_index = 0;
451 rx_size = 0;
452
453 status_t r = i2c->read( address, rx_buf, n, stop );
454
455 check_timeout( r );
456 if ( r )
457 return 0;
458
459 // Set again after the read: this board's own target can take a write
460 // (and hand it to onReceive(), moving these) partway through it
461 rx_data = rx_buf;
462 rx_index = 0;
463 rx_size = n;
464 return n;
465}

467uint8_t TwoWire::requestFrom( uint8_t address, uint8_t quantity, uint32_t iaddress, uint8_t isize, uint8_t sendStop )
468{
469 if ( isize > 0 )
470 {
471 if ( isize > 3 )
472 isize = 3;
473
474 beginTransmission( address );
475 while ( isize-- > 0 )
476 write( (uint8_t)( iaddress >> ( isize * 8 ) ) );
477
478 if ( endTransmission( false ) )
479 {
480 rx_data = rx_buf;
481 rx_index = 0;
482 rx_size = 0;
483 return 0;
484 }
485 }
486
487 return requestFrom( address, (size_t)quantity, (bool)sendStop );
488}
