33/*
34 * Target (slave) mode, on the SDK's LPI2C_Slave* transfer API.
35 *
36 * An LPI2C's master and slave halves run independently on the same pins,
37 * so a board can be a controller and a target at once, even addressing
38 * its own target. Four things the SDK leaves to the caller, each seen on
39 * hardware:
40 *
41 * - FRDM-MCXA153's SDK (fsl_lpi2c 2.5.4) turns the master off when it
42 * arms the slave (LPI2C_SlaveTransferNonBlocking(): "Enable the slave
43 * function and disable the master function"); FRDM-MCXN947's (2.2.4)
44 * doesn't. The master is turned back on after arming, or every later
45 * controller transfer waits forever on a disabled master.
46 * - When the reply a controller reads runs out, the SDK sends nothing and
47 * the slave stretches SCL forever. 0xFF is sent from there on, as AVR
48 * does.
49 * - A write longer than the buffer: the byte that doesn't fit is NAKed
50 * (ACKSTALL, deciding each ACK in the TransmitAck event), and still
51 * read into a scratch byte, since an unread byte leaves the receive
52 * flag set and the interrupt firing.
53 * - Deleting the master (end(), or the reset after a timeout) gates the
54 * LPI2C's clock (A153) or de-initializes its FlexComm (N947), which
55 * takes the slave with it, so start() re-arms the target after
56 * re-creating the master.
57 */
58struct WireTarget
59{
60#if !defined( CPU_MCXC444VLH ) // LPI2C and I3C; FRDM-MCXC444 has the Kinetis I2C
61 lpi2c_slave_handle_t handle;
62#endif
63 uint8_t address;
64 bool active;
65 bool reading; // the controller is reading, not writing
66 bool rx_given; // rx[] handed to the SDK for this write
67 bool rx_full; // ... and it filled up
68 bool tx_given; // onRequest() already asked for this read
69 bool in_request; // inside onRequest(): write() fills tx[]
70 uint8_t rx[ WIRE_BUFFER_SIZE ]; // the SDK receives into this
71 uint8_t received[ WIRE_BUFFER_SIZE ]; // read() reads a finished write here
72 uint8_t tx[ WIRE_BUFFER_SIZE ];
73 size_t tx_size;
74 uint8_t fill; // what a controller gets past the reply
75 uint8_t sink; // where a NAKed byte is read to
76#if defined( CPU_MCXC444VLH )
77 bool in_write; // a controller is writing to this target
78 size_t rx_count; // bytes of that write in rx[]
79 size_t tx_index; // next byte of tx[] to send
80#endif
81};

115void TwoWire::begin( uint8_t address )
116{
117 if ( on_i3c_pins() )
118 panic( "Wire1: target (slave) mode isn't supported on Wire1, which runs on the I3C peripheral. Use Wire" );
119#ifdef CPU_MCXN947VDF
120 /*
121 * LPI2C3 (FlexComm3), behind Wire2, never sees the bus as a target:
122 * its slave status stays 0 through a transfer addressed to it, while
123 * its own master side on the same pins works, and LPI2C2 set up
124 * identically (Wire) works. Every slave register matches LPI2C2's.
125 * Cause not found; checked with Wire and Wire2 jumpered together.
126 */
127 if ( _sda == MB_SDA && _scl == MB_SCL )
128 panic( "Wire2: target (slave) mode doesn't work on Wire2 (LPI2C3 on FRDM-MCXN947). Use Wire" );
129#endif
130 if ( address > 127 )
131 panic( "Wire: begin(address) given an address over 127" );
132
133 if ( !target )
134 target = new WireTarget();
135 target->address = address;
136 target->active = true;
137
138 start( 100000 ); // a controller too, as on AVR; start() arms the target
139}

164#if !defined( CPU_MCXC444VLH ) // LPI2C and I3C; FRDM-MCXC444 has the Kinetis I2C
165void TwoWire::target_arm( void )
166{
167 LPI2C_Type *base = i2c->lpi2c_base();
168 WireTarget *t = target;
169
170 t->reading = false;
171 t->rx_given = false;
172 t->rx_full = false;
173 t->tx_given = false;
174 t->in_request = false;
175 t->fill = 0xFF;
176
177 lpi2c_slave_config_t cfg;
178 LPI2C_SlaveGetDefaultConfig( &cfg );
179 cfg.address0 = t->address;
180 cfg.sclStall.enableAck = true; // ACKSTALL, so a full buffer can NAK
181
182 LPI2C_SlaveInit( base, &cfg, i2c->lpi2c_clock() );
183#if defined( CPU_MCXN236VDF )
184 flexcomm_keep_shared( LPI2C_GetInstance( base ) ); // Serial1's FlexComm too
185#endif
186 LPI2C_SlaveTransferCreateHandle( base, &t->handle, target_callback, this );
187 LPI2C_SlaveTransferNonBlocking( base, &t->handle,
188 kLPI2C_SlaveAddressMatchEvent | kLPI2C_SlaveTransmitAckEvent |
189 kLPI2C_SlaveRepeatedStartEvent | kLPI2C_SlaveCompletionEvent );
190
191 LPI2C_MasterEnable( base, true ); // A153's SDK turned it off (see top)
192 i2c->wait_bus_idle();
193}

211void TwoWire::target_event( void *p )
212{
213 lpi2c_slave_transfer_t *x = static_cast<lpi2c_slave_transfer_t *>( p );
214 WireTarget *t = target;
215
216 switch ( x->event )
217 {
218 case kLPI2C_SlaveAddressMatchEvent:
219 t->reading = x->receivedAddress & 1u;
220 t->rx_given = false;
221 t->rx_full = false;
222 t->tx_given = false;
223 x->data = nullptr;
224 x->dataSize = 0;
225 break;
226
227 case kLPI2C_SlaveTransmitAckEvent:
228 {
229 // Before the byte is stored: ACK it only if it will fit
230 size_t stored = t->rx_given ? sizeof( t->rx ) - x->dataSize : 0;
231 LPI2C_SlaveTransmitAck( i2c->lpi2c_base(), !t->rx_full && stored < sizeof( t->rx ) );
232 break;
233 }
234
235 case kLPI2C_SlaveReceiveEvent:
236 if ( !t->rx_given )
237 {
238 x->data = t->rx;
239 x->dataSize = sizeof( t->rx );
240 t->rx_given = true;
241 }
242 else
243 {
244 t->rx_full = true; // NAKed above; read it and drop it
245 x->data = &t->sink;
246 x->dataSize = 1;
247 }
248 break;
249
250 case kLPI2C_SlaveTransmitEvent:
251 if ( !t->tx_given )
252 {
253 t->tx_given = true;
254 t->tx_size = 0;
255 if ( user_onRequest )
256 {
257 t->in_request = true;
258 user_onRequest();
259 t->in_request = false;
260 }
261 if ( t->tx_size )
262 {
263 x->data = t->tx;
264 x->dataSize = t->tx_size;
265 break;
266 }
267 }
268 x->data = &t->fill; // past the reply: 0xFF, as on AVR
269 x->dataSize = 1;
270 break;
271
272 case kLPI2C_SlaveRepeatedStartEvent:
273 case kLPI2C_SlaveCompletionEvent:
274 // A controller's write just ended: hand it to onReceive(), as
275 // AVR does even for a write of no bytes
276 if ( !t->reading && x->completionStatus == kStatus_Success )
277 {
278 size_t n = t->rx_full ? sizeof( t->rx ) : t->rx_given ? x->transferredCount : 0;
279
280 // Its own buffer, not rx_buf: this can land in the middle of
281 // a requestFrom() of our own, which reads into rx_buf
282 memcpy( t->received, t->rx, n );
283 rx_data = t->received;
284 rx_index = 0;
285 rx_size = n;
286
287 if ( user_onReceive )
288 user_onReceive( (int)n );
289 }
290 t->reading = false;
291 t->rx_given = false;
292 t->rx_full = false;
293 break;
294
295 default:
296 break;
297 }
298}
299#else
300/*
301 * Target (slave) mode on FRDM-MCXC444's Kinetis I2C, driven from the I2C
302 * interrupt here: the SDK's I2C_SlaveInit() would rewrite the master's
303 * settings on the same module too.
304 *
305 * One Kinetis I2C is master or slave at a time, never both at once, so
306 * unlike an LPI2C it can't address its own target. r01lib's controller
307 * transfers turn this interrupt and FLT[SSIE] off while they run (they
308 * poll the same IICIF, which SSIE would set for their own START), and a
309 * controller transfer can't start while another controller is talking to
310 * this target (the bus is busy).
311 *
312 * The STOP and START detection interrupt (FLT[SSIE]) ends a write. Flags
313 * are cleared in the order the reference manual asks: STOPF/STARTF first,
314 * then IICIF.
315 */
316namespace {

318TwoWire *s_target_of[ 2 ] = {}; // the TwoWire armed on I2C0, I2C1

320constexpr uint8_t FLT_W1C = I2C_FLT_STOPF_MASK | I2C_FLT_STARTF_MASK;

322int instance_of( I2C_Type *base )
323{
324 return ( base == I2C0 ) ? 0 : 1;
325}

327void target_irq( int n )
328{
329 if ( s_target_of[ n ] )
330 s_target_of[ n ]->target_event( nullptr );
331 else
332 {

334
335 base->FLT = base->FLT; // clears STOPF/STARTF that are set

354void TwoWire::target_arm( void )
355{
356 I2C_Type *base = i2c->kinetis_base();
357 WireTarget *t = target;
358 int n = instance_of( base );
359
360 t->in_write = false;
361 t->rx_count = 0;
362 t->rx_full = false;
363 t->tx_size = 0;
364 t->tx_index = 0;
365 t->in_request = false;
366 t->fill = 0xFF;
367
368 s_target_of[ n ] = this;
369
370 // The address register is named A1, as is the Arduino pin (io.h)
371#pragma push_macro( "A1" )
372#undef A1
373 base->A1 = (uint8_t)( t->address << 1 );
374#pragma pop_macro( "A1" )
375 base->FLT = ( base->FLT & (uint8_t)~FLT_W1C ) | I2C_FLT_SSIE_MASK;
376 base->FLT = base->FLT; // clear stale STOPF/STARTF
377 base->S = I2C_S_IICIF_MASK | I2C_S_ARBL_MASK;
378 base->C1 = ( base->C1 & (uint8_t)~( I2C_C1_TX_MASK | I2C_C1_TXAK_MASK ) ) | I2C_C1_IICIE_MASK;
379
380 EnableIRQ( n ? I2C1_IRQn : I2C0_IRQn );
381}

404void TwoWire::target_event( void *p )
405{
406 (void)p;
407
408 I2C_Type *base = i2c->kinetis_base();
409 WireTarget *t = target;
410 uint8_t flt = base->FLT;
411
412 if ( flt & FLT_W1C )
413 {
414 base->FLT = flt; // STOPF/STARTF first ...
415 base->S = I2C_S_IICIF_MASK; // ... then IICIF
416 base->C1 &= (uint8_t)~I2C_C1_TXAK_MASK;
417
418 // A controller's write just ended (STOP or repeated START): hand it
419 // to onReceive(), as AVR does even for a write of no bytes
420 if ( t->in_write )
421 {
422 size_t count = t->rx_count;
423
424 // Its own buffer, not rx_buf: this can land in the middle of a
425 // requestFrom() of our own, which reads into rx_buf
426 memcpy( t->received, t->rx, count );
427 rx_data = t->received;
428 rx_index = 0;
429 rx_size = count;
430 t->in_write = false;
431
432 if ( user_onReceive )
433 user_onReceive( (int)count );
434 }
435 return;
436 }
437
438 uint8_t s = base->S;
439
440 if ( !( s & I2C_S_IICIF_MASK ) )
441 return;
442
443 base->S = I2C_S_IICIF_MASK;
444
445 if ( base->C1 & I2C_C1_MST_MASK ) // a controller transfer of our own
446 return;
447
448 if ( s & I2C_S_ARBL_MASK )
449 {
450 base->S = I2C_S_ARBL_MASK;
451 if ( !( s & I2C_S_IAAS_MASK ) )
452 return;
453 }
454
455 if ( s & I2C_S_IAAS_MASK )
456 {
457 if ( s & I2C_S_SRW_MASK ) // the controller reads
458 {
459 base->C1 |= I2C_C1_TX_MASK;
460 t->tx_size = 0;
461 t->tx_index = 0;
462 if ( user_onRequest )
463 {
464 t->in_request = true;
465 user_onRequest();
466 t->in_request = false;
467 }
468 base->D = ( t->tx_index < t->tx_size ) ? t->tx[ t->tx_index++ ] : t->fill;
469 }
470 else // the controller writes
471 {
472 base->C1 &= (uint8_t)~( I2C_C1_TX_MASK | I2C_C1_TXAK_MASK );
473 t->in_write = true;
474 t->rx_count = 0;
475 t->rx_full = false;
476 (void)base->D; // lets go of SCL
477 }
478 return;
479 }
480
481 if ( base->C1 & I2C_C1_TX_MASK ) // sending to the controller
482 {
483 if ( s & I2C_S_RXAK_MASK ) // it NAKed: the read is over
484 {
485 base->C1 &= (uint8_t)~I2C_C1_TX_MASK;
486 (void)base->D;
487 }
488 else // past the reply: 0xFF, as on AVR
489 base->D = ( t->tx_index < t->tx_size ) ? t->tx[ t->tx_index++ ] : t->fill;
490 return;
491 }
492
493 // Receiving. TXAK is the ACK of the byte after this one: NAK it once
494 // the buffer is full (a NAKed byte still arrives here, and is dropped)
495 uint8_t b = base->D;
496
497 if ( t->rx_count < sizeof( t->rx ) )
498 t->rx[ t->rx_count++ ] = b;
499 else
500 t->rx_full = true;
501
502 if ( t->rx_count >= sizeof( t->rx ) )
503 base->C1 |= I2C_C1_TXAK_MASK;
504}

507void TwoWire::start( int baud )
508{
509 baudrate = baud;
510
511 bool is_i3c = on_i3c_pins();
512
513 if ( !i2c )
514 {
515#if !defined( CPU_MCXC444VLH ) // LPI2C and I3C; FRDM-MCXC444 has the Kinetis I2C
516 if ( is_i3c )
517 {
518 I3C *i3c;
519 i3c = new I3C( _sda, _scl );
520
521#ifdef CPU_MCXN947VDF
522 /*
523 * At least one native-I3C-mode bus operation needs to happen on
524 * this bus before switching it down to legacy I2C_MODE -- on
525 * FRDM-MCXN947, without this, I2C-mode reads on this bus
526 * reliably NAK on the address phase (writes still succeed).
527 * Confirmed via a controlled A/B test on real hardware (GDB,
528 * then reflashed with/without this line): consistently fails
529 * without it, consistently works with it. Root cause not fully
530 * understood -- per the I3C spec the master's pull-up assist
531 * should already be active during bus-free/START/address-header
532 * regardless of prior traffic, so this looks like an SDK/
533 * silicon init-ordering quirk (I3C_MasterInit() alone may not
534 * be enough to arm the pull-up-assist state machine; one real
535 * START/STOP cycle might be needed first) rather than a spec
536 * violation we're intentionally working around. This priming
537 * broadcast is expected to itself fail (kStatus_I3C_WriteAbort,
538 * nothing has a dynamic address yet) -- only the bus activity
539 * matters. Matches the flow NXP's own P3T1755_FRDM_MCXN947_demo_DAA
540 * reference example always does (RSTDAA before any I2C-mode
541 * access).
542 *
543 * Confirmed NOT needed on A153 (its I3C_SDA/I3C_SCL are the
544 * same physical pins as the general-purpose Arduino I2C
545 * connector, which has real populated pull-up resistors --
546 * unlike N947's dedicated onboard-sensor-only I3C bus, whose
547 * fixed pull-ups are DNP on the schematic). Scoped to just
548 * this chip rather than done unconditionally, so boards that
549 * don't need it don't get a mystery extra I2C-bus transaction
550 * on every Wire.begin().
551 */
552 i3c->ccc_broadcast( CCC::BROADCAST_RSTDAA, nullptr, 0 );
553#endif // CPU_MCXN947VDF
554
555 i3c->mode( I3C::MODE::I2C_MODE );
556 i2c = i3c;
557 }
558 else
559#endif
560 {
561 i2c = new I2C( _sda, _scl );
562
563 // Internal pull-ups on, as AVR's twi_init() and UNO R4's
564 // begin() do: FRDM-MCXA153 has no pull-up resistors on D18/D19,
565 // so without them a bus with nothing else on it floats, reads
566 // low, and the first transfer hangs or finds the bus busy.
567 // Harmless next to external pull-ups, which set the rise time.
568 i2c->pullup( true );
569 }
570 }
571
572 /*
573 * I2C(sda, scl, no_hw=true) — the base-class constructor I3C delegates
574 * to — skips hardware init entirely (`if (no_hw) return;`), leaving
575 * I2C::unit_base uninitialized. I3C's own
576 * frequency(uint32_t,uint32_t,uint32_t)/frequency(void) only hide, not
577 * override, I2C::frequency(uint32_t) (different signature), so calling
578 * the generic i2c->frequency(baudrate) for an I3C instance would
579 * dispatch to I2C::frequency() and dereference that uninitialized
580 * unit_base -> BusFault. Route to I3C's own overload instead.
581 */
582#if !defined( CPU_MCXC444VLH ) // LPI2C and I3C; FRDM-MCXC444 has the Kinetis I2C
583 if ( is_i3c )
584 static_cast<I3C *>( i2c )->frequency( baudrate, 0, 0 );
585 else
586 {
587 i2c->pin_low_timeout( timeout_us );
588 i2c->frequency( baudrate );
589 }
590#else
591 (void)is_i3c;
592 i2c->pin_low_timeout( timeout_us ); // the Kinetis I2C's SCL low timeout
593 i2c->frequency( baudrate );
594#endif
595
596 i2c->err_callback( nullptr );
597
598 if ( target && target->active )
599 target_arm();
600}

634size_t TwoWire::write( uint8_t data )
635{
636 if ( target && target->in_request ) // onRequest(): the reply

676uint8_t TwoWire::requestFrom( const uint8_t address, const size_t length, bool stop )
677{
678 size_t n = ( length < sizeof( rx_buf ) ) ? length : sizeof( rx_buf );
679
680 rx_data = rx_buf;
681 rx_index = 0;
682 rx_size = 0;
683
684 status_t r = i2c->read( address, rx_buf, n, stop );
685
686 check_timeout( r );
687 if ( r )
688 return 0;
689
690 // Set again after the read: this board's own target can take a write
691 // (and hand it to onReceive(), moving these) partway through it
692 rx_data = rx_buf;
693 rx_index = 0;
694 rx_size = n;
695 return n;
696}

698uint8_t TwoWire::requestFrom( uint8_t address, uint8_t quantity, uint32_t iaddress, uint8_t isize, uint8_t sendStop )
699{
700 if ( isize > 0 )
701 {
702 if ( isize > 3 )
703 isize = 3;
704
705 beginTransmission( address );
706 while ( isize-- > 0 )
707 write( (uint8_t)( iaddress >> ( isize * 8 ) ) );
708
709 if ( endTransmission( false ) )
710 {
711 rx_data = rx_buf;
712 rx_index = 0;
713 rx_size = 0;
714 return 0;
715 }
716 }
717
718 return requestFrom( address, (size_t)quantity, (bool)sendStop );
