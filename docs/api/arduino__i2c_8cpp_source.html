17/*
18 * Target (slave) mode, on the SDK's LPI2C_Slave* transfer API.
19 *
20 * An LPI2C's master and slave halves run independently on the same pins,
21 * so a board can be a controller and a target at once, even addressing
22 * its own target. Four things the SDK leaves to the caller, each seen on
23 * hardware:
24 *
25 * - FRDM-MCXA153's SDK (fsl_lpi2c 2.5.4) turns the master off when it
26 * arms the slave (LPI2C_SlaveTransferNonBlocking(): "Enable the slave
27 * function and disable the master function"); FRDM-MCXN947's (2.2.4)
28 * doesn't. The master is turned back on after arming, or every later
29 * controller transfer waits forever on a disabled master.
30 * - When the reply a controller reads runs out, the SDK sends nothing and
31 * the slave stretches SCL forever. 0xFF is sent from there on, as AVR
32 * does.
33 * - A write longer than the buffer: the byte that doesn't fit is NAKed
34 * (ACKSTALL, deciding each ACK in the TransmitAck event), and still
35 * read into a scratch byte, since an unread byte leaves the receive
36 * flag set and the interrupt firing.
37 * - Deleting the master (end(), or the reset after a timeout) gates the
38 * LPI2C's clock (A153) or de-initializes its FlexComm (N947), which
39 * takes the slave with it, so start() re-arms the target after
40 * re-creating the master.
41 */
42struct WireTarget
43{
44 lpi2c_slave_handle_t handle;
45 uint8_t address;
46 bool active;
47 bool reading; // the controller is reading, not writing
48 bool rx_given; // rx[] handed to the SDK for this write
49 bool rx_full; // ... and it filled up
50 bool tx_given; // onRequest() already asked for this read
51 bool in_request; // inside onRequest(): write() fills tx[]
52 uint8_t rx[ WIRE_BUFFER_SIZE ]; // the SDK receives into this
53 uint8_t received[ WIRE_BUFFER_SIZE ]; // read() reads a finished write here
54 uint8_t tx[ WIRE_BUFFER_SIZE ];
55 size_t tx_size;
56 uint8_t fill; // what a controller gets past the reply
57 uint8_t sink; // where a NAKed byte is read to
58};

82void TwoWire::begin( uint8_t address )
83{
84 if ( on_i3c_pins() )
85 panic( "Wire1: target (slave) mode isn't supported on Wire1, which runs on the I3C peripheral. Use Wire" );
86#ifdef CPU_MCXN947VDF
87 /*
88 * LPI2C3 (FlexComm3), behind Wire2, never sees the bus as a target:
89 * its slave status stays 0 through a transfer addressed to it, while
90 * its own master side on the same pins works, and LPI2C2 set up
91 * identically (Wire) works. Every slave register matches LPI2C2's.
92 * Cause not found; checked with Wire and Wire2 jumpered together.
93 */
94 if ( _sda == MB_SDA && _scl == MB_SCL )
95 panic( "Wire2: target (slave) mode doesn't work on Wire2 (LPI2C3 on FRDM-MCXN947). Use Wire" );
96#endif
97 if ( address > 127 )
98 panic( "Wire: begin(address) given an address over 127" );
99
100 if ( !target )
101 target = new WireTarget();
102 target->address = address;
103 target->active = true;
104
105 start( 100000 ); // a controller too, as on AVR; start() arms the target
106}

131void TwoWire::target_arm( void )
132{
133 LPI2C_Type *base = i2c->lpi2c_base();
134 WireTarget *t = target;
135
136 t->reading = false;
137 t->rx_given = false;
138 t->rx_full = false;
139 t->tx_given = false;
140 t->in_request = false;
141 t->fill = 0xFF;
142
143 lpi2c_slave_config_t cfg;
144 LPI2C_SlaveGetDefaultConfig( &cfg );
145 cfg.address0 = t->address;
146 cfg.sclStall.enableAck = true; // ACKSTALL, so a full buffer can NAK
147
148 LPI2C_SlaveInit( base, &cfg, i2c->lpi2c_clock() );
149 LPI2C_SlaveTransferCreateHandle( base, &t->handle, target_callback, this );
150 LPI2C_SlaveTransferNonBlocking( base, &t->handle,
151 kLPI2C_SlaveAddressMatchEvent | kLPI2C_SlaveTransmitAckEvent |
152 kLPI2C_SlaveRepeatedStartEvent | kLPI2C_SlaveCompletionEvent );
153
154 LPI2C_MasterEnable( base, true ); // A153's SDK turned it off (see top)
155 i2c->wait_bus_idle();
156}

174void TwoWire::target_event( void *p )
175{
176 lpi2c_slave_transfer_t *x = static_cast<lpi2c_slave_transfer_t *>( p );
177 WireTarget *t = target;
178
179 switch ( x->event )
180 {
181 case kLPI2C_SlaveAddressMatchEvent:
182 t->reading = x->receivedAddress & 1u;
183 t->rx_given = false;
184 t->rx_full = false;
185 t->tx_given = false;
186 x->data = nullptr;
187 x->dataSize = 0;
188 break;
189
190 case kLPI2C_SlaveTransmitAckEvent:
191 {
192 // Before the byte is stored: ACK it only if it will fit
193 size_t stored = t->rx_given ? sizeof( t->rx ) - x->dataSize : 0;
194 LPI2C_SlaveTransmitAck( i2c->lpi2c_base(), !t->rx_full && stored < sizeof( t->rx ) );
195 break;
196 }
197
198 case kLPI2C_SlaveReceiveEvent:
199 if ( !t->rx_given )
200 {
201 x->data = t->rx;
202 x->dataSize = sizeof( t->rx );
203 t->rx_given = true;
204 }
205 else
206 {
207 t->rx_full = true; // NAKed above; read it and drop it
208 x->data = &t->sink;
209 x->dataSize = 1;
210 }
211 break;
212
213 case kLPI2C_SlaveTransmitEvent:
214 if ( !t->tx_given )
215 {
216 t->tx_given = true;
217 t->tx_size = 0;
218 if ( user_onRequest )
219 {
220 t->in_request = true;
221 user_onRequest();
222 t->in_request = false;
223 }
224 if ( t->tx_size )
225 {
226 x->data = t->tx;
227 x->dataSize = t->tx_size;
228 break;
229 }
230 }
231 x->data = &t->fill; // past the reply: 0xFF, as on AVR
232 x->dataSize = 1;
233 break;
234
235 case kLPI2C_SlaveRepeatedStartEvent:
236 case kLPI2C_SlaveCompletionEvent:
237 // A controller's write just ended: hand it to onReceive(), as
238 // AVR does even for a write of no bytes
239 if ( !t->reading && x->completionStatus == kStatus_Success )
240 {
241 size_t n = t->rx_full ? sizeof( t->rx ) : t->rx_given ? x->transferredCount : 0;
242
243 // Its own buffer, not rx_buf: this can land in the middle of
244 // a requestFrom() of our own, which reads into rx_buf
245 memcpy( t->received, t->rx, n );
246 rx_data = t->received;
247 rx_index = 0;
248 rx_size = n;
249
250 if ( user_onReceive )
251 user_onReceive( (int)n );
252 }
253 t->reading = false;
254 t->rx_given = false;
255 t->rx_full = false;
256 break;
257
258 default:
259 break;
260 }
261}

263void TwoWire::start( int baud )
264{
265 baudrate = baud;
266
267 bool is_i3c = on_i3c_pins();
268
269 if ( !i2c )
270 {
271 if ( is_i3c )
272 {
273 I3C *i3c;
274 i3c = new I3C( _sda, _scl );
275
276#ifdef CPU_MCXN947VDF
277 /*
278 * At least one native-I3C-mode bus operation needs to happen on
279 * this bus before switching it down to legacy I2C_MODE -- on
280 * FRDM-MCXN947, without this, I2C-mode reads on this bus
281 * reliably NAK on the address phase (writes still succeed).
282 * Confirmed via a controlled A/B test on real hardware (GDB,
283 * then reflashed with/without this line): consistently fails
284 * without it, consistently works with it. Root cause not fully
285 * understood -- per the I3C spec the master's pull-up assist
286 * should already be active during bus-free/START/address-header
287 * regardless of prior traffic, so this looks like an SDK/
288 * silicon init-ordering quirk (I3C_MasterInit() alone may not
289 * be enough to arm the pull-up-assist state machine; one real
290 * START/STOP cycle might be needed first) rather than a spec
291 * violation we're intentionally working around. This priming
292 * broadcast is expected to itself fail (kStatus_I3C_WriteAbort,
293 * nothing has a dynamic address yet) -- only the bus activity
294 * matters. Matches the flow NXP's own P3T1755_FRDM_MCXN947_demo_DAA
295 * reference example always does (RSTDAA before any I2C-mode
296 * access).
297 *
298 * Confirmed NOT needed on A153 (its I3C_SDA/I3C_SCL are the
299 * same physical pins as the general-purpose Arduino I2C
300 * connector, which has real populated pull-up resistors --
301 * unlike N947's dedicated onboard-sensor-only I3C bus, whose
302 * fixed pull-ups are DNP on the schematic). Scoped to just
303 * this chip rather than done unconditionally, so boards that
304 * don't need it don't get a mystery extra I2C-bus transaction
305 * on every Wire.begin().
306 */
307 i3c->ccc_broadcast( CCC::BROADCAST_RSTDAA, nullptr, 0 );
308#endif // CPU_MCXN947VDF
309
310 i3c->mode( I3C::MODE::I2C_MODE );
311 i2c = i3c;
312 }
313 else
314 {
315 i2c = new I2C( _sda, _scl );
316
317 // Internal pull-ups on, as AVR's twi_init() and UNO R4's
318 // begin() do: FRDM-MCXA153 has no pull-up resistors on D18/D19,
319 // so without them a bus with nothing else on it floats, reads
320 // low, and the first transfer hangs or finds the bus busy.
321 // Harmless next to external pull-ups, which set the rise time.
322 i2c->pullup( true );
323 }
324 }
325
326 /*
327 * I2C(sda, scl, no_hw=true) — the base-class constructor I3C delegates
328 * to — skips hardware init entirely (`if (no_hw) return;`), leaving
329 * I2C::unit_base uninitialized. I3C's own
330 * frequency(uint32_t,uint32_t,uint32_t)/frequency(void) only hide, not
331 * override, I2C::frequency(uint32_t) (different signature), so calling
332 * the generic i2c->frequency(baudrate) for an I3C instance would
333 * dispatch to I2C::frequency() and dereference that uninitialized
334 * unit_base -> BusFault. Route to I3C's own overload instead.
335 */
336 if ( is_i3c )
337 static_cast<I3C *>( i2c )->frequency( baudrate, 0, 0 );
338 else
339 {
340 i2c->pin_low_timeout( timeout_us );
341 i2c->frequency( baudrate );
342 }
343
344 i2c->err_callback( nullptr );
345
346 if ( target && target->active )
347 target_arm();
348}

380size_t TwoWire::write( uint8_t data )
381{
382 if ( target && target->in_request ) // onRequest(): the reply
383 {
384 if ( target->tx_size >= sizeof( target->tx ) )
385 {
386 setWriteError();
387 return 0;
388 }
389 target->tx[ target->tx_size++ ] = data;
390 return 1;
391 }
392
393 if ( tx_size >= sizeof( tx_buf ) )
394 {
395 setWriteError();
396 return 0;
397 }
398
399 tx_buf[ tx_size++ ] = data;
400
401 return 1;
402}

422uint8_t TwoWire::requestFrom( const uint8_t address, const size_t length, bool stop )
423{
424 size_t n = ( length < sizeof( rx_buf ) ) ? length : sizeof( rx_buf );
425
426 rx_data = rx_buf;
427 rx_index = 0;
428 rx_size = 0;
429
430 status_t r = i2c->read( address, rx_buf, n, stop );
431
432 check_timeout( r );
433 if ( r )
434 return 0;
435
436 // Set again after the read: this board's own target can take a write
437 // (and hand it to onReceive(), moving these) partway through it
438 rx_data = rx_buf;
439 rx_index = 0;
440 rx_size = n;
441 return n;
442}

444uint8_t TwoWire::requestFrom( uint8_t address, uint8_t quantity, uint32_t iaddress, uint8_t isize, uint8_t sendStop )
445{
446 if ( isize > 0 )
447 {
448 if ( isize > 3 )
449 isize = 3;
450
451 beginTransmission( address );
452 while ( isize-- > 0 )
453 write( (uint8_t)( iaddress >> ( isize * 8 ) ) );
454
455 if ( endTransmission( false ) )
456 {
457 rx_data = rx_buf;
458 rx_index = 0;
459 rx_size = 0;
460 return 0;
461 }
462 }
463
464 return requestFrom( address, (size_t)quantity, (bool)sendStop );
465}
