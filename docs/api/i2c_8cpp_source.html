192void I2C::frequency( uint32_t frequency )
193{
194#if CPU_MCXC444VLH
195 I2C_MasterSetBaudRate( unit_base, I2C_MASTER_CLOCK_FREQUENCY, frequency );
196#else
197 /* Clear the glitch filters before handing over to the SDK.
198 *
199 * LPI2C_MasterSetBaudRate() only ever raises FILTSDA, never lowers it:
200 * the block that writes it is guarded by
201 * (sourceClock / baudRate / 20) > (divider + 2)
202 * which is false for fast baud rates, so switching *down* in period
203 * (i.e. up in speed) skips the write entirely and leaves whatever a
204 * previous, slower setting left behind.
205 *
206 * Worked example on this hardware (12MHz source), all three values
207 * confirmed by reading MCFGR2 back on a real board:
208 * begin() @100kHz -> 12e6/100e3/20 = 6 > 3 -> FILTSDA = 6-1-2 = 3
209 * setClock(10kHz) -> 12e6/10e3/20 = 60 > 18 -> FILTSDA = 60-16-2 = 42,
210 * truncated to the 4-bit field as 42 & 0xF = 10
211 * setClock(400kHz)-> 12e6/400e3/20 = 1 > 3? no -> skipped, stays 10
212 *
213 * FILTSDA 10 at 12MHz is ~833ns of SDA filtering, against a 2.5us bit
214 * time at 400kHz. On a logic analyzer that showed up as the address
215 * phase being cut short after three SCL pulses and restarted with a
216 * repeated START, over and over -- and, before the STOP handling above
217 * was fixed, as an outright hang. It only ever reproduced when a fast
218 * rate followed a slow one, which is exactly what this guard predicts.
219 *
220 * Zeroing both filter fields first makes each call recompute from a
221 * clean slate: the SDK then writes the right value when it wants one,
222 * and leaving it at 0 when it skips the block is the intended "no
223 * extra filtering needed at this speed" result rather than stale state.
224 *
225 * MCFGR2 is only writable with the module disabled -- the SDK writes it
226 * inside its own disable/restore window for the same reason -- so do
227 * the same here, and hand the module back in the state it was found in
228 * so LPI2C_MasterSetBaudRate()'s own save/restore still sees the truth.
229 */
230 bool was_enabled = ( unit_base->MCR & LPI2C_MCR_MEN_MASK ) != 0U;
231
232 LPI2C_MasterEnable( unit_base, false );
233 unit_base->MCFGR2 &= ~( LPI2C_MCFGR2_FILTSDA_MASK | LPI2C_MCFGR2_FILTSCL_MASK );
234 LPI2C_MasterEnable( unit_base, was_enabled );
235
236 LPI2C_MasterSetBaudRate( unit_base, LPI2C_MASTER_CLOCK_FREQUENCY, frequency );
237#endif
238}

315/* Emit a STOP, working around LPI2C_MasterStop()'s own error handling.
316 *
317 * LPI2C_MasterStop() opens with LPI2C_MasterWaitForTxReady(), which runs
318 * LPI2C_MasterCheckAndClearError(). When the target NAKed, that check
319 * consumes the NAK and makes STOP return kStatus_LPI2C_Nak *before* it
320 * ever writes the stop command -- so the transaction is never terminated
321 * on the bus.
322 *
323 * This is not a rare corner: writing one byte to an absent address, the
324 * NAK is not yet visible at either of the earlier checks in write_core()
325 * (LPI2C_MasterStart() only queues the address, the FIFO-drain loop only
326 * waits for it to leave the FIFO, and LPI2C_MasterSend() can still place
327 * the data byte before the bus has answered), so LPI2C_MasterStop() is
328 * where it consistently surfaces. Measured on real hardware: 1500 of 1500
329 * probe transfers, with a logic analyzer showing each transaction's STOP
330 * missing and only appearing once the *next* transaction started.
331 *
332 * CheckAndClearError has cleared the flag by the time it returns, so a
333 * second call goes through and actually emits the STOP. Only retried for
334 * NAK: the other error flags (arbitration lost, pin-low timeout) mean the
335 * bus itself is in trouble, and LPI2C_MasterStop() has no timeout of its
336 * own to fall back on (I2C_RETRY_TIMES is 0), so retrying those could
337 * block forever. The caller still gets the original status.
338 */
339static status_t stop_with_nak_recovery( LPI2C_Type *unit_base )
340{
341 status_t r = LPI2C_MasterStop( unit_base );
342
343 if ( r == kStatus_LPI2C_Nak )
344 LPI2C_MasterStop( unit_base );
345
346 return r;
347}

349status_t I2C::write_core( uint8_t address, const uint8_t *dp, int length, bool stop )
350{
351 status_t reVal = kStatus_Fail;
352 size_t txCount = 0xFFU;
353
354 if ( kStatus_Success == (reVal = LPI2C_MasterStart( unit_base, address, kLPI2C_Write)) )
355 {
356 LPI2C_MasterGetFifoCounts( unit_base, NULL, &txCount );
357 while ( txCount )
358 {
359 LPI2C_MasterGetFifoCounts( unit_base, NULL, &txCount );
360 }
361
362 uint32_t status = LPI2C_MasterGetStatusFlags( unit_base );
363
364 if ( status & kLPI2C_MasterNackDetectFlag )
365 {
366 // Address-phase NAK, on the occasions it is already visible
367 // here (on a slow enough SCL it can be). Clear the error first:
368 // leaving the flag set would make the STOP below bail out without
369 // emitting anything -- see stop_with_nak_recovery() above, which
370 // is the same trap reached from the other direction.
371 (void)LPI2C_MasterCheckAndClearError( unit_base, status );
372 LPI2C_MasterStop( unit_base );
373
374 return kStatus_LPI2C_Nak;
375 }
376
377 reVal = LPI2C_MasterSend( unit_base, (uint8_t *)dp, length );
378 if (reVal != kStatus_Success)
379 {
380 if ( reVal == kStatus_LPI2C_Nak )
381 {
382 LPI2C_MasterStop( unit_base );
383 }
384 return reVal;
385 }
386
387 if ( stop )
388 {
389 reVal = stop_with_nak_recovery( unit_base );
390 if ( reVal != kStatus_Success )
391 {
392 return reVal;
393 }
394 }
395 }
396 return reVal;
397}

508void I2C::scan( uint8_t start, uint8_t last, bool *result )
509{
510 // result[] is always indexed by the full address (0..last), even
511 // though only [start, last] is actually pinged -- addresses below
512 // start are marked "not found" rather than left uninitialized, so
513 // scan(start, last)'s display loop below can print the whole table.
514 for ( uint8_t i = 0; i <= last; i++ )
515 result[i] = ( i >= start ) ? ping( i ) : false;
516}

518void I2C::scan( uint8_t start, uint8_t last )
519{
520 bool result[ 128 ];
521
522 scan( start, last, result );
523
524 printf( "\r\nI2C scan result (in range of 0x%02X ~ 0x%02X)\r\n ", start, last );
525 for ( uint8_t x = 0; x < 16; x++ )
526 printf( " x%01X", x );
527
528 // <= last, not < last -- last is the last address to include (see the
529 // scan(uint8_t last) overload's doc comment), so excluding it here
530 // silently dropped the requested upper-bound address from the table.
531 for ( uint8_t i = 0; i <= last; i++ )
532 {
533 if ( !( i % 16) )
534 printf( "\r\n%01Xx:", i / 16 );
535
536 if ( result[ i ] )
537 printf( " %02X", i );
538 else
539 printf( " --" );
540 }
541 printf( "\r\n\r\n" );
542}
