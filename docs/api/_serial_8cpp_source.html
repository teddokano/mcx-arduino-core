1/*
2 * Serial.cpp - Mbed-compatible Serial class for r01lib
3 *
4 * Supports:
5 * FRDM-MCXC444 (CPU_MCXC444VLH) LPUART0/1
6 * FRDM-MCXA153 (CPU_MCXA153VLH) LPUART0/1/2
7 * FRDM-MCXA156 (CPU_MCXA156VLL) LPUART0
8 * FRDM-MCXN236 (CPU_MCXN236VDF) LP_FLEXCOMM4 (LPUART4)
9 * FRDM-MCXN947 (CPU_MCXN947VDF) LP_FLEXCOMM4 (LPUART4)
10 *
11 * Design note:
12 * Both TX and RX use software ring buffers driven directly by the LPUART
13 * interrupt, without the SDK Transfer API.
14 *
15 * RX path:
16 * kLPUART_RxDataRegFullInterruptEnable fires ->
17 * ISR reads one byte -> pushes into _rx_buf -> calls _rx_callback.
18 *
19 * TX path:
20 * putc/write/printf push bytes into _tx_buf and enable
21 * kLPUART_TxDataRegEmptyInterruptEnable ->
22 * ISR pops one byte -> writes to hardware.
23 * When _tx_buf is empty the ISR disables TX interrupt and calls
24 * _tx_callback (if attached).
25 *
26 * @author Based on r01lib pattern by Tedd OKANO
27 * Released under the MIT License
28 */

352Serial::Serial( int tx, int rx, int baud )
353 : Obj( true ),
354 _base( nullptr ), _config{}, _clk_freq( 0U ),
355 _instance( 0U ), _tx_mux( kPORT_MuxAlt2 ), _rx_mux( kPORT_MuxAlt2 ), _irqn( NotAvail_IRQn ),
356 _tx_pin( tx ), _rx_pin( rx ),
357 _rx_head( 0 ), _rx_tail( 0 ),
358 _tx_head( 0 ), _tx_tail( 0 ),
359 _rx_callback( nullptr ), _tx_callback( nullptr )
360{
361 resolve_pins( tx, rx );
362
363 if ( !_base )
364 {
365 panic( "Serial: unsupported TX/RX pin combination" );
366 return;
367 }
368
369 _register_instance();
370
371 /*
372 * Note what is deliberately NOT done here: claiming the TX/RX pins.
373 * apply_pin_mux() does that, and begin() is what calls it.
374 *
375 * The global Serial/Serial1 instances are constructed during static
376 * initialization, before main(), whether or not the sketch ever
377 * begin()s them. Muxing the pins here meant Serial1 seized its pins
378 * unconditionally at that point -- and on FRDM-MCXN947 Serial1 sits
379 * on MB_TX/MB_RX, the very same physical pins as I3C_SCL/I3C_SDA.
380 * A sketch with its own global I3C object was therefore in an
381 * unwinnable race: static init order across translation units is
382 * unspecified, so whichever constructor ran last owned the pins. When
383 * Serial1 lost that race nothing happened; when it won, the I3C
384 * peripheral was left disconnected from the bus, SDA/SCL idled high
385 * with no START ever generated, and every transfer returned
386 * kStatus_I3C_Nak. Wire1 escaped this only because TwoWire::begin()
387 * constructs its I3C lazily, long after static init.
388 *
389 * Deferring to begin() also just matches how Arduino is supposed to
390 * behave -- a port you never begin() has no business holding pins.
391 */
392
393 // Clock must be set up before reset release and LPUART_Init
394 _setup_clock();
395 _release_reset();
396
397 LPUART_GetDefaultConfig( &_config );
398 _config.baudRate_Bps = (uint32_t)baud;
399 _config.enableTx = true;
400 _config.enableRx = true;
401
402 _clk_freq = _get_clk_freq();
403 LPUART_Init( _base, &_config, _clk_freq );
404}

406void Serial::apply_pin_mux( void )
407{
408 if ( !_base )
409 return;
410
411 DigitalInOut tx_io( (uint8_t)_tx_pin );
412 DigitalInOut rx_io( (uint8_t)_rx_pin );
413
414 tx_io.pin_mux( (int)_tx_mux );
415 rx_io.pin_mux( (int)_rx_mux );
416
417 // RX pins not otherwise pre-configured at boot (pin_mux.c) reset
418 // with the digital input buffer disabled -- MUX alone isn't enough,
419 // the peripheral never sees the incoming signal without this.
420 rx_io.input_buffer( true );
421
422 uint8_t pins[ 2 ] = { (uint8_t)_tx_pin, (uint8_t)_rx_pin };
423 pin_registry_note( this, "Serial", pins, 2, (uint8_t)_tx_mux );
424}

446void Serial::tx_enqueue( uint8_t b )
447{
448 uint16_t next;
449 do {
450 next = (uint16_t)(( _tx_head + 1U ) & ( TX_RING_BUF_SIZE - 1U ));
451 } while ( next == _tx_tail ); // spin if TX buffer full
452
453 _tx_buf[ _tx_head ] = b;
454 _tx_head = next;
455
456 EnableIRQ( _irqn );
457 LPUART_EnableInterrupts( _base, kLPUART_TxDataRegEmptyInterruptEnable );
458}

460void Serial::update_irq_enables( void )
461{
462 if ( _rx_callback )
463 {
464 EnableIRQ( _irqn );
465 LPUART_EnableInterrupts( _base, kLPUART_RxDataRegFullInterruptEnable );
466 }
467 else
468 {
469 LPUART_DisableInterrupts( _base, kLPUART_RxDataRegFullInterruptEnable );
470 if ( _tx_head == _tx_tail )
471 DisableIRQ( _irqn );
472 }

492void Serial::_irq_handler( void )
493{
494 uint32_t flags = LPUART_GetStatusFlags( _base );
495
496 // ---- RX: one byte received ----
497 if ( flags & kLPUART_RxDataRegFullFlag )
498 {
499 uint8_t byte = LPUART_ReadByte( _base );
500 uint16_t next = (uint16_t)(( _rx_head + 1U ) & ( RX_RING_BUF_SIZE - 1U ));
501
502 if ( next != _rx_tail ) // drop silently if ring buffer full
503 {
504 _rx_buf[ _rx_head ] = byte;
505 _rx_head = next;
506 }
507
508 if ( _rx_callback )
509 _rx_callback();
510 }
511
512 // ---- TX: register empty — send next byte from ring buffer ----
513 if ( flags & kLPUART_TxDataRegEmptyFlag )
514 {
515 if ( _tx_head != _tx_tail )
516 {
517 LPUART_WriteByte( _base, _tx_buf[ _tx_tail ] );
518 _tx_tail = (uint16_t)(( _tx_tail + 1U ) & ( TX_RING_BUF_SIZE - 1U ));
519 }
520 else
521 {
522 // TX buffer drained: disable interrupt, fire callback
523 LPUART_DisableInterrupts( _base, kLPUART_TxDataRegEmptyInterruptEnable );
524 if ( _tx_callback )
525 _tx_callback();
526 }
527 }
528
529 // ---- Overrun: clear flag ----
530 if ( flags & kLPUART_RxOverrunFlag )
531 LPUART_ClearStatusFlags( _base, kLPUART_RxOverrunFlag );
532}

534void Serial::baud( int baudrate )
535{
536 LPUART_DisableInterrupts( _base,
537 kLPUART_RxDataRegFullInterruptEnable |
538 kLPUART_TxDataRegEmptyInterruptEnable );
539
540 _config.baudRate_Bps = (uint32_t)baudrate;
541 LPUART_Deinit( _base );
542 LPUART_Init( _base, &_config, _clk_freq );
543
544 update_irq_enables();
545
546 if ( _tx_head != _tx_tail )
547 LPUART_EnableInterrupts( _base, kLPUART_TxDataRegEmptyInterruptEnable );
548}

556int Serial::getc( void )
557{
558 if ( _rx_callback )
559 {
560 if ( _rx_head == _rx_tail )
561 return -1;
562
563 uint8_t b = _rx_buf[ _rx_tail ];
564 _rx_tail = (uint16_t)(( _rx_tail + 1U ) & ( RX_RING_BUF_SIZE - 1U ));
565 return (int)b;
566 }
567 else
