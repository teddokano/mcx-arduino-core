74extern "C" void CTIMER2_IRQHandler( void )
75{
76 CTIMER_ClearStatusFlags( CTIMER2, CTIMER_GetStatusFlags( CTIMER2 ) );
77
78 uint32_t start = micros();
79
80 ETHERNET_DriverIRQHandler(); // frees sent frames; the ENET interrupt itself stays off
81 ethernetif_input( &enet_netif );
82 sys_check_timeouts();
83
84 uint32_t t = micros() - start;
85 if ( t > tick_max_us )
86 tick_max_us = t;
87
88 SDK_ISR_EXIT_BARRIER;
89}

91static void tick_start()
92{
93 CLOCK_AttachClk( kFRO12M_to_CTIMER2 );
94 CLOCK_SetClkDiv( kCLOCK_DivCtimer2Clk, 1u );
95
96 ctimer_config_t cfg;
97 CTIMER_GetDefaultConfig( &cfg );
98 CTIMER_Init( CTIMER2, &cfg );
99
100 ctimer_match_config_t match = {};
101 match.matchValue = CLOCK_GetCTimerClkFreq( 2 ) / 1000 - 1;
102 match.enableCounterReset = true;
103 match.enableInterrupt = true;
104 match.outControl = kCTIMER_Output_NoAction;
105 CTIMER_SetupMatch( CTIMER2, kCTIMER_Match_0, &match );
106
107 // the lowest priority, as SysTick's (millis()): the peripherals' interrupts come first
108 NVIC_SetPriority( CTIMER2_IRQn, ( 1u << __NVIC_PRIO_BITS ) - 1 );
109
110 running = true;
111 if ( lock_depth == 0 )
112 NVIC_EnableIRQ( CTIMER2_IRQn );
113 CTIMER_StartTimer( CTIMER2 );
114}

162// ELS (the chip's security block): wait for a command, false on an error or if it never ends
163static bool els_wait()
164{
165 for ( uint32_t i = 0; ELS->ELS_STATUS & S50_ELS_STATUS_ELS_BUSY_MASK; i++ )
166 if ( i > 1000000 )
167 return false;
168
169 if ( ELS->ELS_STATUS & S50_ELS_STATUS_ELS_ERR_MASK )
170 {
171 ELS->ELS_ERR_STATUS_CLR = S50_ELS_ERR_STATUS_CLR_ERR_CLR_MASK;
172 return false;
173 }
174 return true;
175}
176
177// A word from the ELS's PRNG, which the ELS seeds from its TRNG, or 0 if the
178// ELS doesn't come up. The steps are the SDK's ELS_PowerDownWakeupInit()
179// (els_pkc/src/platforms/mcxn/mcux_els.c): the clocks, the ELS enabled, and
180// if the PRNG isn't ready yet, a KDELETE of a key slot that holds no key,
181// which raises the entropy level and seeds the PRNG. A slot with a key in
182// it is never touched
183static uint32_t els_random()
184{
185 SYSCON->REF_CLK_CTRL_SET = SYSCON_REF_CLK_CTRL_SET_GDET_REFCLK_EN_SET_MASK | SYSCON_REF_CLK_CTRL_SET_TRNG_REFCLK_EN_SET_MASK;
186 CLOCK_EnableClock( kCLOCK_Css );
187
188 ELS->ELS_CTRL = S50_ELS_CTRL_ELS_EN_MASK;
189 if ( !els_wait() )
190 return 0;
191
192 if ( !( ELS->ELS_STATUS & S50_ELS_STATUS_PRNG_RDY_MASK ) )
193 {
194 const volatile uint32_t *ks = &ELS->ELS_KS0;
195 const uint32_t slots = 20; // ELS_KS0 to ELS_KS19
196 uint32_t i;
197
198 for ( i = 0; i < slots; i++ )
199 if ( !( ks[ i ] & S50_ELS_KS0_KS0_KACT_MASK ) )
200 break;
201 if ( i == slots )
202 return 0;
203
204 ELS->ELS_KIDX0 = i;
205 ELS->ELS_CMDCFG0 = 0;
206 ELS->ELS_CTRL = S50_ELS_CTRL_ELS_CMD( 11 ) // KDELETE
207 | S50_ELS_CTRL_ELS_START_MASK | S50_ELS_CTRL_ELS_EN_MASK
208 | S50_ELS_CTRL_BYTE_ORDER_MASK;
209 if ( !els_wait() )
210 return 0;
211 }
212
213 uint32_t r = ELS->ELS_PRNG_DATOUT;
214 if ( ELS->ELS_ERR_STATUS & S50_ELS_ERR_STATUS_PRNG_ERR_MASK ) // not enough entropy
215 {
216 ELS->ELS_ERR_STATUS_CLR = S50_ELS_ERR_STATUS_CLR_ERR_CLR_MASK;
217 return 0;
218 }
219 return r;
220}

222// @return true once the interface is up
223static bool interface_start( const uint8_t *mac )
224{
225 if ( !hardware_init() )
226 return false;
227
228 if ( running )
229 {
230 Lock lock;
231 if ( memcmp( mac_address, mac, 6 ) != 0 )
232 {
233 memcpy( mac_address, mac, 6 );
234 memcpy( enet_netif.hwaddr, mac, 6 );
235 ENET_SetMacAddr( ENET0, mac_address );
236 }
237 return true;
238 }
239
240 memcpy( mac_address, mac, 6 );
241 rand_state ^= ( (uint32_t)mac[ 2 ] << 24 | (uint32_t)mac[ 3 ] << 16 | (uint32_t)mac[ 4 ] << 8 | mac[ 5 ] ) ^ micros();
242 rand_state ^= els_random(); // before lwip_init(), which takes its first TCP and UDP ports from it
243 if ( rand_state == 0 )
244 rand_state = 0x2545F491;
245
246 phy_resource.read = mdio_read;
247 phy_resource.write = mdio_write;
248
249 enet_config.phyHandle = &phy_handle;
250 enet_config.phyAddr = PHY_ADDRESS;
251 enet_config.phyOps = &phylan8741_ops;
252 enet_config.phyResource = &phy_resource;
253 enet_config.srcClockHz = 50000000U;
254 memcpy( enet_config.macAddress, mac, 6 );
255
256 lwip_init();
257 netif_add( &enet_netif, NULL, NULL, NULL, &enet_config, ethernetif0_init, ethernet_input );
258 DisableIRQ( ETHERNET_IRQn ); // ENET_CreateHandler() enables it; the tick polls the driver instead
259 netif_set_default( &enet_netif );
260 netif_set_up( &enet_netif );
261
262 tick_start();
263 return true;
264}

266// a locally administered MAC address from the chip's 16-byte UUID
267static void mac_from_uuid( uint8_t *mac )
268{
269 uint8_t uuid[ FLASH_FFR_UUID_SIZE ] = {};
270 flash_config_t config = {};
271
272 if ( FLASH_Init( &config ) == kStatus_Success && FFR_Init( &config ) == kStatus_Success )
273 FFR_GetUUID( &config, uuid );
274
275 uint64_t h = 0xCBF29CE484222325ull; // FNV-1a
276 for ( unsigned i = 0; i < sizeof( uuid ); i++ )
277 {
278 h ^= uuid[ i ];
279 h *= 0x100000001B3ull;
280 }
281
282 mac[ 0 ] = 0x02; // locally administered, unicast
283 for ( int i = 1; i < 6; i++ )
284 mac[ i ] = (uint8_t)( h >> ( 8 * ( i - 1 ) ) );
285}

416void EthernetClass::begin( uint8_t *mac, IPAddress ip, IPAddress dns, IPAddress gateway, IPAddress subnet )
417{
418 if ( !interface_start( mac ) )
419 return;
420
421 {
422 Lock lock;
423 ip_addr_t a = to_lwip( ip );
424 ip_addr_t m = to_lwip( subnet );
425 ip_addr_t g = to_lwip( gateway );
426 ip_addr_t d = to_lwip( dns );
427
428 dhcp_stop( &enet_netif );
429 netif_set_addr( &enet_netif, ip_2_ip4( &a ), ip_2_ip4( &m ), ip_2_ip4( &g ) );
430 dns_setserver( 0, &d );
431 }
432
433 // Waits up to LINK_WAIT_MS for the PHY's auto-negotiation (~1.7 s), as a
434 // W5x00 has its link up from power-up: sketches test linkStatus() right
435 // after this. Asking the PHY here tells lwIP at once, not at its next
436 // 1.5 s poll, so a connect() just after doesn't find no route
437 uint32_t start = millis();
438 while ( millis() - start < LINK_WAIT_MS )
439 {
440 {
441 Lock lock;
442 ethernetif_probe_link( &enet_netif );
443 if ( netif_is_link_up( &enet_netif ) )
444 return;
445 }
446 delay( 10 );
447 }

554int EthernetClass::hostByName( const char *hostname, IPAddress &result )
555{
556 if ( result.fromString( hostname ) )
557 return 1;
558 if ( !running )
559 return 0;
560
561 ip_addr_t addr;
562 err_t err;
563 {
564 Lock lock;
565 lookup.id++;
566 lookup.done = false;
567 lookup.found = false;
568 err = dns_gethostbyname( hostname, &addr, dns_found, (void *)(uintptr_t)lookup.id );
569 }
570
571 if ( err == ERR_OK )
572 {
573 result = from_lwip( &addr );
574 return 1;
575 }
576 if ( err != ERR_INPROGRESS )
577 return 0;
578
579 uint32_t start = millis();
580 while ( !lookup.done && millis() - start < 10000 )
581 idle();
582
583 Lock lock;
584 if ( !lookup.done || !lookup.found )
585 {
586 lookup.id++; // ignore a late answer
587 return 0;
588 }
589 result = from_lwip( &lookup.addr );
590 return 1;
591}
