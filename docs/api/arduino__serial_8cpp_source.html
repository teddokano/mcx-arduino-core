9// Captured here, before arduino_io.h's #include below redefines MB_TX/MB_RX
10// as small Arduino-renumbered indices (they're part of its
11// ARDUINO_PIN_RENUMBERING table, unlike USBTX/USBRX). Serial1 needs the raw
12// r01lib io.h physical-pin values Serial::resolve_pins()'s s_pinMap[]
13// actually matches against -- passing the renumbered ones through
14// unconditionally panic()s ("unsupported TX/RX pin combination") at
15// static-init time, before setup() even runs (confirmed on real hardware).
16#ifdef CPU_MCXN947VDF

18constexpr int SERIAL1_RX_PIN = MB_RX;
19#endif

23// Global Arduino-compatible Serial instance(s) (no heap, directly inherit
24// r01lib Serial)
25// Serial: USB-bridged UART (USBTX/USBRX, not part of the pin-renumbering table)
26SerialClass Serial( USBTX, USBRX );
27
28#ifdef CPU_MCXN947VDF
29// Serial1 (D0/D1 hardware UART) is intentionally NOT supported on this board:
30// D0/D1's only UART-capable alt-function is FC2_P2/FC2_P3, the same FlexComm2
31// instance Wire already uses for I2C -- a FlexComm can only be one peripheral
32// mode at a time, so the two can't coexist in a sketch. See
33// variants/frdm_mcxn947/README.md for the full writeup.
34//
35// Serial1 IS supported on the MikroBus header instead (MB_TX/MB_RX, physical
36// P1_17/P1_16) via FC5_P1/FC5_P0 -> LPUART5 -- a FlexComm instance nothing
37// else on this board claims. These are the same physical pins as I3C_SDA/
38// I3C_SCL (used by Wire1) and are also plain GPIO-capable; pinMode()
39// explicitly reclaims ALT0 on every call and Wire1/Serial1's begin() each
40// explicitly set their own ALT, so all three switch cleanly on the same two
41// pins (just not simultaneously).
42SerialClass Serial1( SERIAL1_TX_PIN, SERIAL1_RX_PIN );
43#else
44// Serial1: hardware UART on D0(RX)/D1(TX) -- arduino_pin_by_number[] converts
45// the renumbered D0/D1 indices back to the raw physical pin macros
46// Serial::resolve_pins() expects (same trick pinMode()/digitalWrite()
47// use).
48SerialClass Serial1( arduino_pin_by_number[ D1 ], arduino_pin_by_number[ D0 ] );
49#endif
