9// Captured here, before arduino_io.h's #include below redefines MB_TX/MB_RX
10// as small Arduino-renumbered indices (they're part of its
11// ARDUINO_PIN_RENUMBERING table, unlike USBTX/USBRX). Serial1 needs the raw
12// r01lib io.h physical-pin values Serial::resolve_pins()'s s_pinMap[]

14// unconditionally panic()s ("unsupported TX/RX pin combination") at
15// static-init time, before setup() even runs (confirmed on real hardware).

23void SerialClass::begin( unsigned long baud, uint16_t config )
24{
25 int bits = 0;
26 Parity parity = Forced0; // anything format() refuses
27 int stop = 0;
28
29 switch ( config & SERIAL_DATA_MASK )
30 {
31 case SERIAL_DATA_7: bits = 7; break;
32 case SERIAL_DATA_8: bits = 8; break;
33 }
34 switch ( config & SERIAL_PARITY_MASK )
35 {
36 case SERIAL_PARITY_NONE: parity = None; break;
37 case SERIAL_PARITY_EVEN: parity = Even; break;
38 case SERIAL_PARITY_ODD: parity = Odd; break;
39 }
40 switch ( config & SERIAL_STOP_BIT_MASK )
41 {
42 case SERIAL_STOP_BIT_1: stop = 1; break;
43 case SERIAL_STOP_BIT_2: stop = 2; break;
44 }
45
46 apply_pin_mux();
47 format( bits, parity, stop ); // panic()s on anything left unset above
48 this->baud( (int)baud );
49 attach( []{}, RxIrq );
50}

52// Global Arduino-compatible Serial instance(s) (no heap, directly inherit
53// r01lib Serial)
54// Serial: USB-bridged UART (USBTX/USBRX, not part of the pin-renumbering table)
55SerialClass Serial( USBTX, USBRX );
56
57#ifdef CPU_MCXN947VDF
58// Serial1 (D0/D1 hardware UART) is intentionally NOT supported on this board:
59// D0/D1's only UART-capable alt-function is FC2_P2/FC2_P3, the same FlexComm2
60// instance Wire already uses for I2C -- a FlexComm can only be one peripheral
61// mode at a time, so the two can't coexist in a sketch. See
62// variants/frdm_mcxn947/README.md for the full writeup.
63//
64// Serial1 IS supported on the MikroBus header instead (MB_TX/MB_RX, physical
65// P1_17/P1_16) via FC5_P1/FC5_P0 -> LPUART5 -- a FlexComm instance nothing
66// else on this board claims. These are the same physical pins as I3C_SDA/
67// I3C_SCL (used by Wire1) and are also plain GPIO-capable; pinMode()
68// explicitly reclaims ALT0 on every call and Wire1/Serial1's begin() each
69// explicitly set their own ALT, so all three switch cleanly on the same two
70// pins (just not simultaneously).
71SerialClass Serial1( SERIAL1_TX_PIN, SERIAL1_RX_PIN );
72#else
73// Serial1: hardware UART on D0(RX)/D1(TX) -- arduino_pin_by_number[] converts
74// the renumbered D0/D1 indices back to the raw physical pin macros
75// Serial::resolve_pins() expects (same trick pinMode()/digitalWrite()
76// use).
77SerialClass Serial1( arduino_pin_by_number[ D1 ], arduino_pin_by_number[ D0 ] );
78#endif
