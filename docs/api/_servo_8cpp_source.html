27// The pins without PWM: one timer interrupt raises each attached pin in
28// turn, by index, and lowers it after its pulse width, then waits for the
29// rest of the 20ms. The timer counts freely and each interrupt sets the
30// next match from the last one, not from when it ran, so a late
31// interrupt shifts both ends of a pulse alike.

127void hw_init( void )
128{
129 // As TimerOne's CTIMER1: FRO_HF, which the core clock comes from too,
130 // divided to 12MHz
131 CLOCK_AttachClk( SERVO_CLOCK_ATTACH );
132 set_divider( 1u );
133 set_divider( CLOCK_GetCTimerClkFreq( SERVO_CTIMER_INDEX ) / 12000000u );
134
135 ctimer_config_t cfg;
136
137 CTIMER_GetDefaultConfig( &cfg );
138 CTIMER_Init( SERVO_CTIMER, &cfg );
139
140 clock_khz = CLOCK_GetCTimerClkFreq( SERVO_CTIMER_INDEX ) / 1000u;
141
142 ctimer_match_config_t match = {};
143 match.matchValue = 0;
144 match.enableCounterReset = false; // counts freely
145 match.enableInterrupt = true;
146 match.outControl = kCTIMER_Output_NoAction;
147 CTIMER_SetupMatch( SERVO_CTIMER, kCTIMER_Match_0, &match );
148
149 NVIC_DisableIRQ( SERVO_IRQn ); // CTIMER_SetupMatch() turned it on

170constexpr uint32_t HALF_TURN = COUNT_MASK / 2u + 1u;
171
172uint32_t us_to_ticks( uint32_t us )
173{
174 return us * clock_khz / 1000u;
175}

177// Whether a comes after b, on a counter that wraps
178bool after( uint32_t a, uint32_t b )
179{
180 uint32_t d = ( a - b ) & COUNT_MASK;
181
182 return d != 0 && d < HALF_TURN;
183}

185// The conversions done once, so that the interrupt doesn't divide (in
186// software on the Cortex-M0+)
187void init_ticks( void )
188{
189 margin_ticks = us_to_ticks( 10u );
190 refresh_ticks = us_to_ticks( REFRESH_INTERVAL );
191}

193void schedule( uint32_t t )
194{
195 t &= COUNT_MASK;
196 hw_set_match( t );
197
198 // A match already passed, or too close to set safely, would wait for
199 // the counter to come round again (6 minutes on CTIMER): run late
200 // instead. The flag, if any, is that passed match's
201 if ( !after( t, ( hw_now() + margin_ticks / 2u ) & COUNT_MASK ) )
202 {
203 t = ( hw_now() + margin_ticks ) & COUNT_MASK;
204 hw_set_match( t );
205 hw_clear_flag();
206 }
207
208 next_at = t;
209}

211void service( void )
212{
213 uint32_t t = next_at;
214
215 if ( current >= 0 && channels[ current ].clear_reg )
216 *channels[ current ].clear_reg = channels[ current ].mask;
217
218 while ( ++current < MAX_SERVOS && !channels[ current ].active )
219 ;
220
221 if ( current < MAX_SERVOS )
222 {
223 Channel &c = channels[ current ];
224
225 *c.set_reg = c.mask;
226 t += c.ticks;
227 }
228 else
229 {
230 current = -1;
231
232 bool any = false;
233
234 for ( const Channel &c : channels )
235 any = any || c.active;
236
237 if ( !any )
238 {
239 NVIC_DisableIRQ( SERVO_IRQn );
240 running = false;
241 return;
242 }
243
244 // The next frame 20ms after this one began, or at once if the
245 // pulses took longer
246 uint32_t frame_end = frame_start + refresh_ticks;
247 uint32_t soonest = t + margin_ticks;
248
249 t = after( frame_end, soonest ) ? frame_end : soonest;
250 frame_start = t;
251 }
252
253 schedule( t );
254}
255
256void start( void )
257{
258 if ( running )
259 return;
260
261 current = -1;
262 schedule( hw_now() + margin_ticks );
263 frame_start = next_at;
264
265 hw_clear_flag();
266 NVIC_ClearPendingIRQ( SERVO_IRQn );
267 NVIC_EnableIRQ( SERVO_IRQn );
268 running = true;
269}

302uint8_t Servo::attach( int pin, int min, int max )
303{
304 if ( index == INVALID_SERVO )
305 return INVALID_SERVO;
306
307 // The official library takes any pin, so sketches don't check what
308 // attach() returns: failing quietly here would leave the servo still
309 // with no hint why. Stop and say so, as analogWriteFrequency() does.
310 if ( arduino_pin_raw( pin ) < 0 )
311 panic( "Servo: no such pin" );
312
313 PwmOut *p = analog_pwm_out( pin );
314
315 if ( attached() && !( p && p == pwm ) )
316 detach(); // moving to another pin: stop the old one's pulses
317
318 this->pin = pin;
319 min_us = min;
320 max_us = max;
321
322 if ( p )
323 {
324 pwm = p;
325 pwm->period_us( REFRESH_INTERVAL );
326 writeMicroseconds( pulse_us ); // clamps a width written before to the new range
327
328 return index;
329 }
330
331 pinMode( pin, OUTPUT );
332 digitalWrite( pin, LOW );
333
334 GPIO_Type *port = digitalPinToPort( pin );
335
336 if ( port == nullptr )
337 panic( "Servo: no such pin" );
338
339 if ( !ready )
340 {
341 hw_init();
342 init_ticks();
343 ready = true;
344 }
345
346 Channel &c = channels[ index ];
347 uint32_t primask = irq_save();
348
349 c.set_reg = &port->PSOR;
350 c.clear_reg = &port->PCOR;
351 c.mask = digitalPinToBitMask( pin );
352 irq_restore( primask );
353
354 on_timer = true;
355 writeMicroseconds( pulse_us );
356
357 primask = irq_save();
358 c.active = true;
359 start();
360 irq_restore( primask );
361
362 return index;
363}

365void Servo::detach()
366{
367 if ( pwm )
368 pwm->pulsewidth_us( 0 );
369
370 if ( on_timer )
371 {
372 Channel &c = channels[ index ];
373 uint32_t primask = irq_save();
374
375 c.active = false;
376 *c.clear_reg = c.mask; // cuts a pulse that is on; the timer stops after the frame if none is left
377 irq_restore( primask );
378 }
379
380 pwm = nullptr;
381 on_timer = false;
382 pin = -1;
383}
