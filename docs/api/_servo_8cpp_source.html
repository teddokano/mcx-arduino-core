24uint8_t Servo::attach( int pin, int min, int max )
25{
26 if ( index == INVALID_SERVO )
27 return INVALID_SERVO;
28
29 PwmOut *p = analog_pwm_out( pin );
30
31 // The official library takes any pin, so sketches don't check what
32 // attach() returns: failing quietly here would leave the servo still
33 // with no hint why. Stop and say so, as analogWriteFrequency() does.
34 if ( p == nullptr )
35#if defined( FRDM_MCXC444 )
36 panic( "Servo: pin has no PWM -- use D3, D5, D6, D9 or D10" );
37#else
38 panic( "Servo: pin has no PWM -- use PWM0-PWM5 (or D3/D5/D6/D9 on FRDM-MCXA156/N236)" );
39#endif
40
41 if ( pwm && pwm != p )
42 detach(); // moving to another pin: stop the old one's pulses
43
44 this->pin = pin;
45 pwm = p;
46 min_us = min;
47 max_us = max;
48
49 pwm->period_us( REFRESH_INTERVAL );
50 writeMicroseconds( pulse_us ); // clamps a width written before to the new range
51
52 return index;
53}
