11#define PWM_PERIOD_US 1000 // 1kHz, common Arduino-like PWM rate

13static AnalogIn* analog_in_pins[ MAX_ANALOG_PINS ] = {};
14static PwmOut* pwm_out_pins[ MAX_ANALOG_PINS ] = {};

16// Defaults match classic Arduino (10bit analogRead, 8bit analogWrite).
17// Adjustable via analogReadResolution()/analogWriteResolution() (Due/Zero/
18// MKR-style extension); LPADC's raw reading is always 16bit internally.
19static int adc_resolution_bits = 10;
20static int pwm_resolution_bits = 8;

22// Sketches speak Arduino's renumbering; the pin tables and peripheral
23// classes below speak r01lib's raw pin values. Out of range becomes -1,
24// which every caller below already treats as "no such pin".
25static int raw_pin( int pin_num )
26{
27#ifdef ARDUINO_PIN_RENUMBERING
28 constexpr int count = (int)( sizeof( arduino_pin_by_number ) / sizeof( arduino_pin_by_number[0] ) );
29
30 return ( 0 <= pin_num && pin_num < count ) ? arduino_pin_by_number[ pin_num ] : -1;
31#else
32 return pin_num;
33#endif
34}

36// The PwmOut driving this pin, made on first use. nullptr means FlexPWM
37// cannot reach it: only the six dedicated PWM0-PWM5 pins exist on either
38// board. Asking PwmOut directly instead would panic() -- hence is_pwm_pin().
39static PwmOut* pwm_for( int pin )
40{
41 if ( pin < 0 || pin >= MAX_ANALOG_PINS || !PwmOut::is_pwm_pin( pin ) )
42 return nullptr;
43
44 if ( pwm_out_pins[ pin ] == nullptr )
45 {
46 pwm_out_pins[ pin ] = new PwmOut( pin );
47
48 if ( pwm_out_pins[ pin ] == nullptr )
49 panic( "error @ new, in analogWrite()" );
50
51 pwm_out_pins[ pin ]->period_us( PWM_PERIOD_US );
52 }
53
54 return pwm_out_pins[ pin ];
55}

57int analogRead( int pin_num )
58{
59 int pin = raw_pin( pin_num );
60
61 if ( pin < 0 || pin >= MAX_ANALOG_PINS )
62 return 0;
63
64 if ( analog_in_pins[ pin ] == nullptr )
65 {
66 analog_in_pins[ pin ] = new AnalogIn( pin );
67
68 if ( analog_in_pins[ pin ] == nullptr )
69 panic( "error @ new, in analogRead()" );
70 }
71
72 return (int)( analog_in_pins[ pin ]->read_u16() >> ( 16 - adc_resolution_bits ) );
73}

75void analogWrite( int pin_num, int value )
76{
77 const int max_value = ( 1 << pwm_resolution_bits ) - 1;
78
79 if ( value < 0 )
80 value = 0;
81 else if ( value > max_value )
82 value = max_value;
83
84 if ( PwmOut *pwm = pwm_for( raw_pin( pin_num ) ) )
85 {
86 pwm->write( (float)value / (float)max_value );
87 return;
88 }
89
90 // Landing here is the normal case, not an error: no D-pin on N947
91 // reaches FlexPWM at all, and on A153 only D3/D7 do, on the channels
92 // PWM5/PWM4 already own -- so analogWrite( 9, ... ), written for a
93 // classic Arduino, can never be PWM on these boards. Drive the pin
94 // high or low instead, as AVR's core does for a pin with no timer
95 // behind it (wiring_analog.c, "case NOT_ON_TIMER"). The midpoint
96 // follows analogWriteResolution() rather than AVR's hardcoded 128.
97 // pin_num is still the sketch's own numbering here -- pinMode() and
98 // digitalWrite() renumber it themselves.
99 pinMode( pin_num, OUTPUT ); // this core needs it before digitalWrite()
100 digitalWrite( pin_num, value >= ( max_value + 1 ) / 2 );
101}

103void analogWriteFrequency( int pin_num, uint32_t frequency )
104{
105 PwmOut *pwm = pwm_for( raw_pin( pin_num ) );
106
107 // Unlike analogWrite(), there is nothing to fall back to -- a plain
108 // GPIO has no period. This is also this core's own extension rather
109 // than a standard Arduino call, so no ported sketch reaches it by
110 // accident: asking for it on a pin that cannot do PWM is a mistake
111 // worth saying out loud, as every other unsupported-pin case here does.
112 if ( pwm == nullptr )
113 panic( "analogWriteFrequency: pin has no PWM -- use PWM0-PWM5" );
114
115 if ( frequency < 1 )
116 frequency = 1;
117
118 pwm->period_us( (int)( 1000000UL / frequency ) );
119}
