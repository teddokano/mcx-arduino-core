16// Defaults match classic Arduino (10bit analogRead, 8bit analogWrite).
17// Adjustable via analogReadResolution()/analogWriteResolution() (Due/Zero/
18// MKR-style extension); LPADC's raw reading is always 16bit internally.
19static int adc_resolution_bits = 10;
20static int pwm_resolution_bits = 8;

22int analogRead( int pin_num )
23{
24#ifdef ARDUINO_PIN_RENUMBERING
25 pin_num = arduino_pin_by_number[ pin_num ];
26#endif
27
28 if ( pin_num < 0 || pin_num >= MAX_ANALOG_PINS )
29 return 0;
30
31 if ( analog_in_pins[ pin_num ] == nullptr )
32 {
33 analog_in_pins[ pin_num ] = new AnalogIn( pin_num );
34
35 if ( analog_in_pins[ pin_num ] == nullptr )
36 panic( "error @ new, in analogRead()" );
37 }
38
39 return (int)( analog_in_pins[ pin_num ]->read_u16() >> ( 16 - adc_resolution_bits ) );
40}

42void analogWrite( int pin_num, int value )
43{
44#ifdef ARDUINO_PIN_RENUMBERING
45 pin_num = arduino_pin_by_number[ pin_num ];
46#endif
47
48 if ( pin_num < 0 || pin_num >= MAX_ANALOG_PINS )
49 return;
50
51 if ( pwm_out_pins[ pin_num ] == nullptr )
52 {
53 pwm_out_pins[ pin_num ] = new PwmOut( pin_num );
54
55 if ( pwm_out_pins[ pin_num ] == nullptr )
56 panic( "error @ new, in analogWrite()" );
57
58 pwm_out_pins[ pin_num ]->period_us( PWM_PERIOD_US );
59 }
60
61 int max_value = ( 1 << pwm_resolution_bits ) - 1;
62
63 if ( value < 0 )
64 value = 0;
65 else if ( value > max_value )
66 value = max_value;
67
68 pwm_out_pins[ pin_num ]->write( (float)value / (float)max_value );
69}
