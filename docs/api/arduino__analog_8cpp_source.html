11#define PWM_PERIOD_US 1000 // 1kHz, common Arduino-like PWM rate

13static AnalogIn* analog_in_pins[ MAX_ANALOG_PINS ] = {};
14static PwmOut* pwm_out_pins[ MAX_ANALOG_PINS ] = {};

16// Defaults match classic Arduino (10bit analogRead, 8bit analogWrite).
17// Adjustable via analogReadResolution()/analogWriteResolution() (Due/Zero/
18// MKR-style extension); LPADC's raw reading is always 16bit internally.
19static int adc_resolution_bits = 10;
20static int pwm_resolution_bits = 8;

22// Sketches speak Arduino's renumbering; the pin tables and peripheral
23// classes below speak r01lib's raw pin values. Out of range becomes -1,
24// which every caller below already treats as "no such pin".
25static int raw_pin( int pin_num )
26{
27#ifdef ARDUINO_PIN_RENUMBERING
28 constexpr int count = (int)( sizeof( arduino_pin_by_number ) / sizeof( arduino_pin_by_number[0] ) );
29
30 return ( 0 <= pin_num && pin_num < count ) ? arduino_pin_by_number[ pin_num ] : -1;
31#else
32 return pin_num;
33#endif
34}

36// Set by pinMode() (through analog_pin_to_gpio()) on a pin whose PwmOut it
37// switched to GPIO; the next analogWrite() puts the pin back on FlexPWM
38static bool pwm_pin_released[ MAX_ANALOG_PINS ] = {};
39
40// arduino_io.cpp: drops the pin's pinMode() GPIO object, if it has one
41bool digital_pin_drop( int raw_pin );
42
43// The PwmOut driving this pin, made on first use. nullptr means FlexPWM
44// cannot reach it: PWM0-PWM5 on every board, and D3/D5/D6/D9 on
45// FRDM-MCXA156 (on FRDM-MCXN236 those are PWM0-PWM5 pins themselves).
46// Asking PwmOut directly instead would panic() -- hence is_pwm_pin().
47static PwmOut* pwm_for( int pin )
48{
49 if ( pin < 0 || pin >= MAX_ANALOG_PINS || !PwmOut::is_pwm_pin( pin ) )
50 return nullptr;
51
52 if ( pwm_out_pins[ pin ] == nullptr )
53 {
54 // A pin whose submodule partner is already running joins it at
55 // its period, which analogWriteFrequency() may have set: putting
56 // the default back here would change the partner's frequency.
57 bool joins_running = PwmOut::shares_running_period( pin );
58
59 pwm_out_pins[ pin ] = new PwmOut( pin );
60
61 if ( pwm_out_pins[ pin ] == nullptr )
62 panic( "error @ new, in analogWrite()" );
63
64 if ( !joins_running )
65 pwm_out_pins[ pin ]->period_us( PWM_PERIOD_US );
66 }
67
68 return pwm_out_pins[ pin ];
69}

71// Called by pinMode() before it switches a pin to GPIO. A PwmOut on the
72// pin stops owning it, and analogWrite() takes it back.
73void analog_pin_to_gpio( int raw_pin )
74{
75 if ( raw_pin < 0 || raw_pin >= MAX_ANALOG_PINS || pwm_out_pins[ raw_pin ] == nullptr )
76 return;
77
78 pwm_out_pins[ raw_pin ]->release_pin();
79 pwm_pin_released[ raw_pin ] = true;
80}

82int analogRead( int pin_num )
83{
84 int pin = raw_pin( pin_num );
85
86 if ( pin < 0 || pin >= MAX_ANALOG_PINS )
87 return 0;
88
89#if defined( FRDM_MCXN236 )
90 // A3 is also the blue LED's pin, which cannot be cut off, so the core
91 // keeps it a digital pin. The chip does have an ADC input there, so
92 // rather than stopping in panic() as for a pin with none, say so with
93 // a value no reading at any resolution can take.
94 if ( pin == arduino_pin_by_number[ A3 ] )
95 return -1;
96#endif
97
98 if ( analog_in_pins[ pin ] == nullptr )
99 {
100 analog_in_pins[ pin ] = new AnalogIn( pin );
101
102 if ( analog_in_pins[ pin ] == nullptr )
103 panic( "error @ new, in analogRead()" );
104 }
105
106 return (int)( analog_in_pins[ pin ]->read_u16() >> ( 16 - adc_resolution_bits ) );
107}

109void analogWrite( int pin_num, int value )
110{
111 const int max_value = ( 1 << pwm_resolution_bits ) - 1;
112
113 if ( value < 0 )
114 value = 0;
115 else if ( value > max_value )
116 value = max_value;
117
118 int pin = raw_pin( pin_num );
119
120 if ( PwmOut *pwm = pwm_for( pin ) )
121 {
122 // The pin is FlexPWM's from here on: a pinMode() GPIO object
123 // made before would still claim it, and a pinMode() since the
124 // last analogWrite() has switched it to GPIO
125 digital_pin_drop( pin );
126
127 if ( pwm_pin_released[ pin ] )
128 {
129 pwm->claim_pin();
130 pwm_pin_released[ pin ] = false;
131 }
132
133 pwm->write( (float)value / (float)max_value );
134 return;
135 }
136
137 // Landing here is the normal case, not an error: no D-pin on N947
138 // reaches FlexPWM at all, and on A153 only D3/D7 do, on the channels
139 // PWM5/PWM4 already own -- so analogWrite( 9, ... ), written for a
140 // classic Arduino, can be PWM only on FRDM-MCXA156 and FRDM-MCXN236
141 // (D3/D5/D6/D9 there). Elsewhere, drive the pin
142 // high or low instead, as AVR's core does for a pin with no timer
143 // behind it (wiring_analog.c, "case NOT_ON_TIMER"). The midpoint
144 // follows analogWriteResolution() rather than AVR's hardcoded 128.
145 // pin_num is still the sketch's own numbering here -- pinMode() and
146 // digitalWrite() renumber it themselves.
147 pinMode( pin_num, OUTPUT ); // this core needs it before digitalWrite()
148 digitalWrite( pin_num, value >= ( max_value + 1 ) / 2 );
149}

151void analogWriteFrequency( int pin_num, uint32_t frequency )
152{
153 PwmOut *pwm = pwm_for( raw_pin( pin_num ) );
154
155 // Unlike analogWrite(), there is nothing to fall back to -- a plain
156 // GPIO has no period. This is also this core's own extension rather
157 // than a standard Arduino call, so no ported sketch reaches it by
158 // accident: asking for it on a pin that cannot do PWM is a mistake
159 // worth saying out loud, as every other unsupported-pin case here does.
160 if ( pwm == nullptr )
161 panic( "analogWriteFrequency: pin has no PWM -- use PWM0-PWM5 (or D3/D5/D6/D9 on FRDM-MCXA156/N236)" );
162
163 if ( frequency < 1 )
164 frequency = 1;
165
166 pwm->period_us( (int)( 1000000UL / frequency ) );
167}
