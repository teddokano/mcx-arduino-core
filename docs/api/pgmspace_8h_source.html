1/*
2 * @author Tedd OKANO
3 *
4 * Released under the MIT license
5 *
6 * avr-libc's <avr/pgmspace.h> for this core, so AVR-era sketches and
7 * libraries that put constants in flash build unchanged. Arduino.h
8 * includes it, as AVR's does.
9 *
10 * AVR is a Harvard-architecture chip: flash and RAM are separate address
11 * spaces, so data tagged PROGMEM needs special instructions
12 * (pgm_read_byte() and the _P string functions) to read back. On this
13 * Cortex-M, flash is ordinary memory, readable like anything else, so
14 * PROGMEM is empty, the reads are plain pointer reads and each _P function
15 * is the ordinary one. The "near" and "far" forms are the same thing too:
16 * there is only one 32-bit address space.
17 *
18 * Original implementation written to avr-libc's documented interface
19 * (avr-libc is under the Modified BSD License; none of its source is
20 * copied). Everything is a macro, as in avr-libc, so the common
21 * "#ifndef pgm_read_byte_near" checks in libraries see them.
22 * Usable from C as well as C++.
23 */

63/* The multi-byte reads go through memcpy(), so they work at any address,
64 * as on AVR, where a word is read a byte at a time: AVR code reads words
65 * out of byte tables at odd offsets. A Cortex-M33 reads an unaligned word
66 * anyway, and the compiler turns these into the same single load there,
67 * but a Cortex-M0+ would fault. */
68static inline uint16_t mcx_pgm_read_word( const void *p ) { uint16_t v; memcpy( &v, p, sizeof( v ) ); return v; }
69static inline uint32_t mcx_pgm_read_dword( const void *p ) { uint32_t v; memcpy( &v, p, sizeof( v ) ); return v; }
70static inline uint64_t mcx_pgm_read_qword( const void *p ) { uint64_t v; memcpy( &v, p, sizeof( v ) ); return v; }
71static inline float mcx_pgm_read_float( const void *p ) { float v; memcpy( &v, p, sizeof( v ) ); return v; }
72static inline void *mcx_pgm_read_ptr( const void *p ) { void *v; memcpy( &v, p, sizeof( v ) ); return v; }
