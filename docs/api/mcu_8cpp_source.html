46void init_mcu( void )
47{
48#ifdef CPU_MCXN947VDF
49 CLOCK_SetClkDiv(kCLOCK_DivFlexcom4Clk, 1);
50 CLOCK_AttachClk(BOARD_DEBUG_UART_CLK_ATTACH);
51
52 /* Attach PLL0 clock to I3C, 150MHz / 6 = 25MHz. */
53 CLOCK_SetClkDiv(kCLOCK_DivI3c1FClk, 6U);
54 CLOCK_AttachClk(kPLL0_to_I3C1FCLK);
55
56 /* I2C */
57 CLOCK_SetClkDiv(kCLOCK_DivFlexcom2Clk, 1u);
58 CLOCK_AttachClk(kFRO12M_to_FLEXCOMM2);
59
60 /* SPI -- FRO_HF_DIV (48MHz, see BOARD_BootClockPLL150M's
61 CLOCK_SetupFROHFClocking() call), not FRO12M: this chip's LPSPI
62 can't exceed roughly half its source clock's rate (basic SPI
63 clock-division limit), so a 12MHz FRO12M source caps real
64 throughput far below what SPISettings() requests -- confirmed on
65 real hardware, a 24MHz SPISettings request measured as ~31kHz
66 actual SCK on FRO12M (FRDM-MCXA153's equivalent default SPI
67 already uses FRO_HF_DIV for exactly this reason, at the same
68 48MHz, and measures the requested 24MHz correctly). */
69 CLOCK_SetClkDiv(kCLOCK_DivFlexcom1Clk, 1u);
70 CLOCK_AttachClk(kFRO_HF_DIV_to_FLEXCOMM1);
71
72 /* I2C on MikroBus (MB_SDA/MB_SCL -> LPI2C3) */
73 CLOCK_SetClkDiv(kCLOCK_DivFlexcom3Clk, 1u);
74 CLOCK_AttachClk(kFRO12M_to_FLEXCOMM3);
75
76 /* SPI on MikroBus (MB_MOSI/MB_MISO/MB_SCK/MB_CS -> LPSPI6) -- FRO_HF_DIV,
77 same reasoning as the default SPI above (FRO12M is too coarse a
78 source for LPSPI's baud-rate divider search to land on the exact
79 requested rate). Confirmed on real hardware: a 1MHz SPISettings
80 request on FRO12M measured as 500kHz actual SCK; FRDM-MCXA153's
81 equivalent SPI1 (LPSPI0) already uses FRO_HF_DIV via its own
82 clock_config.c and hits the requested rate correctly. */
83 CLOCK_SetClkDiv(kCLOCK_DivFlexcom6Clk, 1u);
84 CLOCK_AttachClk(kFRO_HF_DIV_to_FLEXCOMM6);
85
86 /* Serial1 on MikroBus (MB_TX/MB_RX -> LPUART5); the attach itself is
87 also done lazily by Serial::_setup_clock() when the instance begin()s,
88 but the divider needs a known value up front like every other FlexComm here. */
89 CLOCK_SetClkDiv(kCLOCK_DivFlexcom5Clk, 1u);
90
91 SYSCON->CLOCK_CTRL |= SYSCON_CLOCK_CTRL_FRO1MHZ_ENA_MASK; // UTICK
92
93 CLOCK_EnableClock( kCLOCK_Gpio0 );
94 CLOCK_EnableClock( kCLOCK_Gpio1 );
95 CLOCK_EnableClock( kCLOCK_Gpio2 );
96 CLOCK_EnableClock( kCLOCK_Gpio3 );
97 CLOCK_EnableClock( kCLOCK_Gpio4 );
98
99 /* Init board hardware. */
100 BOARD_InitBootPins();
101 BOARD_InitBootClocks();
102 BOARD_InitBootPeripherals();
103 #ifndef BOARD_INIT_DEBUG_CONSOLE_PERIPHERAL
104 /* Init FSL debug console. */
105 BOARD_InitDebugConsole();
106 #endif
107
108#elif CPU_MCXN236VDF
109 CLOCK_SetClkDiv(kCLOCK_DivFlexcom4Clk, 1);
110 CLOCK_AttachClk(BOARD_DEBUG_UART_CLK_ATTACH);
111
112 /* Attach PLL0 clock to I3C, 150MHz / 12 = 12.5MHz. */
113 CLOCK_SetClkDiv(kCLOCK_DivI3c1FClk, 12U);
114 CLOCK_AttachClk(kPLL0_to_I3C1FCLK);
