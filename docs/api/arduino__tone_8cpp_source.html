6 * tone()/noTone() toggle an arbitrary GPIO pin from a CTIMER0 match
7 * interrupt (square wave, 2 toggles per cycle) so any digital pin can be
8 * used, matching classic Arduino tone() — unlike analogWrite() which is
9 * limited to the FlexPWM0-backed PWM0-PWM5 pins. Only 1 tone can be
10 * active at a time (same limitation as most Arduino cores).
11 */

32void ctimer_init( void )
33{
34 /*
35 * CTIMER0's clock mux is unattached out of reset (CLOCK_GetCTimerClkFreq()
36 * returns 0 until a source is attached), so CTIMER_Init() alone (clock
37 * gate + reset only) leaves the peripheral clockless — match never fires.
38 * Attach FRO12M, same fixed/always-available source AnalogIn.cpp uses.
39 *
40 * N947 note: the divider symbol is named differently here
41 * (`kCLOCK_DivCtimer0Clk`, mixed case + "Clk" suffix) than on A153
42 * (`kCLOCK_DivCTIMER0`, all-caps) -- confirmed via NXP's own
43 * FRDM-MCXN947 CTIMER driver example, which also uses CLOCK_SetClkDiv
44 * (not CLOCK_SetClockDiv -- also a different function name on this
45 * chip's SDK).
46 */
47 CLOCK_AttachClk( kFRO12M_to_CTIMER0 );
48#if defined( CPU_MCXN947VDF )
49 CLOCK_SetClkDiv( kCLOCK_DivCtimer0Clk, 1u );
50#else
51 CLOCK_SetClockDiv( kCLOCK_DivCTIMER0, 1u );
52#endif
53
54 ctimer_config_t cfg;
55
56 CTIMER_GetDefaultConfig( &cfg );
57 CTIMER_Init( CTIMER0, &cfg );
58
59 ctimer_ready = true;
60}

64extern "C" void CTIMER0_IRQHandler( void )
65{
66 uint32_t flags = CTIMER_GetStatusFlags( CTIMER0 );
67
68 CTIMER_ClearStatusFlags( CTIMER0, flags );
69
70 if ( !( flags & kCTIMER_Match0Flag ) || ( tone_pin == nullptr ) )
71 return;
72
73 if ( toggles_left != TONE_INDEFINITE )
74 {
75 if ( toggles_left == 0 )
76 {
77 CTIMER_StopTimer( CTIMER0 );
78 *tone_pin = false;
79 return;
80 }
81
82 toggles_left--;
83 }
84
85 tone_state = !tone_state;
86 *tone_pin = tone_state;
87}

89void tone( int pin_num, unsigned int frequency, unsigned long duration )
90{
91#ifdef ARDUINO_PIN_RENUMBERING
92 pin_num = arduino_pin_by_number[ pin_num ];
93#endif
94
95 if ( ( pin_num < 0 ) || ( frequency == 0 ) )
96 return;
97
98 if ( !ctimer_ready )
99 ctimer_init();
100
101 CTIMER_StopTimer( CTIMER0 );
102
103 if ( tone_pin_num != pin_num )
104 {
105 delete tone_pin;
106 tone_pin = new DigitalInOut( (uint8_t)pin_num, DigitalInOut::OUTPUT );
107 tone_pin_num = pin_num;
108 }
109
110 tone_state = false;
111 *tone_pin = tone_state;
112
113 uint32_t timer_clk = CLOCK_GetCTimerClkFreq( 0 );
114 uint32_t match = ( timer_clk / ( 2u * frequency ) ) - 1u;
115
116 toggles_left = ( duration == 0 )
117 ? TONE_INDEFINITE
118 : (uint32_t)( ( (uint64_t)2u * frequency * duration ) / 1000u );
119
120 ctimer_match_config_t match_cfg = {};
121 match_cfg.matchValue = match;
122 match_cfg.enableCounterReset = true;
123 match_cfg.enableCounterStop = false;
124 match_cfg.outControl = kCTIMER_Output_NoAction;
125 match_cfg.outPinInitState = false;
126 match_cfg.enableInterrupt = true;
127
128 CTIMER_SetupMatch( CTIMER0, kCTIMER_Match_0, &match_cfg );
129 EnableIRQ( CTIMER0_IRQn );
130
131 /*
132 * StopTimer() only clears CEN — it leaves TC wherever it was. If the new
133 * match value is smaller than the stale TC, TC won't hit it until it
134 * wraps all the way around (~6 min @ 12MHz), so the tone never starts.
135 * Reset TC/PR to 0 before (re)starting.
136 */
137 CTIMER_Reset( CTIMER0 );
138 CTIMER_StartTimer( CTIMER0 );
139}
