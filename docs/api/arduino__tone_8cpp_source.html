1/*
2 * @author Tedd OKANO
3 *
4 * Released under the MIT license
5 *
6 * tone()/noTone() toggle an arbitrary GPIO pin from a periodic timer
7 * interrupt (square wave, 2 toggles per cycle) so any digital pin can be
8 * used, matching classic Arduino tone() — unlike analogWrite() which is
9 * limited to the PWM pins. Only 1 tone can be active at a time (same
10 * limitation as most Arduino cores).
11 *
12 * The timer is CTIMER0's match 0 on the MCX A/N chips, and PIT channel 0
13 * on FRDM-MCXC444, which has no CTIMER. Only timer_init(), timer_start()
14 * and timer_stop() differ. PIT channel 1 is the bundled TimerOne's; the
15 * PIT's one interrupt handler, here, serves both.
16 */
17

56// One interrupt every half period. The PIT counts the bus clock.
57void timer_start( unsigned int frequency )
58{
59 uint32_t ticks = CLOCK_GetFreq( kCLOCK_BusClk ) / ( 2u * frequency );
60
61 PIT_SetTimerPeriod( PIT, kPIT_Chnl_0, ticks ? ticks : 1u );
62 PIT_ClearStatusFlags( PIT, kPIT_Chnl_0, kPIT_TimerFlag );
63 PIT_EnableInterrupts( PIT, kPIT_Chnl_0, kPIT_TimerInterruptEnable );
64 EnableIRQ( PIT_IRQn );
65 PIT_StartTimer( PIT, kPIT_Chnl_0 ); // loads the period afresh
66}

70void timer_init( void )
71{
72 /*
73 * CTIMER0's clock mux is unattached out of reset (CLOCK_GetCTimerClkFreq()
74 * returns 0 until a source is attached), so CTIMER_Init() alone (clock
75 * gate + reset only) leaves the peripheral clockless — match never fires.
76 * Attach FRO12M, same fixed/always-available source AnalogIn.cpp uses.
77 *
78 * N947 note: the divider symbol is named differently here
79 * (`kCLOCK_DivCtimer0Clk`, mixed case + "Clk" suffix) than on A153
80 * (`kCLOCK_DivCTIMER0`, all-caps) -- confirmed via NXP's own
81 * FRDM-MCXN947 CTIMER driver example, which also uses CLOCK_SetClkDiv
82 * (not CLOCK_SetClockDiv -- also a different function name on this
83 * chip's SDK).
84 */
85 CLOCK_AttachClk( kFRO12M_to_CTIMER0 );
86#if defined( CPU_MCXN947VDF ) || defined( CPU_MCXN236VDF )
87 CLOCK_SetClkDiv( kCLOCK_DivCtimer0Clk, 1u );
88#elif defined( CPU_MCXA153VLH ) || defined( CPU_MCXA156VLL )
89 CLOCK_SetClockDiv( kCLOCK_DivCTIMER0, 1u );
90#else
91#error "arduino_tone.cpp: set CTIMER0's clock divider for this chip"
92#endif
93
94 ctimer_config_t cfg;
95
96 CTIMER_GetDefaultConfig( &cfg );
97 CTIMER_Init( CTIMER0, &cfg );
98
99 timer_ready = true;
100}

107void timer_start( unsigned int frequency )
108{
109 uint32_t timer_clk = CLOCK_GetCTimerClkFreq( 0 );
110 uint32_t match = ( timer_clk / ( 2u * frequency ) ) - 1u;
111
112 ctimer_match_config_t match_cfg = {};
113 match_cfg.matchValue = match;
114 match_cfg.enableCounterReset = true;
115 match_cfg.enableCounterStop = false;
116 match_cfg.outControl = kCTIMER_Output_NoAction;
117 match_cfg.outPinInitState = false;
118 match_cfg.enableInterrupt = true;
119
120 CTIMER_SetupMatch( CTIMER0, kCTIMER_Match_0, &match_cfg );
121 EnableIRQ( CTIMER0_IRQn );
122
123 /*
124 * StopTimer() only clears CEN — it leaves TC wherever it was. If the new
125 * match value is smaller than the stale TC, TC won't hit it until it
126 * wraps all the way around (~6 min @ 12MHz), so the tone never starts.
127 * Reset TC/PR to 0 before (re)starting.
128 */
129 CTIMER_Reset( CTIMER0 );
130 CTIMER_StartTimer( CTIMER0 );
131}

135// Called from the timer interrupt, every half period
136void tone_tick( void )
137{
138 if ( tone_pin == nullptr )
139 return;
140
141 if ( toggles_left != TONE_INDEFINITE )
142 {
143 if ( toggles_left == 0 )
144 {
145 timer_stop();
146 *tone_pin = false;
147 return;
148 }
149
150 toggles_left--;
151 }
152
153 tone_state = !tone_state;
154 *tone_pin = tone_state;
155}

161void (* volatile pit_channel1_hook)( void ) = nullptr;

163// PIT_Init() without its stopping every channel: tone() and TimerOne each
164// call this the first time they use their channel, and mustn't stop the
165// other's. It does the rest as PIT_Init() with the default configuration
166// does: the clock on, the module on (after a read of MCR, for erratum
167// 7914), and the timers stopped while a debugger halts the core
168void pit_module_init( void )
169{
170 CLOCK_EnableClock( kCLOCK_Pit0 );
171 (void)PIT->MCR;
172 PIT->MCR = ( PIT->MCR & ~PIT_MCR_MDIS_MASK ) | PIT_MCR_FRZ_MASK;
173}

175// Channel 1 is the bundled TimerOne's (arduino_timer_hooks.h). Each
176// channel's flag is only looked at while its interrupt is on: TimerOne's
177// flag also sets while it runs with no interrupt attached, and has to stay
178// set then, as Timer1's overflow flag does on AVR
179extern "C" void PIT_IRQHandler( void )
180{
181 if ( PIT->CHANNEL[ kPIT_Chnl_0 ].TCTRL & PIT_TCTRL_TIE_MASK )
182 {
183 uint32_t flags = PIT_GetStatusFlags( PIT, kPIT_Chnl_0 );
184
185 PIT_ClearStatusFlags( PIT, kPIT_Chnl_0, flags );
186
187 if ( flags & kPIT_TimerFlag )
188 tone_tick();
189 }
190
191 if ( PIT->CHANNEL[ kPIT_Chnl_1 ].TCTRL & PIT_TCTRL_TIE_MASK )
192 {
193 uint32_t flags = PIT_GetStatusFlags( PIT, kPIT_Chnl_1 );
194
195 PIT_ClearStatusFlags( PIT, kPIT_Chnl_1, flags );
196
197 void ( *hook )( void ) = pit_channel1_hook;
198
199 if ( ( flags & kPIT_TimerFlag ) && hook )
200 hook();
201 }
202}

218void tone( int pin_num, unsigned int frequency, unsigned long duration )
219{
220#ifdef ARDUINO_PIN_RENUMBERING
221 pin_num = arduino_pin_by_number[ pin_num ];
222#endif
223
224 if ( ( pin_num < 0 ) || ( frequency == 0 ) )
225 return;
226
227 if ( !timer_ready )
228 timer_init();
229
230 timer_stop();
231
232 if ( tone_pin_num != pin_num )
233 {
234 delete tone_pin;
235 tone_pin = new DigitalInOut( (uint8_t)pin_num, DigitalInOut::OUTPUT );
236 tone_pin_num = pin_num;
237 }
238
239 tone_state = false;
240 *tone_pin = tone_state;
241
242 toggles_left = ( duration == 0 )
243 ? TONE_INDEFINITE
244 : (uint32_t)( ( (uint64_t)2u * frequency * duration ) / 1000u );
245
246 timer_start( frequency );
247}

249void noTone( int pin_num )
250{
251#ifdef ARDUINO_PIN_RENUMBERING
252 pin_num = arduino_pin_by_number[ pin_num ];
253#endif
254
255 if ( pin_num != tone_pin_num )
256 return;
257
258 if ( timer_ready )
259 timer_stop();
260
261 if ( tone_pin != nullptr )
262 *tone_pin = false;
263}
