32static void set_pin_mode( int pin_num, int mode )
33{
34 analog_pin_to_gpio( pin_num );
35
36 pin_modes[ pin_num ] = (uint8_t)mode;
37
38 int dir = (mode == OUTPUT || mode == OUTPUT_OPENDRAIN) ? DigitalInOut::OUTPUT : DigitalInOut::INPUT;
39 int pin_mode = DigitalInOut::PullNone;
40
41 if ( mode == INPUT_PULLUP )
42 pin_mode = DigitalInOut::PullUp;
43 else if ( mode == INPUT_PULLDOWN )
44 pin_mode = DigitalInOut::PullDown;
45 else if ( mode == OUTPUT_OPENDRAIN )
46 pin_mode = DigitalInOut::OpenDrain;
47
48 if ( digital_pins[ pin_num ] != nullptr )
49 {
50 if ( dir == OUTPUT )
51 digital_pins[ pin_num ]->output();
52 else
53 digital_pins[ pin_num ]->input();
54
55 digital_pins[ pin_num ]->mode( pin_mode );
56 digital_pins[ pin_num ]->pin_mux( 0 ); // reclaim as plain GPIO (ALT0), regardless of any peripheral mode a prior begin() (I2C/I3C/SPI/PWM) left the pin in
57 }
58 else
59 {
60 digital_pins[ pin_num ] = new DigitalInOut( pin_num, dir, 0, pin_mode );
61
62 if ( digital_pins[ pin_num ] == nullptr )
63 panic( "error @ new, in pinMode()" );
64
65 digital_pins[ pin_num ]->pin_mux( 0 ); // some pins (e.g. MB_RX/MB_TX = I3C_SDA/I3C_SCL) boot up muxed to a non-GPIO peripheral function instead of the usual ALT0 default
66 }
67}

109void attachInterrupt( int pin_num, void (*callback)(void), int mode )
110{
111#ifdef ARDUINO_PIN_RENUMBERING
112 pin_num = arduino_pin_by_number[ pin_num ];
113#endif
114
115 if ( pin_num < 0 || pin_num >= MAX_DIGITAL_PINS )
116 return;
117
118 InterruptIn* int_pin = interrupt_pins[ pin_num ];
119
120 if ( int_pin == nullptr )
121 {
122 int_pin = new InterruptIn( pin_num );
123
124 if ( int_pin == nullptr )
125 panic( "error @ new, in attachInterrupt()" );
126
127 interrupt_pins[ pin_num ] = int_pin;
128
129 // The InterruptIn's own DigitalInOut constructor has just set
130 // the pin up afresh: input, no pull. Arduino's attachInterrupt()
131 // leaves the pin as pinMode() set it -- pinMode(INPUT_PULLUP)
132 // then attachInterrupt() is the usual way to wire a button or
133 // an open-drain INT line -- so put that back.
134 if ( digital_pins[ pin_num ] != nullptr )
135 set_pin_mode( pin_num, pin_modes[ pin_num ] );
136 }
137
138 switch ( mode )
139 {
140 case RISING:
141 int_pin->rise( callback );
142 break;
143
144 case FALLING:
145 int_pin->fall( callback );
146 break;
147
148 case CHANGE:
149 int_pin->change( callback );
150 break;
151
152 case LOW:
153 int_pin->low( callback );
154 break;
155
156 default:
157 panic( "error @ attachInterrupt(), unknown mode" );
158 break;
159 }
160}

219void shiftOut( int dataPin, int clockPin, int bitOrder, uint8_t val )
220{
221 for ( int i = 0; i < 8; i++ )
222 {
223 if ( bitOrder == LSBFIRST )
224 digitalWrite( dataPin, !!(val & (1 << i)) );
225 else
226 digitalWrite( dataPin, !!(val & (1 << (7 - i))) );
227
228 digitalWrite( clockPin, HIGH );
229 digitalWrite( clockPin, LOW );
230 }
231}

233uint8_t shiftIn( int dataPin, int clockPin, int bitOrder )
234{
235 uint8_t value = 0;
236
237 for ( int i = 0; i < 8; i++ )
238 {
239 digitalWrite( clockPin, HIGH );
240
241 if ( bitOrder == LSBFIRST )
242 value |= digitalRead( dataPin ) << i;
243 else
244 value |= digitalRead( dataPin ) << (7 - i);
245
246 digitalWrite( clockPin, LOW );
247 }
248
249 return value;
250}

252unsigned long pulseIn( int pin_num, bool state, unsigned long timeout )
253{
254 unsigned long start_wait = micros();
255
256 while ( digitalRead( pin_num ) == state )
257 if ( (micros() - start_wait) > timeout )
258 return 0;
259
260 while ( digitalRead( pin_num ) != state )
261 if ( (micros() - start_wait) > timeout )
262 return 0;
263
264 unsigned long start_pulse = micros();
265
266 while ( digitalRead( pin_num ) == state )
267 if ( (micros() - start_pulse) > timeout )
268 return 0;
269
270 return micros() - start_pulse;
271}
