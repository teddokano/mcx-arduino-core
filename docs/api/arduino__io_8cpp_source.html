15void pinMode( int pin_num, int mode )
16{
17#ifdef ARDUINO_PIN_RENUMBERING
18 pin_num = arduino_pin_by_number[ pin_num ];
19#endif
20
21 if ( pin_num < 0 || pin_num >= MAX_DIGITAL_PINS )
22 return;
23
24 int dir = (mode == OUTPUT || mode == OUTPUT_OPENDRAIN) ? DigitalInOut::OUTPUT : DigitalInOut::INPUT;
25 int pin_mode = DigitalInOut::PullNone;
26
27 if ( mode == INPUT_PULLUP )
28 pin_mode = DigitalInOut::PullUp;
29 else if ( mode == INPUT_PULLDOWN )
30 pin_mode = DigitalInOut::PullDown;
31 else if ( mode == OUTPUT_OPENDRAIN )
32 pin_mode = DigitalInOut::OpenDrain;
33
34 if ( digital_pins[ pin_num ] != nullptr )
35 {
36 if ( dir == OUTPUT )
37 digital_pins[ pin_num ]->output();
38 else
39 digital_pins[ pin_num ]->input();
40
41 digital_pins[ pin_num ]->mode( pin_mode );
42 digital_pins[ pin_num ]->pin_mux( 0 ); // reclaim as plain GPIO (ALT0), regardless of any peripheral mode a prior begin() (I2C/I3C/SPI/PWM) left the pin in
43 }
44 else
45 {
46 digital_pins[ pin_num ] = new DigitalInOut( pin_num, dir, 0, pin_mode );
47
48 if ( digital_pins[ pin_num ] == nullptr )
49 panic( "error @ new, in pinMode()" );
50
51 digital_pins[ pin_num ]->pin_mux( 0 ); // some pins (e.g. MB_RX/MB_TX = I3C_SDA/I3C_SCL) boot up muxed to a non-GPIO peripheral function instead of the usual ALT0 default
52 }
53}

83void attachInterrupt( int pin_num, void (*callback)(void), int mode )
84{
85#ifdef ARDUINO_PIN_RENUMBERING
86 pin_num = arduino_pin_by_number[ pin_num ];
87#endif
88
89 if ( pin_num < 0 || pin_num >= MAX_DIGITAL_PINS )
90 return;
91
92 InterruptIn* int_pin = interrupt_pins[ pin_num ];
93
94 if ( int_pin == nullptr )
95 {
96 int_pin = new InterruptIn( pin_num );
97
98 if ( int_pin == nullptr )
99 panic( "error @ new, in attachInterrupt()" );
100
101 interrupt_pins[ pin_num ] = int_pin;
102 }
103
104 switch ( mode )
105 {
106 case RISING:
107 int_pin->rise( callback );
108 break;
109
110 case FALLING:
111 int_pin->fall( callback );
112 break;
113
114 case CHANGE:
115 int_pin->rise( callback );
116 int_pin->fall( callback );
117 break;
118
119 case LOW:
120 int_pin->low( callback );
121 break;
122
123 default:
124 panic( "error @ attachInterrupt(), unknown mode" );
125 break;
126 }
127}

186void shiftOut( int dataPin, int clockPin, int bitOrder, uint8_t val )
187{
188 for ( int i = 0; i < 8; i++ )
189 {
190 if ( bitOrder == LSBFIRST )
191 digitalWrite( dataPin, !!(val & (1 << i)) );
192 else
193 digitalWrite( dataPin, !!(val & (1 << (7 - i))) );
194
195 digitalWrite( clockPin, HIGH );
196 digitalWrite( clockPin, LOW );
197 }
198}

200uint8_t shiftIn( int dataPin, int clockPin, int bitOrder )
201{
202 uint8_t value = 0;
203
204 for ( int i = 0; i < 8; i++ )
205 {
206 digitalWrite( clockPin, HIGH );
207
208 if ( bitOrder == LSBFIRST )
209 value |= digitalRead( dataPin ) << i;
210 else
211 value |= digitalRead( dataPin ) << (7 - i);
212
213 digitalWrite( clockPin, LOW );
214 }
215
216 return value;
217}

219unsigned long pulseIn( int pin_num, bool state, unsigned long timeout )
220{
221 unsigned long start_wait = micros();
222
223 while ( digitalRead( pin_num ) == state )
224 if ( (micros() - start_wait) > timeout )
225 return 0;
226
227 while ( digitalRead( pin_num ) != state )
228 if ( (micros() - start_wait) > timeout )
229 return 0;
230
231 unsigned long start_pulse = micros();
232
233 while ( digitalRead( pin_num ) == state )
234 if ( (micros() - start_pulse) > timeout )
235 return 0;
236
237 return micros() - start_pulse;
238}
