4 * Received packets wait in a short queue per socket, filled by lwIP's callback in the

17static constexpr int RX_QUEUE = 16; // packets that wait per socket (over a millisecond of full-size ones at 100 Mbit/s); more are dropped
18static constexpr size_t TX_MAX = 1472; // a UDP payload in one Ethernet frame
19
20struct UdpSlot
21{
22 bool used;
23 struct udp_pcb *pcb;
24 struct pbuf *queue[ RX_QUEUE ];
25 ip_addr_t queue_ip[ RX_QUEUE ];
26 uint16_t queue_port[ RX_QUEUE ];
27 volatile uint8_t head;
28 volatile uint8_t count;
29 struct pbuf *cur; // the packet parsePacket() returned, what is left of it
30 ip_addr_t cur_ip;
31 uint16_t cur_port;
32 bool joined; // beginMulticast() joined `group`
33 ip4_addr_t group;
34};

36static UdpSlot udp_slots[ MAX_SOCK_NUM ];
37
38static void on_recv( void *arg, struct udp_pcb *pcb, struct pbuf *p, const ip_addr_t *addr, u16_t port )
39{
40 (void)pcb;
41 UdpSlot *s = (UdpSlot *)arg;
42
43 // copy out of the driver's receive buffer, so packets waiting here don't hold it
44 struct pbuf *q = s->count < RX_QUEUE ? pbuf_clone( PBUF_RAW, PBUF_RAM, p ) : NULL;
45 pbuf_free( p );
46 if ( !q )
47 return;
48
49 int i = ( s->head + s->count ) % RX_QUEUE;
50 s->queue[ i ] = q;
51 s->queue_ip[ i ] = *addr;
52 s->queue_port[ i ] = port;
53 s->count = s->count + 1;
54}

65uint8_t EthernetUDP::begin( uint16_t port )
66{
67 if ( !started() )
68 return 0;
69 if ( _sock >= 0 )
70 stop();
71
72 Lock lock;
73 int sock = -1;
74 for ( int i = 0; i < MAX_SOCK_NUM; i++ )
75 {
76 if ( !udp_slots[ i ].used )
77 {
78 sock = i;
79 break;
80 }
81 }
82 if ( sock < 0 )
83 return 0;
84
85 struct udp_pcb *pcb = udp_new();
86 if ( !pcb )
87 return 0;
88
89 ip_set_option( pcb, SOF_REUSEADDR );
90 if ( udp_bind( pcb, IP_ANY_TYPE, port ) != ERR_OK )
91 {
92 udp_remove( pcb );
93 return 0;
94 }
95
96 UdpSlot *s = &udp_slots[ sock ];
97 memset( s, 0, sizeof( *s ) );
98 s->used = true;
99 s->pcb = pcb;
100 udp_recv( pcb, on_recv, s );
101 _sock = sock;
102 return 1;
103}

122void EthernetUDP::stop()
123{
124 if ( _sock < 0 )
125 return;
126
127 Lock lock;
128 UdpSlot *s = &udp_slots[ _sock ];
129
130 if ( s->joined )
131 igmp_leavegroup( IP4_ADDR_ANY4, &s->group );
132 udp_remove( s->pcb );
133 for ( int i = 0; i < s->count; i++ )
134 pbuf_free( s->queue[ ( s->head + i ) % RX_QUEUE ] );
135 if ( s->cur )
136 pbuf_free( s->cur );
137 if ( _tx )
138 pbuf_free( (struct pbuf *)_tx );
139 memset( s, 0, sizeof( *s ) );
140
141 _tx = NULL;
142 _sock = -1;
143}

154int EthernetUDP::beginPacket( IPAddress ip, uint16_t port )
155{
156 // as the official library: no packet to 0.0.0.0 or port 0 (an IPv6 address reads as 0.0.0.0 here)
157 if ( _sock < 0 || (uint32_t)ip == 0 || port == 0 )
158 return 0;
159

163 _tx = pbuf_alloc( PBUF_TRANSPORT, TX_MAX, PBUF_RAM );
164 _tx_len = 0;
165 _tx_ip = ip;
166 _tx_port = port;

186int EthernetUDP::endPacket()
187{
188 if ( _sock < 0 || !_tx )
189 return 0;
190
191 Lock lock;
192 struct pbuf *p = (struct pbuf *)_tx;
193 ip_addr_t a = to_lwip( _tx_ip );
194
195 pbuf_realloc( p, (u16_t)_tx_len );
196 err_t err = udp_sendto( udp_slots[ _sock ].pcb, p, &a, _tx_port );
197 pbuf_free( p );
198 _tx = NULL;
199 return err == ERR_OK;
200}

202int EthernetUDP::parsePacket()
203{
204 if ( _sock < 0 )
205 return 0;
206
207 Lock lock;
208 UdpSlot *s = &udp_slots[ _sock ];
209
210 if ( s->cur )
211 {
212 pbuf_free( s->cur );
213 s->cur = NULL;
214 }
215 if ( !s->count )
216 return 0;
217
218 s->cur = s->queue[ s->head ];
219 s->cur_ip = s->queue_ip[ s->head ];
220 s->cur_port = s->queue_port[ s->head ];
221 s->head = ( s->head + 1 ) % RX_QUEUE;
222 s->count = s->count - 1;
223 return s->cur->tot_len;
224}
