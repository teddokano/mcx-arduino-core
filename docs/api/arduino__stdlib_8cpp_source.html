14// long and int are the same width here, so newlib's int versions already
15// give avr-libc's long ones exactly.
16static_assert( sizeof( long ) == sizeof( int ), "ltoa()/ultoa() assume a 32-bit long" );

18extern "C" char *ltoa( long value, char *str, int base )

28// platform.txt links with -u _printf_float, so snprintf()'s %f is the full
29// one: rounded, and with width/alignment, as avr-libc's dtostrf() is. No
30// length limit is passed on, same as avr-libc -- the caller sizes str.
31extern "C" char *dtostrf( double val, signed char width, unsigned char prec, char *str )
32{
33 sprintf( str, "%*.*f", (int)width, (int)prec, val );
