38/*
39 * snprintf()/printf() have no format specifier for arbitrary radixes (no
40 * %b for binary, and nothing at all for bases other than 8/10/16), so
41 * BIN -- and any other non-DEC/HEX/OCT base a sketch might pass, matching
42 * real Arduino's support for any radix 2-36 -- has to be hand-converted.
43 */
44void _utoa_radix( unsigned long long value, int base, char *buf, size_t bufsize )
45{
46 static const char digits[] = "0123456789abcdefghijklmnopqrstuvwxyz";
47
48 if ( base < 2 || base > 36 )
49 base = 10;
50
51 char tmp[ 66 ];
52 int i = 0;
53
54 if ( value == 0 )
55 tmp[ i++ ] = '0';
56
57 while ( value > 0 && i < (int)sizeof(tmp) - 1 )
58 {
59 tmp[ i++ ] = digits[ value % (unsigned)base ];
60 value /= (unsigned)base;
61 }
62
63 size_t j = 0;
64 while ( i > 0 && j < bufsize - 1 )
65 buf[ j++ ] = tmp[ --i ];
66 buf[ j ] = '\0';
67}

81size_t Print::_print_num( long n, int base )
82{
83 char buf[ 34 ];
84 if ( base == DEC )
85 snprintf( buf, sizeof(buf), "%ld", n );
86 else if ( base == HEX )
87 snprintf( buf, sizeof(buf), "%lx", (unsigned long)n );
88 else if ( base == OCT )
89 snprintf( buf, sizeof(buf), "%lo", (unsigned long)n );
90 else if ( n < 0 )
91 {
92 size_t bytes = write( '-' );
93 _utoa_radix( (unsigned long long)( -(long long)n ), base, buf, sizeof(buf) );
94 return bytes + print( buf );
95 }
96 else
97 _utoa_radix( (unsigned long long)n, base, buf, sizeof(buf) );
98 return print( buf );
99}

200size_t Print::_print_double( double val, int digits )
201{
202 size_t bytes = 0;
203
204 if ( val < 0.0 ) { bytes += write( '-' ); val = -val; }
205
206 long integer = (long)val;
207 double frac = val - integer;
208 char buf[ 32 ];
209 snprintf( buf, sizeof(buf), "%ld.", integer );
210 bytes += print( buf );
211
212 for ( int i = 0; i < digits; i++ )
213 {
214 frac *= 10;
215 int d = (int)frac;
216 bytes += write( (uint8_t)('0' + d) );
217 frac -= d;
218 }
219
220 return bytes;
221}
