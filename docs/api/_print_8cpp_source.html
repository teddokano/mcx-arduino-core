39/*
40 * snprintf()/printf() have no format specifier for arbitrary radixes (no
41 * %b for binary, and nothing at all for bases other than 8/10/16), so
42 * BIN -- and any other non-DEC/HEX/OCT base a sketch might pass, matching
43 * real Arduino's support for any radix 2-36 -- has to be hand-converted.
44 */
45void _utoa_radix( unsigned long long value, int base, char *buf, size_t bufsize )
46{
47 static const char digits[] = "0123456789abcdefghijklmnopqrstuvwxyz";
48
49 if ( base < 2 || base > 36 )
50 base = 10;
51
52 char tmp[ 66 ];
53 int i = 0;
54
55 if ( value == 0 )
56 tmp[ i++ ] = '0';
57
58 while ( value > 0 && i < (int)sizeof(tmp) - 1 )
59 {
60 tmp[ i++ ] = digits[ value % (unsigned)base ];
61 value /= (unsigned)base;
62 }
63
64 size_t j = 0;
65 while ( i > 0 && j < bufsize - 1 )
66 buf[ j++ ] = tmp[ --i ];
67 buf[ j ] = '\0';
68}

82size_t Print::_print_num( long n, int base )
83{
84 char buf[ 34 ];
85 if ( base == DEC )
86 snprintf( buf, sizeof(buf), "%ld", n );
87 else if ( base == HEX )
88 snprintf( buf, sizeof(buf), "%lx", (unsigned long)n );
89 else if ( base == OCT )
90 snprintf( buf, sizeof(buf), "%lo", (unsigned long)n );
91 else if ( n < 0 )
92 {
93 size_t bytes = write( '-' );
94 _utoa_radix( (unsigned long long)( -(long long)n ), base, buf, sizeof(buf) );
95 return bytes + print( buf );
96 }
97 else
98 _utoa_radix( (unsigned long long)n, base, buf, sizeof(buf) );
99 return print( buf );
100}

201// Written to give the same output as ArduinoCore-avr/ArduinoCore-API's
202// printFloat(), not copied from it: rounded to `digits` places by adding
203// half of the last place before truncating, no decimal point when digits is
204// 0, and "nan"/"inf"/"ovf" where the integer part won't fit 32 bits.
205size_t Print::_print_double( double val, int digits )
206{
207 if ( std::isnan( val ) )
208 return print( "nan" );
209 if ( std::isinf( val ) )
210 return print( "inf" );
211 if ( val > 4294967040.0 || val < -4294967040.0 )
212 return print( "ovf" );
213
214 if ( digits < 0 )
215 digits = 0;
216
217 size_t bytes = 0;
218
219 if ( val < 0.0 ) { bytes += write( '-' ); val = -val; }
220
221 double rounding = 0.5;
222 for ( int i = 0; i < digits; i++ )
223 rounding /= 10.0;
224 val += rounding;
225
226 unsigned long integer = (unsigned long)val;
227 double frac = val - (double)integer;
228 bytes += print( integer );
229
230 if ( digits > 0 )
231 bytes += write( '.' );
232
233 for ( int i = 0; i < digits; i++ )
234 {
235 frac *= 10;
236 int d = (int)frac;
237 bytes += write( (uint8_t)('0' + d) );
238 frac -= d;
239 }
240
241 return bytes;
242}
