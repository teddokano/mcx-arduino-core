297 // cs defaults to manual/GPIO control, never the LPSPI hardware PCS
298 // function: one physical SPI bus is routinely shared by several
299 // devices (e.g. an LCD + SD card), each with its own independently
300 // sketch-managed CS pin via pinMode()/digitalWrite(). If this pin were
301 // hardware-PCS-driven instead, LPSPI would auto-pulse it on *every*
302 // SPI.transfer() call for *any* device sharing the bus -- not just
303 // transfers meant for the device on this pin -- corrupting whichever
304 // device thinks it's currently deselected.
305 cs_manual_control( true );
306 manual_cs_control = true;
307
308 // chip_select's own DigitalInOut constructor self-registered it as a
309 // generic "GPIO" pin owner, and cs_manual_control() above just did it
310 // again via DigitalInOut::pin_mux() (which keeps the registry's
311 // wanted-ALT in sync -- see that method). Undo both here, now that
312 // nothing else in this constructor still needs to call pin_mux() on
313 // chip_select: CS is deliberately left under the sketch's own
314 // pinMode()/digitalWrite() control (see the "cs defaults to manual/
315 // GPIO control" comment above), so a sketch following the documented
316 // `pinMode(SS,...); SPI.begin();` pattern always ends up with two
317 // independent DigitalInOut objects on the same physical pin --
318 // functionally harmless (both read/write the same GPIO register), but
319 // reported as a false-positive CONFLICT by pin-ownership debug
320 // tooling. This member exists only for cs_manual_control()'s internal
321 // bookkeeping, not to compete with the sketch for pin ownership. Must
322 // stay the last statement in this constructor: anything added below
323 // that calls chip_select.pin_mux() again would silently re-register it.

335// frequency()/mode()/bit_order() reconfigure the peripheral in place --
336// disable, poke just the registers that actually need to change, re-enable
337// -- instead of a full LPSPI_Deinit()+LPSPI_MasterInit() (which gates the
338// peripheral clock off/on and rebuilds every register, CFGR1/FIFO
339// watermarks/dummy-data included, from scratch). A sketch that interleaves
340// two devices at different clocks/modes on one bus (e.g. an LCD + an SD
341// card sharing MOSI/MISO/SCK with separate CS pins) calls these on *every*
342// beginTransaction() where settings differ from the last transaction, so
343// the old full-reinit cost was paid on every single alternation -- this is
344// the fix for issue #2.
345void SPI::frequency( uint32_t frequency )
346{
347 masterConfig.baudRate = frequency;
348
349 masterConfig.pcsToSckDelayInNanoSec = 1000000000U / (masterConfig.baudRate * 2U);
350 masterConfig.lastSckToPcsDelayInNanoSec = 1000000000U / (masterConfig.baudRate * 2U);
351 masterConfig.betweenTransferDelayInNanoSec = 1000000000U / (masterConfig.baudRate * 2U);
352
353 uint32_t tcr_prescale;
354
355 LPSPI_Enable( unit_base, false );
356
357 LPSPI_MasterSetBaudRate( unit_base, masterConfig.baudRate, master_clk_freq, &tcr_prescale );
358 unit_base->TCR = (unit_base->TCR & ~LPSPI_TCR_PRESCALE_MASK) | LPSPI_TCR_PRESCALE( tcr_prescale );
359
360 LPSPI_Enable( unit_base, true );
361
362 // delay times depend on the (now-updated) TCR prescale field, so these
363 // must come after it's written -- matching LPSPI_MasterInit()'s own
364 // ordering, which also sets them after enabling
365 LPSPI_MasterSetDelayTimes( unit_base, masterConfig.pcsToSckDelayInNanoSec, kLPSPI_PcsToSck, master_clk_freq );
366 LPSPI_MasterSetDelayTimes( unit_base, masterConfig.lastSckToPcsDelayInNanoSec, kLPSPI_LastSckToPcs, master_clk_freq );
367 LPSPI_MasterSetDelayTimes( unit_base, masterConfig.betweenTransferDelayInNanoSec, kLPSPI_BetweenTransfer, master_clk_freq );
368}

402uint8_t SPI::transfer_byte( uint8_t out )
403{
404 // Fast path for a single scalar byte, bypassing write()/
405 // LPSPI_MasterTransferBlocking()'s per-call disable+flush-FIFO+
406 // clear-status-flags+re-enable dance -- measured at ~40us of fixed
407 // overhead per call on real hardware, versus ~2us of actual bit-
408 // clocking at 4MHz. That overhead is negligible for the occasional
409 // bulk transfer (e.g. this project's own LCD writePixels() bursts),
410 // but dominates when a caller moves data one byte at a time in a
411 // tight loop -- as the standard Arduino SD library's spiRec()/
412 // spiSend() do (measured as the actual cause of "very slow" SD-backed
413 // rendering, once the interleaved-reinit issue #2 was independently
414 // fixed and made no difference).
415 //
416 // TCR's dynamic per-transfer bits (PCS/CONT/CONTC/RXMSK/TXMSK) are
417 // rewritten to a fixed one-shot/non-continuous/TX+RX-both-active
418 // state on every call -- cheap (no FIFO-empty wait needed, since the
419 // peripheral is always left idle between calls here, unlike write()'s
420 // multi-byte bursts) -- then the byte is pushed/popped directly via
421 // TDR/RDR, polling the same TDF/RDF flags the SDK's own blocking
422 // transfer waits on internally. CPOL/CPHA/PRESCALE/LSBF/FRAMESZ are
423 // untouched here -- they're the sticky fields frequency()/mode()/
424 // bit_order() already maintain.
425 uint32_t whichPcs = (master_pcs_4_xfer & LPSPI_MASTER_PCS_MASK) >> LPSPI_MASTER_PCS_SHIFT;
426
427 unit_base->TCR = (unit_base->TCR & ~(LPSPI_TCR_CONT_MASK | LPSPI_TCR_CONTC_MASK | LPSPI_TCR_RXMSK_MASK |
428 LPSPI_TCR_TXMSK_MASK | LPSPI_TCR_PCS_MASK))
429 | LPSPI_TCR_PCS( whichPcs );
430
431 while ( 0U == ( LPSPI_GetStatusFlags( unit_base ) & (uint32_t)kLPSPI_TxDataRequestFlag ) )
432 ;
433 LPSPI_WriteData( unit_base, out );
434
435 while ( 0U == ( LPSPI_GetStatusFlags( unit_base ) & (uint32_t)kLPSPI_RxDataReadyFlag ) )
436 ;
437
438 return (uint8_t)LPSPI_ReadData( unit_base );
439}
