94double Stream::_parseNumber( bool allow_decimal )
95{
96 bool negative = false;
97 double value = 0;
98 double frac = 0;
99 double frac_scale = 1;
100 bool in_frac = false;
101
102 int c = _timed_peek();
103
104 // discard anything that isn't the start of a number
105 while ( c >= 0 && c != '-' && ( c < '0' || c > '9' ) && !( allow_decimal && c == '.' ) )
106 {
107 read();
108 c = _timed_peek();
109 }
110
111 if ( c == '-' )
112 {
113 negative = true;
114 read();
115 c = _timed_peek();
116 }
117
118 while ( c >= 0 )
119 {
120 if ( c >= '0' && c <= '9' )
121 {
122 if ( in_frac )
123 {
124 frac_scale *= 0.1;
125 frac += ( c - '0' ) * frac_scale;
126 }
127 else
128 {
129 value = value * 10 + ( c - '0' );
130 }
131 read();
132 }
133 else if ( allow_decimal && c == '.' && !in_frac )
134 {
135 in_frac = true;
136 read();
137 }
138 else
139 {
140 break;
141 }
142
143 c = _timed_peek();
144 }
145
146 value += frac;
147 return negative ? -value : value;
148}
