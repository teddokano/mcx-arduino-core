56void AnalogIn::resolve_pin( int pin )
57{
58 for ( size_t i = 0; i < sizeof( s_pins ) / sizeof( s_pins[0] ); i++ )
59 {
60 if ( s_pins[ i ].pin == pin )
61 {
62 _input_positive = (uint8_t)s_pins[ i ].input_positive;
63 _channel_id = (uint8_t)( i + 1 ); // dedicated LPADC CMD1..CMD4
64 _pin = pin;
65
66 port_pin_config_t cfg = {
67 kPORT_PullDisable,
68 kPORT_LowPullResistor,
69 kPORT_FastSlewRate,
70 kPORT_PassiveFilterDisable,
71 kPORT_OpenDrainDisable,
72 kPORT_LowDriveStrength,
73 kPORT_NormalDriveStrength,
74 kPORT_MuxAlt0, // ADC0_An
75 kPORT_InputBufferDisable, // required for analog function
76 kPORT_InputNormal,
77 kPORT_UnlockRegister
78 };
79 PORT_SetPinConfig( s_pins[ i ].port, s_pins[ i ].port_pin, &cfg );
80 return;
81 }
82 }

84 _input_positive = 0;
85 _channel_id = 0;
86 _pin = -1;

89void AnalogIn::_acquire_peripheral( void )
90{
91 if ( _instance_count == 0 )
92 {
93 CLOCK_EnableClock( kCLOCK_GateADC0 );
94 RESET_ReleasePeripheralReset( kADC0_RST_SHIFT_RSTn );
95
96 // ADCK (conversion clock) source — without this, LPADC_Init()
97 // succeeds but no conversion/calibration cycle ever completes and
98 // LPADC_DoAutoCalibration() spins forever polling GCC[0].RDY.
99 CLOCK_SetClockDiv( kCLOCK_DivADC0, 1u );
100 CLOCK_AttachClk( kFRO12M_to_ADC0 );
101
102 lpadc_config_t cfg;
103 LPADC_GetDefaultConfig( &cfg );
104 cfg.enableAnalogPreliminary = true;
105 cfg.referenceVoltageSource = kLPADC_ReferenceVoltageAlt3; // VDDA (FRDM-MCXA153)
106 cfg.powerLevelMode = kLPADC_PowerLevelAlt4; // highest accuracy
107 LPADC_Init( ADC0, &cfg );
108
109#if defined( FSL_FEATURE_LPADC_HAS_CTRL_CAL_REQ ) && FSL_FEATURE_LPADC_HAS_CTRL_CAL_REQ
110 if ( !_calibrated )
111 {
112 LPADC_DoAutoCalibration( ADC0 );
113 _calibrated = true;
114 }
115#endif
116 }
117
118 _instance_count++;
119}

121void AnalogIn::_release_peripheral( void )
122{
123 _instance_count--;
124
125 if ( _instance_count == 0 )
126 {
127 LPADC_Deinit( ADC0 );

165uint16_t AnalogIn::sample_raw( void )
166{
167 if ( -1 == _pin )
168 return 0;
169
170 lpadc_conv_trigger_config_t trig;
171 LPADC_GetDefaultConvTriggerConfig( &trig );
172 trig.targetCommandId = _channel_id;
173 trig.enableHardwareTrigger = false;
174 LPADC_SetConvTriggerConfig( ADC0, 0u, &trig );
175
176 LPADC_DoResetFIFO( ADC0 );
177 LPADC_DoSoftwareTrigger( ADC0, 1u ); // trigger0 mask
178
179 lpadc_conv_result_t result;
180 while ( !LPADC_GetConvResult( ADC0, &result ) )
181 ;
182
183 return (uint16_t)( ( result.convValue >> 3u ) & 0x0FFFu );
184}

230void AnalogIn::resolve_pin( int pin )
231{
232 for ( size_t i = 0; i < sizeof( s_pins ) / sizeof( s_pins[0] ); i++ )
233 {
234 if ( s_pins[ i ].pin == pin )
235 {
236 _channel_number = (uint8_t)s_pins[ i ].channel_number;
237 _side = s_pins[ i ].side;
238 _channel_id = (uint8_t)( i + 1 ); // dedicated LPADC CMD1..CMD4
239 _pin = pin;
240
241 // Named-field init, not positional: port_pin_config_t is a
242 // bitfield struct whose field set depends on which
243 // FSL_FEATURE_PORT_HAS_* macros this chip defines (N947 has no
244 // driveStrength1 field, unlike some other MCX parts), so
245 // positional aggregate init is fragile across chips.
246 port_pin_config_t cfg = {};
247 cfg.pullSelect = kPORT_PullDisable;
248 cfg.slewRate = kPORT_FastSlewRate;
249 cfg.passiveFilterEnable = kPORT_PassiveFilterDisable;
250 cfg.openDrainEnable = kPORT_OpenDrainDisable;
251 cfg.driveStrength = kPORT_LowDriveStrength;
252 cfg.mux = kPORT_MuxAlt0; // ADC0_Bxx/ADC0_Axx
253 cfg.inputBuffer = kPORT_InputBufferDisable; // required for analog function
254 cfg.lockRegister = kPORT_UnlockRegister;
255 PORT_SetPinConfig( s_pins[ i ].port, s_pins[ i ].port_pin, &cfg );
256 return;
257 }
258 }

260 _channel_number = 0;
261 _channel_id = 0;
262 _pin = -1;

265void AnalogIn::_acquire_peripheral( void )
266{
267 if ( _instance_count == 0 )
268 {
269 CLOCK_EnableClock( kCLOCK_Adc0 );
270 RESET_ReleasePeripheralReset( kADC0_RST_SHIFT_RSTn );
271
272 // ADCK (conversion clock) source -- same requirement as A153: without
273 // this LPADC_Init() succeeds but no conversion/calibration cycle ever
274 // completes.
275 CLOCK_SetClkDiv( kCLOCK_DivAdc0Clk, 1u );
276 CLOCK_AttachClk( kFRO_HF_to_ADC0 );
277
278 // N947-specific: the VREF block supplies LPADC's bias current on this
279 // chip (A153 didn't need this) -- must be enabled/initialized before
280 // LPADC_Init(), per NXP's own FRDM-MCXN947 LPADC polling example.
281 SPC_EnableActiveModeAnalogModules( SPC0, kSPC_controlVref );
282
283 vref_config_t vref_cfg;
284 VREF_GetDefaultConfig( &vref_cfg );
285 vref_cfg.bufferMode = kVREF_ModeBandgapOnly;
286 VREF_Init( VREF0, &vref_cfg );
287
288 lpadc_config_t cfg;
289 LPADC_GetDefaultConfig( &cfg );
290 cfg.enableAnalogPreliminary = true;
291 cfg.referenceVoltageSource = kLPADC_ReferenceVoltageAlt3; // VDDA
292 LPADC_Init( ADC0, &cfg );
293
294#if defined( FSL_FEATURE_LPADC_HAS_CTRL_CAL_REQ ) && FSL_FEATURE_LPADC_HAS_CTRL_CAL_REQ
295 if ( !_calibrated )
296 {
297 LPADC_DoAutoCalibration( ADC0 );
298 _calibrated = true;
299 }
300#endif
301 }
302
303 _instance_count++;
