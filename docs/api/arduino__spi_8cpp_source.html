65uint16_t SPIClass::transfer16( uint16_t data )
66{
67 union { uint16_t val; struct { uint8_t lo; uint8_t hi; } b; } t;
68
69 t.val = data;
70
71 // MSBFIRST: high byte goes out first on the wire; LSBFIRST: low byte first
72 if ( _last_order == MSBFIRST )
73 {
74 t.b.hi = transfer( t.b.hi );
75 t.b.lo = transfer( t.b.lo );
76 }
77 else
78 {
79 t.b.lo = transfer( t.b.lo );
80 t.b.hi = transfer( t.b.hi );
81 }
82
83 return t.val;
84}

109void SPIClass::setClockDivider( uint8_t divider )
110{
111 if ( !spi )
112 begin();
113
114 uint32_t div;
115
116 switch ( divider )
117 {
118 case SPI_CLOCK_DIV2: div = 2; break;
119 case SPI_CLOCK_DIV4: div = 4; break;
120 case SPI_CLOCK_DIV8: div = 8; break;
121 case SPI_CLOCK_DIV16: div = 16; break;
122 case SPI_CLOCK_DIV32: div = 32; break;
123 case SPI_CLOCK_DIV64: div = 64; break;
124 case SPI_CLOCK_DIV128: div = 128; break;
125 default: div = 4; break;
126 }
127
128 uint32_t freq = spi->clock_freq() / div;
129
130 spi->frequency( freq );
131 _last_clock = freq;
132}

134void SPIClass::txrx( uint8_t *data, size_t size )
135{
136 // Chunked through a fixed-size stack buffer rather than a single
137 // size-sized one -- a caller-supplied count has no compile-time bound,
138 // and CS stays under the sketch's own digitalWrite() control across
139 // this whole call (see SPIClass's header comment), so splitting into
140 // multiple back-to-back r01lib write() calls is transparent to the bus.
141 static constexpr size_t READ_BUFFER_SIZE = 128;
142
143 uint8_t r_data[ READ_BUFFER_SIZE ];
144
145 while ( size > 0 )
146 {
147 size_t chunk = size < READ_BUFFER_SIZE ? size : READ_BUFFER_SIZE;
148
149 spi->write( data, r_data, (int)chunk );
150 memcpy( data, r_data, chunk );
151
152 data += chunk;
153 size -= chunk;
154 }
155}
