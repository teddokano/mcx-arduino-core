15static err_t on_recv( void *arg, struct tcp_pcb *pcb, struct pbuf *p, err_t err )
16{
17 (void)pcb;
18 (void)err;
19 TcpSlot *s = (TcpSlot *)arg;
20
21 if ( !p )
22 {
23 s->state = TCP_PEER_CLOSED;
24 return ERR_OK;
25 }
26
27 // copy out of the driver's receive buffer, so data the sketch hasn't read doesn't hold it
28 struct pbuf *q = pbuf_clone( PBUF_RAW, PBUF_RAM, p );
29 if ( !q )
30 return ERR_MEM; // lwIP keeps p and offers it again
31 pbuf_free( p );
32
33 if ( s->rx )
34 pbuf_cat( s->rx, q );
35 else
36 s->rx = q;
37 return ERR_OK;
38}

58int tcp_slot_take( struct tcp_pcb *pcb, TcpState state, uint16_t server_port )
59{
60 for ( int i = 0; i < MAX_SOCK_NUM; i++ )
61 {
62 TcpSlot *s = &tcp_slots[ i ];
63 if ( s->state != TCP_FREE )
64 continue;
65
66 s->pcb = pcb;
67 s->rx = NULL;
68 s->state = state;
69 s->server_port = server_port;
70 s->remote_ip = pcb->remote_ip;
71 s->remote_port = pcb->remote_port;
72 s->local_port = pcb->local_port;
73
74 tcp_arg( pcb, s );
75 tcp_recv( pcb, on_recv );
76 tcp_err( pcb, on_err );
77 tcp_nagle_disable( pcb ); // send each write() at once, as a W5x00 does
78 return i;
79 }
80 return -1;
81}

83void tcp_slot_release( int sock )
84{
85 TcpSlot *s = &tcp_slots[ sock ];
86
87 if ( s->pcb )
88 {
89 // take unread data off the window first: lwIP answers a close with data
90 // the sketch hasn't read with a reset, where a W5x00 closes gracefully
91 if ( s->rx )
92 tcp_recved( s->pcb, s->rx->tot_len );
93
94 tcp_arg( s->pcb, NULL );
95 tcp_recv( s->pcb, NULL );
96 tcp_err( s->pcb, NULL );
97 if ( s->state == TCP_CONNECTING || tcp_close( s->pcb ) != ERR_OK )
98 tcp_abort( s->pcb );
99 s->pcb = NULL;
100 }
101 if ( s->rx )
102 {
103 pbuf_free( s->rx );
104 s->rx = NULL;
105 }
106 s->state = TCP_FREE;
107}

114uint8_t EthernetClient::status()
115{
116 if ( _sock < 0 )
117 return 0x00; // CLOSED
118
119 switch ( tcp_slots[ _sock ].state )
120 {
121 case TCP_CONNECTING: return 0x15; // SYNSENT
122 case TCP_CONNECTED: return 0x17; // ESTABLISHED
123 case TCP_PEER_CLOSED: return 0x1C; // CLOSE_WAIT
124 default: return 0x00;
125 }
126}

137int EthernetClient::connect( IPAddress ip, uint16_t port )
138{
139 if ( !started() || (uint32_t)ip == 0 || (uint32_t)ip == 0xFFFFFFFF )
140 return 0;
141
142 if ( _sock >= 0 )
143 stop();
144
145 {
146 Lock lock;
147 struct tcp_pcb *pcb = tcp_new();
148 if ( !pcb )
149 return 0;
150
151 ip_addr_t a = to_lwip( ip );
152 _sock = tcp_slot_take( pcb, TCP_CONNECTING, 0 );
153 if ( _sock < 0 )
154 {
155 tcp_abort( pcb );
156 return 0;
157 }
158
159 if ( tcp_connect( pcb, &a, port, on_connected ) != ERR_OK )
160 {
161 tcp_slot_release( _sock );
162 _sock = -1;
163 return 0;
164 }
165 tcp_slots[ _sock ].remote_ip = a;
166 tcp_slots[ _sock ].remote_port = port;
167 tcp_slots[ _sock ].local_port = pcb->local_port;
168 }
169
170 uint32_t start = millis();
171 while ( tcp_slots[ _sock ].state == TCP_CONNECTING && millis() - start < _timeout )
172 idle();
173
174 if ( tcp_slots[ _sock ].state == TCP_CONNECTED )
175 return 1;
176
177 Lock lock;
178 tcp_slot_release( _sock );
179 _sock = -1;
180 return 0;
181}

198size_t EthernetClient::write( const uint8_t *buf, size_t size )
199{
200 if ( _sock < 0 )
201 {
202 setWriteError();
203 return 0;
204 }
205
206 // as the official library, wait for room as long as the connection is open, with no
207 // time limit; the peer having sent its FIN (CLOSE_WAIT) still takes data
208 size_t sent = 0;
209
210 while ( sent < size )
211 {
212 size_t n = 0;
213 {
214 Lock lock;
215 TcpSlot *s = &tcp_slots[ _sock ];
216 if ( !s->pcb || ( s->state != TCP_CONNECTED && s->state != TCP_PEER_CLOSED ) )
217 break;
218
219 n = min( (size_t)tcp_sndbuf( s->pcb ), size - sent );
220 n = min( n, (size_t)0xFFFF );
221 if ( n && tcp_write( s->pcb, buf + sent, n, TCP_WRITE_FLAG_COPY ) != ERR_OK )
222 n = 0; // out of segments: wait for ACKs as for a full buffer
223 tcp_output( s->pcb );
224 }
225
226 if ( n )
227 sent += n;
228 else
229 idle();
230 }
231
232 if ( sent < size )
233 setWriteError();
234 return sent;
235}

253int EthernetClient::read( uint8_t *buf, size_t size )
254{
255 if ( _sock < 0 )
256 return -1;
257
258 Lock lock;
259 TcpSlot *s = &tcp_slots[ _sock ];
260 if ( !s->rx )
261 return -1;
262
263 uint16_t n = pbuf_copy_partial( s->rx, buf, (uint16_t)min( size, (size_t)s->rx->tot_len ), 0 );
264 s->rx = pbuf_free_header( s->rx, n );
265 if ( s->pcb )
266 tcp_recved( s->pcb, n );
267 return n;
