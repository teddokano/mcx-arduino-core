21/*
22 * PROGMEM / F() / pgm_read_*() compatibility -- no-ops on this Cortex-M
23 * target. AVR is a Harvard-architecture chip where flash and RAM are
24 * separate address spaces, so PROGMEM-tagged data needs special
25 * instructions (pgm_read_byte() etc.) to read back; on this von-Neumann
26 * Cortex-M, flash is just regular memory-mapped, readable like anything
27 * else. These are declared purely so AVR-era sketches/libraries that use
28 * them still compile -- __FlashStringHelper is forward-declared here
29 * (before arduino_string.h/arduino_serial.h) since String and SerialClass
30 * both need to know the type exists for their F()-string overloads.
31 */
37#define PROGMEM
38#define PGM_P const char *
40#define PSTR( s ) ( s )
41#define pgm_read_byte( addr ) ( *(const unsigned char *)(addr) )
42#define pgm_read_word( addr ) ( *(const unsigned short *)(addr) )
43#define pgm_read_dword( addr ) ( *(const unsigned long *)(addr) )
44#define pgm_read_float( addr ) ( *(const float *)(addr) )
45#define pgm_read_ptr( addr ) ( *(const void * const *)(addr) )

106/*
107 * Real Arduino declares LSBFIRST/MSBFIRST as enumerators of an actual
108 * `BitOrder` enum type (api/Common.h), so any type named BitOrder is
109 * automatically usable. This project defines LSBFIRST/MSBFIRST as plain
110 * #define macros instead (established earlier -- see arduino_spi.h's
111 * `enum endian`), so a real `enum BitOrder { LSBFIRST, MSBFIRST }` can't
112 * be declared here too: the preprocessor would substitute those tokens
113 * away before the compiler ever saw them as enumerator names. BitOrder
114 * is provided as a plain integer typedef instead, just so the TYPE NAME
115 * exists for libraries (e.g. Adafruit BusIO's `typedef BitOrder
116 * BusIOBitOrder;`) that expect the core to declare it.
117 */
118typedef uint8_t BitOrder;

120// Clock-cycle conversion macros (matches UNO R3/R4's Arduino.h). Each board
121// in this core always runs its core clock at a fixed frequency (no
122// user-selectable F_CPU like AVR), supplied on the command line via
123// platform.txt's compiler.defines (-DF_CPU={build.f_cpu}, from boards.txt)
124// since it differs per board (e.g. 96MHz on FRDM-MCXA153, 150MHz on
125// FRDM-MCXN947). Falls back to the FRDM-MCXA153 value if not supplied.
126#ifndef F_CPU
127#define F_CPU 96000000UL
128#endif
129#define clockCyclesPerMicrosecond() ( F_CPU / 1000000UL )
130#define clockCyclesToMicroseconds( a ) ( (a) / clockCyclesPerMicrosecond() )
131#define microsecondsToClockCycles( a ) ( (a) * clockCyclesPerMicrosecond() )
