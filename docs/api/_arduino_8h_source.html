23/*
24 * PROGMEM / F() / pgm_read_*() compatibility -- no-ops on this Cortex-M
25 * target. AVR is a Harvard-architecture chip where flash and RAM are
26 * separate address spaces, so PROGMEM-tagged data needs special
27 * instructions (pgm_read_byte() etc.) to read back; on this von-Neumann
28 * Cortex-M, flash is just regular memory-mapped, readable like anything
29 * else. These are declared purely so AVR-era sketches/libraries that use
30 * them still compile -- __FlashStringHelper is forward-declared here
31 * (before arduino_string.h/arduino_serial.h) since String and SerialClass
32 * both need to know the type exists for their F()-string overloads.
33 */

42#define PSTR( s ) ( s )

46#define pgm_read_float( addr ) ( *(const float *)(addr) )

106// Clock-cycle conversion macros (matches UNO R3/R4's Arduino.h). Each board
107// in this core always runs its core clock at a fixed frequency (no
108// user-selectable F_CPU like AVR), supplied on the command line via
109// platform.txt's compiler.defines (-DF_CPU={build.f_cpu}, from boards.txt)
